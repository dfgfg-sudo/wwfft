    console.log('接收到插件生成请求:', params);

    if (!params || typeof params !== 'object') {
      throw new Error('参数必须是有效的JSON对象');
    }

    const userRequirement = params.plugin_requirement || params.request || params.query || '创建一个基本的Coze插件';

    const pluginCode = generatePluginCode(userRequirement);
    const pluginConfig = generatePluginConfig(userRequirement);

    return {
      code: 0,
      message: '成功生成Coze插件代码',
      data: {
        pluginName: pluginConfig.pluginName,
        pluginDescription: pluginConfig.pluginDescription,
        code: pluginCode,
        config: pluginConfig,
        usageInstructions: '将此代码复制到Coze IDE中，或使用配置导入功能直接创建插件',
        generationTime: new Date().toISOString()
      }
    };
  } catch (error) {
    console.error('插件生成失败:', error);
    return {
      code: -1,
      message: `生成失败: ${error instanceof Error ? error.message : String(error)}`,
      data: null
    };
  }
}

const validateInputParams = (params) => {
  if (params.plugin_name === undefined || params.plugin_name === null) {
    return {
      code: "INVALID_INPUT",
      message: "缺少必选参数：plugin_name字段",
      suggestion: "请确保传递了plugin_name字段，键名必须完全匹配"
    };
  }

  if (params.plugin_name.trim() === "") {
    return {
      code: "INVALID_INPUT",
      message: "插件名称不能为空",
      suggestion: "请为plugin_name字段提供有效的插件名称"
    };
  }

  if (!params.function_desc?.trim()) {
    return {
      code: "INVALID_INPUT",
      message: "缺少必选参数：功能描述",
      suggestion: "请为function_desc字段提供插件功能描述"
    };
  }

  if (!params.plugin_spec?.trim()) {
    return {
      code: "INVALID_INPUT",
      message: "缺少必选参数：合并参数",
      suggestion: "请为plugin_spec字段提供参数说明和错误处理规则"
    };
  }

  return null;
};

const extractPluginFeatures = (params) => {
  const name = params.plugin_name.toLowerCase().replace(/\s+/g, "_").replace(/[^\w]/g, "");
  const permissions = params.permissions?.trim()
    ? params.permissions.split(/,|、/).map(p => p.trim().toLowerCase())
    : ["network"];
  const platforms = params.platforms?.trim()
    ? params.platforms.split(/,|、/).map(p => `coze-${p.trim().toLowerCase()}`)
    : ["coze-web", "coze-mobile"];

  const specLines = params.plugin_spec.split('\n').filter(line => line.trim());
  const parameters = [];
  const errorTypes = [];

  specLines.forEach(line => {
    if (line.startsWith('-') && line.includes('（') && line.includes('）')) {
      const nameMatch = line.match(/^-\s*(.*?)\s*\(/);
      const typeMatch = line.match(/（(字符串|数字|布尔|枚举值)/);
      const required = line.includes("必选");
      const enumMatch = line.match(/枚举值：(.*?)[，,）]/);
      const descMatch = line.match(/，(.*?)[）]/);

      if (nameMatch) {
        parameters.push({
          name: nameMatch[1].trim().replace(/\s+/g, "_"),
          type: typeMatch?.[1] || "string",
          required: required,
          enum: enumMatch?.[1]?.split("/").map(v => v.trim()) || [],
          description: descMatch?.[1]?.trim() || "无描述"
        });
      }
    } else if (line.includes('：')) {
      const [scene, msg] = line.split('：').map(item => item.trim());
      if (scene && msg) {
        const codeMap = {
          "参数": "INVALID_PARAM",
          "网络": "NETWORK_ERROR",
          "超时": "TIMEOUT",
          "权限": "PERMISSION_DENIED"
        };
        const code = codeMap[scene] || "UNKNOWN_ERROR";

        errorTypes.push({
          code,
          message: scene,
          defaultSuggestion: msg
        });
      }
    }
  });

  if (errorTypes.length === 0) {
    errorTypes.push(
      { code: "INVALID_PARAM", message: "参数无效", defaultSuggestion: "检查参数格式" },
      { code: "ACTION_FAILED", message: "操作失败", defaultSuggestion: "请稍后重试" }
    );
  }

  const actions = ["execute"];

  return {
    meta: {
      ...PluginMeta,
      name,
      description: params.function_desc.trim(),
      requiredPermissions: permissions,
      compatiblePlatforms: platforms,
      author: "Auto Generated"
    },
    parameters,
    errorTypes,
    actions
  };
};

      })

    required: param.required,

    description: param.description,
    placeholder: param.required ? `请输入${param.description}` : undefined
  }));

  const uiParameters = [
    {
      name: "action",
      label: "动作类型",
      type: "select",
      required: true,
      options: features.actions.map(a => ({ label: a, value: a })),
      description: "选择要执行的动作"

  const uiConfigCode = `export const UIConfig = {

  return `/**

export async function generateHandler(params) {
  try {
    const inputParams = params?.parameters || {};

    const validationError = validateInputParams(inputParams);
    if (validationError) {
      return { success: false, error: validationError };
    }
    const validParams = inputParams;

    const features = extractPluginFeatures(validParams);
    const pluginCode = generatePluginCode(features);

    return {
      success: true,
      data: {
        plugin_code: pluginCode,
        plugin_meta: features.meta
      }
    };
  } catch (error) {
    return {
      success: false,
      error: {
        code: "GENERATION_FAILED",
        message: `生成失败：${error.message}`,
        suggestion: "请检查输入格式是否正确"
      }
    };
  }
}

export async function workflowHandler(args) {
  try {
    const { workflow_json } = args.input;

    if (!workflow_json) {
      return { success: false, message: "缺少必填参数：workflow_json" };
    }

    let parsed;
    try {
      parsed = JSON.parse(workflow_json);
    } catch (parseErr) {
      return { success: false, message: `JSON解析失败：${parseErr.message}` };
    }

    const nodes = parsed.nodes || parsed.json?.nodes || [];
    const edges = parsed.edges || parsed.json?.edges || [];

    if (nodes.length === 0) {
      return { success: false, message: "JSON数据中未找到有效的节点(nodes)信息" };
    }
    if (edges.length === 0) {
      return { success: false, message: "JSON数据中未找到有效的连接(edges)信息" };
    }

    const workflowId = await CozeInternalTool.createWorkflow({
      name: `导入工作流_${new Date().getTime()}`
    });

    const nodeMap = {};

    for (const node of nodes) {
      const safeData = { ...node.data };
      if (safeData.inputs) {
        safeData.inputs = safeData.inputs.filter(
          (input) => {
            const isSensitiveName = ["apikey", "token", "secret", "password", "key"].some(
              key => (input.name || "").toLowerCase().includes(key)
            );
            const isSensitiveValue = typeof input.value === "string" &&
              ["sk-", "api_", "token=", "secret="].some(prefix =>
                input.value.toLowerCase().includes(prefix)
              );
            return !isSensitiveName && !isSensitiveValue;
          }
        );
      }

      const nodeIndex = nodes.indexOf(node);
      const position = node.meta?.position || {
        x: 100 + (nodeIndex % 5) * 200,
        y: 100 + Math.floor(nodeIndex / 5) * 150
      };

      const newNodeId = await CozeInternalTool.createNode(workflowId, {
        type: node.type || "default",
        name: node.data?.nodeMeta?.title || `节点_${node.id}`,
        position,
        data: safeData
      });
      nodeMap[node.id] = newNodeId;
    }

    let validEdges = 0;
    for (const edge of edges) {
      const sourceId = nodeMap[edge.sourceNodeID];
      const targetId = nodeMap[edge.targetNodeID];

      if (sourceId && targetId) {
        try {
          await CozeInternalTool.createEdge(workflowId, {
            sourceNodeId: sourceId,
            targetNodeId: targetId
          });
          validEdges++;
        } catch (edgeErr) {
          console.warn(`创建连接失败 ${edge.sourceNodeID} -> ${edge.targetNodeID}：${edgeErr.message}`);
        }
      } else {
        console.warn(`跳过无效连接: ${edge.sourceNodeID} -> ${edge.targetNodeID}`);
      }
    }

    return {
      success: true,
      message: `工作流导入成功，包含${nodes.length}个节点和${validEdges}个有效连接`,
      workflow_id: workflowId
    };

  } catch (err) {
    return {
      success: false,
      message: `导入失败：${err.message}`
    };
  }
}

 * @description 自动处理数据可视化、格式转换、文本处理、网页解析及OpenAPI/Swagger/cURL解析
 * @parameters {
 *   "task": {
 *     "type": "object",
 *     "required": true,
 *     "description": "任务信息，格式：{需求: '...', 数据: {...}, 选项: {...}}"
 *   }
 * }
 */
export async function smartProcessorHandler(event) {
  try {
    const { task } = event.input;
    const funcType = identifyTaskType(task.需求);

    switch (funcType) {
      case 'visualize': return handleVisualize(task);
      case 'format': return handleFormat(task);
      case 'md2html': return handleMd2Html(task);
      case 'web_parse': return handleWebParse(task);
      case 'api_parse': return handleApiParse(task);
      default: return { status: 'error', message: '未识别任务类型', result: null };
    }
  } catch (error) {
    return { status: 'error', message: `处理失败：${error.message}`, result: null };
  }
}

function identifyTaskType(需求) {
  const keywords = {
    visualize: ['图表', '可视化', '柱状图', '折线图'],
    format: ['转换', 'JSON', 'CSV', '格式'],
    md2html: ['Markdown', 'MD', 'HTML', '文本处理'],
    web_parse: ['网页', '解析', 'URL', '链接'],
    api_parse: ['OpenAPI', 'Swagger', 'cURL', 'API文档']
  };
  for (const [type, keys] of Object.entries(keywords)) {
    if (keys.some(key => 需求.includes(key))) return type;
  }
  return 'unknown';
}

async function handleVisualize(task) {
  const { datasources } = task.数据;
  const chartType = task.选项?.图表类型 || '折线图';
  const chartUrl = `https://coze-visualize.example.com?type=${chartType}&data=${JSON.stringify(datasources)}`;
  return { status: 'success', message: `生成${chartType}`, result: { chartUrl, 类型: chartType } };
}

async function handleFormat(task) {
  const { content } = task.数据;
  const target = task.选项?.目标格式 || 'csv';
  if (target === 'csv' && content.includes('{')) {
    const jsonData = JSON.parse(content);
    const headers = Object.keys(jsonData[0]);
    const csv = [headers.join(','), ...jsonData.map(item => headers.map(k => item[k]).join(','))].join('\n');
    return { status: 'success', message: 'JSON转CSV', result: { 内容: csv } };
  } else if (target === 'json' && content.includes(',')) {
    const [headers, ...rows] = content.split('\n').map(l => l.split(','));
    const json = JSON.stringify(rows.map(r => headers.reduce((o, k, i) => ({ ...o, [k]: r[i] }), {})), null, 2);
    return { status: 'success', message: 'CSV转JSON', result: { 内容: json } };
  }
  return { status: 'error', message: '不支持的转换', result: null };
}

async function handleMd2Html(task) {
  const html = task.数据.content.split('\n').map(line => {
    if (line.startsWith('# ')) return `<h1>${line.slice(2)}</h1>`;
    if (line.startsWith('## ')) return `<h2>${line.slice(3)}</h2>`;
    if (line.startsWith('**') && line.endsWith('**')) return `<strong>${line.slice(2, -2)}</strong>`;
    return `<p>${line}</p>`;
  }).join('\n');
  return { status: 'success', message: 'MD转HTML', result: { HTML内容: html } };
}

async function handleWebParse(task) {
  const { url } = task.数据;
  return { status: 'success', message: `解析${url}`, result: { 文本: '网页文本摘要...', 图片: ['https://example.com/img1.jpg'] } };
}

async function handleApiParse(task) {
  const { content } = task.数据;
  let type = '未知';
  if (content.includes('openapi') || content.includes('swagger')) type = 'OpenAPI/Swagger';
  else if (content.startsWith('curl')) type = 'cURL';
  return { status: 'success', message: `解析${type}`, result: { 类型: type, 摘要: 'API信息解析完成' } };
}

// ==================== Coze插件修复工具 ====================

/**
 * CozePluginRepairTool - Coze插件智能修复工具
 * 自动修复和合并多个Coze插件JSON代码，生成符合导入标准的完整插件定义
 */

export async function repairHandler({ input, logger }) {
  try {
    const {
      json_inputs,
      repair_mode = 'comprehensive',
      output_format = 'coze_import_ready',
      conflict_resolution = 'rename'
    } = input;

    logger.info(`CozePluginRepairTool开始处理，输入插件数量: ${json_inputs ? (Array.isArray(json_inputs) ? json_inputs.length : 1) : 0}`);

    if (!json_inputs) {
      return {
        success: false,
        error: "缺少必要参数：json_inputs",
        timestamp: new Date().toISOString(),
        message: "输入参数错误：请提供需要修复的插件JSON数据",
        repair_engine: "Coze Plugin Repair Engine v2.0"
      };
    }

    let inputsArray = [];

    if (typeof json_inputs === 'string') {
      try {
        const parsed = JSON.parse(json_inputs);
        inputsArray = Array.isArray(parsed) ? parsed : [parsed];
      } catch (e) {
        const repairEngine = new CozePluginRepairEngine();
        const fixedJson = repairEngine.fixJsonString(json_inputs);
        try {
          const parsed = JSON.parse(fixedJson);
          inputsArray = Array.isArray(parsed) ? parsed : [parsed];
        } catch {
          inputsArray = [{
            name: "repaired_from_string",
            version: "1.0.0",
            description: "从字符串自动修复的插件",
            nodes: [],
            edges: []
          }];
        }
      }
    } else if (Array.isArray(json_inputs)) {
      inputsArray = json_inputs;
    } else if (typeof json_inputs === 'object') {
      inputsArray = [json_inputs];
    } else {
      return {
        success: false,
        error: "不支持的输入格式",
        timestamp: new Date().toISOString(),
        message: "输入格式错误：json_inputs应为JSON字符串、对象或数组",
        repair_engine: "Coze Plugin Repair Engine v2.0"
      };
    }

    if (inputsArray.length === 0) {
      return {
        success: false,
        error: "输入数组为空",
        timestamp: new Date().toISOString(),
        message: "没有可处理的插件数据：输入数组为空",
        repair_engine: "Coze Plugin Repair Engine v2.0"
      };
    }

    const repairEngine = new CozePluginRepairEngine({
      repairMode: repair_mode,
      outputFormat: output_format,
      conflictResolution: conflict_resolution
    });

    const result = await repairEngine.processMultiplePlugins(inputsArray);

    return {
      success: true,
      merged_plugin: result.merged_plugin,
      repaired_plugins: result.repaired_plugins,
      formatted_output: result.formatted_output,
      statistics: result.statistics,
      details: result.details,
      timestamp: new Date().toISOString(),
      repair_engine: "Coze Plugin Repair Engine v2.0",
      message: `成功修复 ${result.statistics.repaired_count} 个插件，合并为完整Coze插件，可直接导入使用`
    };

  } catch (error) {
    logger.error(`修复过程出错: ${error.message}`, error);
    return {
      success: false,
      error: `自动修复过程出错: ${error.message}`,
      stack_trace: error.stack,
      timestamp: new Date().toISOString(),
      message: "插件自动修复失败，请检查输入数据格式",
      repair_engine: "Coze Plugin Repair Engine v2.0"
    };
  }
}

class CozePluginRepairEngine {
  constructor(options = {}) {
    this.repairMode = options.repairMode || 'comprehensive';
    this.outputFormat = options.outputFormat || 'coze_import_ready';
    this.conflictResolution = options.conflictResolution || 'rename';
    this.processedCount = 0;
    this.errorCount = 0;
    this.warningCount = 0;
  }

  async processMultiplePlugins(pluginInputs) {
    const startTime = Date.now();
    const results = {
      valid_plugins: [],
      repaired_plugins: [],
      errors: [],
      warnings: []
    };

    for (const input of pluginInputs) {
      try {
        const pluginResult = await this.repairSinglePlugin(input);

        if (pluginResult.success) {
          results.valid_plugins.push(pluginResult.plugin);
          if (pluginResult.repaired) {
            results.repaired_plugins.push(pluginResult.plugin);
          }
        } else {
          results.errors.push({
            input: input,
            error: pluginResult.error
          });
          this.errorCount++;
        }

        results.warnings.push(...(pluginResult.warnings || []));
        this.warningCount += pluginResult.warnings?.length || 0;
        this.processedCount++;

      } catch (error) {
        results.errors.push({
          input: input,
          error: error.message
        });
        this.errorCount++;
      }
    }

    const mergedResult = this.mergePlugins(results.valid_plugins);

    return {
      ...mergedResult,
      statistics: {
        processed_count: this.processedCount,
        valid_count: results.valid_plugins.length,
        repaired_count: results.repaired_plugins.length,
        error_count: this.errorCount,
        warning_count: this.warningCount,
        processing_time_ms: Date.now() - startTime
      },
      details: {
        errors: results.errors,
        warnings: results.warnings
      }
    };
  }

  async repairSinglePlugin(pluginInput) {
    try {
      const normalizedInput = this.normalizeInput(pluginInput);
      const syntaxFixed = this.fixSyntaxErrors(normalizedInput);
      const structureFixed = this.fixStructureErrors(syntaxFixed);
      const semanticFixed = await this.fixSemanticErrors(structureFixed);
      const cozeCompatible = this.ensureCozeCompatibility(semanticFixed);
      const validation = this.validatePlugin(cozeCompatible);

      return {
        success: validation.isValid,
        plugin: cozeCompatible,
        repaired: true,
        warnings: validation.warnings,
        validation_result: validation
      };

    } catch (error) {
      return {
        success: false,
        error: error.message,
        repaired: false
      };
    }
  }

  normalizeInput(input) {
    if (typeof input === 'string') {
      try {
        return JSON.parse(input);
      } catch (e) {
        const fixedJson = this.fixJsonString(input);
        return JSON.parse(fixedJson);
      }
    }
    return input;
  }

  fixJsonString(jsonStr) {
    let fixed = jsonStr
      .replace(/\/\*[\s\S]*?\*\//g, '')
      .replace(/\/\/.*$/gm, '')
      .replace(/([{,]\s*)([a-zA-Z_$][a-zA-Z0-9_$]*)(\s*:)/g, '$1"$2"$3')
      .replace(/'/g, '"')
      .replace(/,\s*([}\]])/g, '$1')
      .replace(/([}\]"])\s*([{["])/g, '$1,$2');

    try {
      JSON.parse(fixed);
      return fixed;
    } catch (e) {
      return this.aggressiveJsonRepair(fixed);
    }
  }

  aggressiveJsonRepair(jsonStr) {
    const jsonObjects = [];
    let depth = 0;
    let start = -1;

    for (let i = 0; i < jsonStr.length; i++) {
      if (jsonStr[i] === '{') {
        if (depth === 0) start = i;
        depth++;
      } else if (jsonStr[i] === '}') {
        depth--;
        if (depth === 0 && start !== -1) {
          const candidate = jsonStr.substring(start, i + 1);
          try {
            JSON.parse(candidate);
            jsonObjects.push(candidate);
          } catch (e) {
          }
          start = -1;
        }
      }
    }

    if (jsonObjects.length > 0) {
      return jsonObjects.reduce((a, b) => a.length > b.length ? a : b);
    }

    return '{"name": "repaired_plugin", "version": "1.0.0", "description": "自动修复的Coze插件", "nodes": [], "edges": []}';
  }

  fixSyntaxErrors(plugin) {
    const fixed = JSON.parse(JSON.stringify(plugin));

    if (!fixed.name) fixed.name = `repaired_plugin_${Date.now()}`;
    if (!fixed.version) fixed.version = "1.0.0";
    if (!fixed.description) fixed.description = "自动修复的Coze插件";

    if (!Array.isArray(fixed.nodes)) fixed.nodes = [];
    if (!Array.isArray(fixed.edges)) fixed.edges = [];

    fixed.nodes = this.fixDuplicateNodeIds(fixed.nodes);

    return fixed;
  }

  fixDuplicateNodeIds(nodes) {
    const idMap = new Map();

    return nodes.map(node => {
      const newNode = { ...node };

      if (!newNode.id) {
        newNode.id = `node_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
        return newNode;
      }

      if (idMap.has(newNode.id)) {
        const count = idMap.get(newNode.id) + 1;
        idMap.set(newNode.id, count);
        newNode.id = `${newNode.id}_${count}`;
      } else {
        idMap.set(newNode.id, 1);
      }

      return newNode;
    });
  }

  fixStructureErrors(plugin) {
    const fixed = JSON.parse(JSON.stringify(plugin));

    fixed.nodes = fixed.nodes.map(node => this.normalizeNode(node));
    fixed.edges = fixed.edges.map(edge => this.normalizeEdge(edge));

    const nodeIds = new Set(fixed.nodes.map(node => node.id));
    fixed.edges = fixed.edges.filter(edge => {
      const sourceExists = nodeIds.has(edge.source);
      const targetExists = nodeIds.has(edge.target);
      return sourceExists && targetExists;
    });

    return fixed;
  }

  normalizeNode(node) {
    const normalized = {
      id: node.id || `node_${Date.now()}`,
      type: node.type || 'function',
      name: node.name || '未命名节点',
      ...node
    };

    if (!normalized.config) normalized.config = {};
    if (!normalized.inputs) normalized.inputs = {};
    if (!normalized.outputs) normalized.outputs = {};

    return normalized;
  }

  normalizeEdge(edge) {
    return {
      source: edge.source,
      target: edge.target,
      sourceHandle: edge.sourceHandle || 'output',
      targetHandle: edge.targetHandle || 'input',
      ...edge
    };
  }

  async fixSemanticErrors(plugin) {
    let fixed = JSON.parse(JSON.stringify(plugin));

    fixed.nodes = fixed.nodes.map(node => this.fixNodeDataTypes(node));
    fixed.edges = this.detectAndFixCircularDependencies(fixed.nodes, fixed.edges);
    fixed = this.fixOrphanedNodes(fixed);

    return fixed;
  }

  fixNodeDataTypes(node) {
    const fixedNode = { ...node };

    if (fixedNode.config && typeof fixedNode.config !== 'object') {
      fixedNode.config = {};
    }

    if (fixedNode.inputs && typeof fixedNode.inputs !== 'object') {
      fixedNode.inputs = {};
    }

    if (fixedNode.outputs && typeof fixedNode.outputs !== 'object') {
      fixedNode.outputs = {};
    }

    return fixedNode;
  }

  detectAndFixCircularDependencies(nodes, edges) {
    const graph = {};
    nodes.forEach(node => {
      graph[node.id] = [];
    });

    edges.forEach(edge => {
      if (graph[edge.source]) {
        graph[edge.source].push(edge.target);
      }
    });

    const visited = new Set();
    const recursionStack = new Set();
    const edgesToRemove = [];

    const detectCycle = (nodeId) => {
      if (recursionStack.has(nodeId)) return true;
      if (visited.has(nodeId)) return false;

      visited.add(nodeId);
      recursionStack.add(nodeId);

      for (const neighbor of graph[nodeId] || []) {
        if (detectCycle(neighbor)) {
          const problematicEdge = edges.find(edge =>
            edge.source === nodeId && edge.target === neighbor
          );
          if (problematicEdge) {
            edgesToRemove.push(problematicEdge);
          }
          return true;
        }
      }

      recursionStack.delete(nodeId);
      return false;
    };

    nodes.forEach(node => {
      if (!visited.has(node.id)) {
        detectCycle(node.id);
      }
    });

    return edges.filter(edge =>
      !edgesToRemove.some(removedEdge =>
        removedEdge.source === edge.source && removedEdge.target === edge.target
      )
    );
  }

  fixOrphanedNodes(plugin) {
    const fixed = { ...plugin };
    const connectedNodes = new Set();

    fixed.edges.forEach(edge => {
      connectedNodes.add(edge.source);
      connectedNodes.add(edge.target);
    });

    fixed.nodes = fixed.nodes.map(node => {
      if (!connectedNodes.has(node.id)) {
        return {
          ...node,
          _isOrphaned: true,
          _warning: '此节点未被任何边连接'
        };
      }
      return node;
    });

    return fixed;
  }

  ensureCozeCompatibility(plugin) {
    const compatible = JSON.parse(JSON.stringify(plugin));

    if (!compatible.metadata) compatible.metadata = {};
    compatible.metadata.coze_compatible = true;
    compatible.metadata.repaired_at = new Date().toISOString();
    compatible.metadata.repair_mode = this.repairMode;
    compatible.metadata.repair_engine = "Coze Plugin Repair Engine v2.0";

    if (!compatible.manifest) {
      compatible.manifest = {
        schema_version: "1.0",
        type: "coze_plugin",
        capabilities: ["node_execution", "workflow"],
        author: "Coze Repair System"
      };
    }

    return compatible;
  }

  validatePlugin(plugin) {
    const errors = [];
    const warnings = [];

    if (!plugin.name) errors.push("缺少插件名称");
    if (!plugin.version) errors.push("缺少版本号");
    if (!plugin.description) warnings.push("建议提供插件描述");

    if (!Array.isArray(plugin.nodes)) {
      errors.push("nodes必须是数组");
    }

    if (!Array.isArray(plugin.edges)) {
      errors.push("edges必须是数组");
    }

    plugin.nodes.forEach((node, index) => {
      if (!node.id) errors.push(`节点${index}缺少ID`);
      if (!node.type) warnings.push(`节点${index}缺少类型`);
      if (!node.name) warnings.push(`节点${index}缺少名称`);
    });

    const nodeIds = new Set(plugin.nodes.map(node => node.id));
    plugin.edges.forEach((edge, index) => {
      if (!nodeIds.has(edge.source)) {
        errors.push(`边${index}引用了不存在的源节点: ${edge.source}`);
      }
      if (!nodeIds.has(edge.target)) {
        errors.push(`边${index}引用了不存在的目标节点: ${edge.target}`);
      }
    });

    if (!plugin.metadata || !plugin.metadata.coze_compatible) {
      warnings.push("插件缺少Coze兼容性元数据");
    }

    return {
      isValid: errors.length === 0,
      errors,
      warnings
    };
  }

  mergePlugins(plugins) {
    if (plugins.length === 0) {
      return {
        merged_plugin: null,
        repaired_plugins: [],
        formatted_output: "{}",
        merge_strategy: "none",
        message: "没有有效的插件可合并"
      };
    }

    if (plugins.length === 1) {
      return {
        merged_plugin: plugins[0],
        repaired_plugins: plugins,
        formatted_output: JSON.stringify(plugins[0], null, 2),
        merge_strategy: "single_plugin"
      };
    }

    const merged = this.mergeMultiplePlugins(plugins);

    return {
      merged_plugin: merged,
      repaired_plugins: plugins,
      formatted_output: JSON.stringify(merged, null, 2),
      merge_strategy: "multi_plugin_merged"
    };
  }

  mergeMultiplePlugins(plugins) {
    const basePlugin = JSON.parse(JSON.stringify(plugins[0]));
    const idMapping = new Map();

    basePlugin.nodes.forEach(node => {
      idMapping.set(node.id, node.id);
    });

    for (let i = 1; i < plugins.length; i++) {
      const currentPlugin = plugins[i];

      currentPlugin.nodes.forEach(node => {
        const originalId = node.id;
        let newId = originalId;
        let counter = 1;

        while (idMapping.has(newId) && idMapping.get(newId) !== newId) {
          newId = `${originalId}_${counter}`;
          counter++;
        }

        idMapping.set(originalId, newId);

        const newNode = {
          ...node,
          id: newId,
          _original_plugin: currentPlugin.name || `plugin_${i}`,
          _merged_from: originalId
        };

        basePlugin.nodes.push(newNode);
      });

      currentPlugin.edges.forEach(edge => {
        const newSource = idMapping.get(edge.source) || edge.source;
        const newTarget = idMapping.get(edge.target) || edge.target;

        const newEdge = {
          ...edge,
          source: newSource,
          target: newTarget,
          _original_plugin: currentPlugin.name || `plugin_${i}`
        };

        basePlugin.edges.push(newEdge);
      });

      this.mergePluginFields(basePlugin, currentPlugin, i);
    }

    basePlugin.name = `merged_${basePlugin.name}_${plugins.length}_plugins`;
    basePlugin.description = `合并了${plugins.length}个插件的复合插件`;
    basePlugin.merged_from = plugins.map(p => p.name || p.id);
    basePlugin.merge_timestamp = new Date().toISOString();
    basePlugin.merge_strategy = this.conflictResolution;
    basePlugin.total_nodes = basePlugin.nodes.length;
    basePlugin.total_edges = basePlugin.edges.length;

    return basePlugin;
  }

  mergePluginFields(basePlugin, currentPlugin, pluginIndex) {
    if (currentPlugin.metadata) {
      basePlugin.metadata = {
        ...basePlugin.metadata,
        ...currentPlugin.metadata,
        [`merged_plugin_${pluginIndex}`]: currentPlugin.name || `plugin_${pluginIndex}`
      };
    }

    if (currentPlugin.manifest) {
      if (!basePlugin.manifest) basePlugin.manifest = {};
      basePlugin.manifest = {
        ...basePlugin.manifest,
        ...currentPlugin.manifest
      };

      if (currentPlugin.manifest.capabilities) {
        if (!basePlugin.manifest.capabilities) basePlugin.manifest.capabilities = [];
        basePlugin.manifest.capabilities = [
          ...new Set([...basePlugin.manifest.capabilities, ...currentPlugin.manifest.capabilities])
        ];
      }
    }
  }
}

/*
 * 工具1：coze_plugin_code_generator - Coze插件代码生成器
 *
 * 工具介绍：根据用户提供的插件功能需求，自动生成符合Coze平台规范的完整插件代码和配置，
 * 支持基础插件和API请求类型插件，可直接用于Coze平台创建插件。
 *
 * 输入参数配置：
 * | 名称 | 描述 | 类型 | 必填 |
 * |------|------|------|------|
 * | plugin_requirement | 插件的功能描述，例如：创建一个能根据股票代码查询实时价格的插件 | string | 是 |
 *
 * 输出参数配置：
 * | 名称 | 描述 | 类型 |
 * |------|------|------|
 * | code | 返回码，0表示成功，非0表示失败 | number |
 * | message | 操作结果描述 | string |
 * | data | 返回的插件数据对象 | object |
 *
 * data对象结构：
 * | 名称 | 描述 | 类型 |
 * |------|------|------|
 * | pluginName | 生成的插件名称 | string |
 * | pluginDescription | 生成的插件描述 | string |
 * | code | 生成的完整插件代码 | string |
 * | config | 插件配置信息 | object |
 * | usageInstructions | 使用说明 | string |
 * | generationTime | 生成时间 | string |
 *
 * config对象结构：
 * | 名称 | 描述 | 类型 |
 * |------|------|------|
 * | pluginName | 插件名称 | string |
 * | pluginDescription | 插件描述 | string |
 * | runtime | 运行环境 | string |
 * | toolCreationMethod | 工具创建方式 | string |
 * | authorizationMethod | 授权方式 | string |
 * | requestUrl | 请求URL | string |
 * | requestHeaders | 请求头 | object |
 * | toolName | 工具名称 | string |
 * | toolDescription | 工具描述 | string |
 * | inputParameters | 输入参数列表 | array |
 * | outputParameters | 输出参数定义 | object |
