// 尝试导入uuid，如果不可用则使用内置替代
function generateUUID() {
  try {
    return require('uuid').v4();
  } catch (e) {
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
      var r = Math.random() * 16 | 0;
      var v = c === 'x' ? r : (r & 0x3 | 0x8);
      return v.toString(16);
    });
  }
}

function diagnoseWorkflow(config) {
  const issues = [];
  const suggestions = [];
  const nodes = (config && config.nodes) ? config.nodes : [];

  if (nodes.length > 1000) {
    issues.push("节点数量 " + nodes.length + " 超过上限1000");
    suggestions.push("拆分为多个子工作流");
  }

  for (const node of nodes) {
    const nodeId = (node.id !== undefined) ? node.id : 'unknown';
    const nodeType = node.type;
    const timeout = node.timeout || 0;

    if (nodeType === 'llm' && timeout > 10) {
      issues.push("节点[" + nodeId + "] LLM超时" + timeout + "分钟 > 10分钟");
      suggestions.push("将超时调整为≤10分钟");
    } else if (nodeType === 'code' && timeout > 1) {
      issues.push("节点[" + nodeId + "] 代码节点超时" + timeout + "分钟 > 1分钟");
      suggestions.push("将超时调整为≤1分钟");
    } else if (nodeType === 'http_request' && timeout > 10) {
      issues.push("节点[" + nodeId + "] HTTP超时" + timeout + "分钟 > 10分钟");
      suggestions.push("将超时调整为≤10分钟");
    }

    if (nodeType === 'code') {
      try {
        const inputData = node.input || {};
        const inputSize = JSON.stringify(inputData).length;
        if (inputSize > 2 * 1024 * 1024) {
          issues.push("节点[" + nodeId + "] 入参超过2MB");
          suggestions.push("添加数据分片或压缩");
        }
      } catch (e) {
        issues.push("节点[" + nodeId + "] 入参包含不可序列化对象");
        suggestions.push("检查输入参数类型");
      }
    }
  }

  const healthScore = Math.max(0, 100 - issues.length * 15);
  return { health_score: healthScore, issues: issues, suggestions: suggestions };
}

// 函数2: 自动修复工作流
// ============================================
function autoFixWorkflow(config) {
  const fixed = JSON.parse(JSON.stringify(config));
  const fixes = [];
  const nodes = (fixed && fixed.nodes) ? fixed.nodes : [];

  for (const node of nodes) {
    const nodeId = node.id;
    const nodeType = node.type;
    const oldTimeout = node.timeout || 0;

    if (nodeType === 'llm' && oldTimeout > 10) {
      node.timeout = 10;
      fixes.push("节点[" + nodeId + "] LLM超时从" + oldTimeout + "修正为10分钟");
    } else if (nodeType === 'code' && oldTimeout > 1) {
      node.timeout = 1;
      fixes.push("节点[" + nodeId + "] 代码节点超时从" + oldTimeout + "修正为1分钟");
    } else if (nodeType === 'http_request' && oldTimeout > 10) {
      node.timeout = 10;
      fixes.push("节点[" + nodeId + "] HTTP超时从" + oldTimeout + "修正为10分钟");
    }

    if (nodeType === 'code') {
      try {
        if (JSON.stringify(node.input || {}).length > 2 * 1024 * 1024) {
          node.needs_chunking = true;
          fixes.push("节点[" + nodeId + "] 已标记需要分批处理");
        }
      } catch (e) {
        // 忽略序列化错误
      }
    }
  }

  return { fixed: fixed, fixes: fixes };
}

function generateReport(workflowName, diagnosis, fixes) {
  const health = diagnosis.health_score;
  let status;
  if (health >= 90) {
    status = "健康";
  } else if (health >= 60) {
    status = "警告";
  } else {
    status = "致命错误";
  }

  const lines = ["工作流诊断报告 - " + workflowName];
  lines.push("健康评分: " + health + "/100 (" + status + ")");

  if (diagnosis.issues && diagnosis.issues.length > 0) {
    lines.push("发现的问题:");
    for (const item of diagnosis.issues) {
      lines.push("- " + item);
    }
  } else {
    lines.push("未发现任何问题，配置规范");
  }

  if (fixes && fixes.length > 0) {
    lines.push("已执行的自动修复:");
    for (const item of fixes) {
      lines.push("- " + item);
    }
  } else {
    lines.push("无需修复操作");
  }

  if (diagnosis.suggestions && diagnosis.suggestions.length > 0) {
    lines.push("修复建议:");
    for (const item of diagnosis.suggestions) {
      lines.push("- " + item);
    }
  }

  return lines.join("\n");
}

    "ui_region": "refresh_token_and_retry"
  };

  const strategy = strategyMap[detected] || "notify_admin";

  return {
    target_module: detected,
    repair_strategy: strategy,
    original_error_code: code,
    suggested_action: "调用 " + strategy + " 修复"

// ============================================
function workflowRepairer(inp) {
  const wfStr = (inp && inp.workflow_json_str) || "{}";
  let wf;
  try {
    wf = JSON.parse(wfStr);
  } catch (e) {
    wf = {};
  }

  const required = ["id", "name", "nodes", "edges"];
  for (const field of required) {
    if (!(field in wf) || wf[field] === null || wf[field] === undefined) {
      if (field === "id") {
        wf[field] = generateUUID();
      } else if (field === "name") {
        wf[field] = "repaired_workflow";
      } else {
        wf[field] = [];
      }
    }
  }

  if (Array.isArray(wf.nodes)) {
    for (const node of wf.nodes) {
      if (!node.id) {
        node.id = generateUUID().substring(0, 8);
      }
      node.id = String(node.id).replace(/[^a-zA-Z0-9_]/g, '_');
    }
  }

  return { repaired_workflow: wf, status: "success" };
}

// 函数6: 知识库分段工具
// ============================================
function knowledgeRechunk(inp) {
  const text = (inp && inp.document_text) || "";
  const chunkSize = (inp && inp.chunk_size) || 500;
  const overlap = (inp && inp.overlap) || 50;

  if (!text.trim()) {
    return { chunks: [], status: "empty" };
  }

  const sentences = text.split(/(?<=[。！？])/);
  const chunks = [];
  let current = "";

  for (const sent of sentences) {
    if ((current + sent).length <= chunkSize) {
      current += sent;
    } else {
      if (current) {
        chunks.push(current.trim());
      }
      const overlapText = (overlap > 0 && current) ? current.slice(-overlap) : "";
      current = overlapText + sent;
    }
  }

  if (current) {
    chunks.push(current.trim());
  }

  return { chunks: chunks, chunk_count: chunks.length, status: "success" };
}

function cardVariableFiller(inp) {
  const template = (inp && inp.card_template) || "";
  let provided = (inp && inp.provided_vars) || {};

  if (typeof provided === 'string') {
    try {
      provided = JSON.parse(provided);
    } catch (e) {
      provided = {};
    }
  }

  const pattern = /\{\{\s*([a-zA-Z_][a-zA-Z0-9_]*)\s*\}\}/g;
  const required = new Set();
  let match;
  while ((match = pattern.exec(template)) !== null) {
    required.add(match[1]);
  }

  const missing = [];
  for (const varName of required) {
    if (!(varName in provided)) {
      missing.push(varName);
    }
  }

  const autoFilled = {};
  for (const varName of missing) {
    let defaultVal;
    if (varName.toLowerCase().includes('title')) {
      defaultVal = "默认标题";
    } else if (varName.toLowerCase().includes('desc')) {
      defaultVal = "暂无描述";
    } else if (varName.toLowerCase().includes('image')) {
      defaultVal = "https://via.placeholder.com/300";
    } else {
      defaultVal = "未提供" + varName;
    }
    autoFilled[varName] = defaultVal;
  }

  const fullVars = Object.assign({}, provided, autoFilled);
  let rendered = template;
  for (const varName in fullVars) {
    const val = fullVars[varName];
    rendered = rendered.split('{{' + varName + '}}').join(String(val));
  }

  return {
    missing_variables: missing,
    auto_filled_values: autoFilled,
    final_variables: fullVars,
    rendered_preview: rendered.substring(0, 200)
  };
}

function validateParameters(params) {
  const errors = [];
  if (!params || typeof params !== 'object') {
    errors.push({ field: 'params', message: '参数必须是对象' });
    return { valid: false, errors };
  }
  if (!params.user_input || typeof params.user_input !== 'string' || params.user_input.trim() === '') {
    errors.push({ field: 'user_input', message: 'user_input必须是非空字符串' });
  }
  if (params.action && typeof params.action !== 'string') {
    errors.push({ field: 'action', message: 'action必须是字符串' });
  }
  return { valid: errors.length === 0, errors };
}

function determineRoute(params) {
  const action = params.action;
  const user_input = params.user_input;
  
  if (action && action !== 'universal' && action !== 'general' && ROUTING_KEYWORDS[action]) {
    return { module: action, sub_action: 'auto_handle', confidence: 1.0 };
  }
  
  const text = (user_input || '').toLowerCase();
  let maxScore = 0;
  let selectedModule = 'universal';

  for (const [module, keywords] of Object.entries(ROUTING_KEYWORDS)) {
    if (module === 'universal' || module === 'general') continue;
    let score = 0;
    for (const keyword of keywords) {
      if (text.includes(keyword.toLowerCase())) {
        score += 1;
      }
    }
    if (score > maxScore) {
      maxScore = score;
      selectedModule = module;
    }
  }

  const confidence = maxScore > 0 ? Math.min(0.5 + maxScore * 0.1, 1.0) : 0.5;
  return { module: selectedModule, sub_action: 'auto_handle', confidence };
}

function unitConvert(value, fromUnit) {

  };
}

export async function handler({ input, logger }) {
  try {
    if (logger && logger.info) {
      logger.info('Coze Ultimate Plugin 开始处理', { input: input });
    }

    const validation = validateParameters(input);
    if (!validation.valid) {
      if (logger && logger.error) {
        logger.error('参数验证失败', { errors: validation.errors });
      }
      return {
        success: false,
        status: 'failed',
        module: 'universal',
        module_name: '统一入口',
        detected_intent: 'error',
        action: 'validate_params',
        result_success: false,
        metadata_timestamp: Date.now(),
        metadata_version: COZE_ULTIMATE_CONFIG.version
      };
    }

    const route = determineRoute(input);

    if (logger && logger.info) {
      logger.info('路由到模块: ' + route.module, { confidence: route.confidence });
    }

    const startTime = Date.now();
    const result = await executeModule(route.module, route.sub_action, input);
    const processingTime = Date.now() - startTime;

    const response = {
      success: true,
      status: 'success',
      module: route.module,
      module_name: route.module,
      detected_intent: route.sub_action,
      action: input.action || 'auto_route',
      performance_metrics_processing_time_ms: processingTime,
      performance_metrics_confidence_score: route.confidence,
      metadata_timestamp: Date.now(),
      metadata_version: COZE_ULTIMATE_CONFIG.version
    };

    for (const key in result) {
      if (result.hasOwnProperty(key)) {
        response[key] = result[key];
      }
    }

    if (logger && logger.info) {
      logger.info('处理完成', {
        module: route.module,
        processingTime: processingTime + 'ms',
        success: result.result_success
      });
    }

    return response;

  } catch (err) {
    const errMsg = err.message || 'Unknown error';
    
    if (logger && logger.error) {
      logger.error('插件执行失败', { error: errMsg });
    }

    return {
      success: false,
      status: 'failed',
      module: 'universal',
      module_name: '统一入口',
      detected_intent: 'error',
      action: 'exception_handler',
      result_success: false,
      metadata_timestamp: Date.now(),
      metadata_version: COZE_ULTIMATE_CONFIG.version
    };
  }
}

const ApiClients = {
  openai: {
    async call(prompt, model) {
      model = model || 'gpt-4';
      if (!process.env.OPENAI_API_KEY) {
        const domain = prompt.includes('设备') || prompt.includes('IoT') ? 'device_control' :
                      prompt.includes('代码') || prompt.includes('编程') || prompt.includes('脚本') || prompt.includes('Python') || prompt.includes('JavaScript') ? 'code_engineering' :
                      prompt.includes('内容') || prompt.includes('生成') || prompt.includes('短文') || prompt.includes('文章') ? 'content_generation' :
                      prompt.includes('行业') || prompt.includes('分析') || prompt.includes('报告') ? 'industry_application' :
                      'generic';
        const action = prompt.includes('部署') ? 'deploy' :
                      prompt.includes('修复') ? 'fix' :
                      prompt.includes('生成') || prompt.includes('写') ? 'generate' :
                      'execute';
        return {
          choices: [{
            message: {
              content: JSON.stringify({ domain, action, details: prompt })
            }
          }]
        };
      }
      
      const response = await fetch('https://api.openai.com/v1/chat/completions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer ' + process.env.OPENAI_API_KEY
        },
        body: JSON.stringify({
          model,
          messages: [{ role: 'user', content: prompt }],
          temperature: 0.2
        })
      });
      
      return response.json();
    }
  },
  
  github: {
    async getRepoInfo(repoUrl) {
      if (!process.env.GITHUB_TOKEN) {
        const parts = repoUrl.split('/');
        const name = parts.pop() || 'unknown';
        return { name, full_name: repoUrl, stargazers_count: 0 };
      }
      
      const parts = repoUrl.split('/');
      const owner = parts[3];
      const repo = parts[4];
      if (!owner || !repo) {
        throw new Error('无效的GitHub仓库URL');
      }
      
      const response = await fetch('https://api.github.com/repos/' + owner + '/' + repo, {
        headers: {
          'Authorization': 'token ' + process.env.GITHUB_TOKEN
        }
      });
      
      return response.json();
    }
  }
};

const ToolModules = {
  requirementParser: {
    async analyze(requirement, useApi) {
      useApi = useApi || false;
      if (!useApi || !process.env.OPENAI_API_KEY) {
        const domain = requirement.includes('设备') || requirement.includes('IoT') ? 'device_control' :
                      requirement.includes('代码') || requirement.includes('编程') || requirement.includes('脚本') || requirement.includes('Python') || requirement.includes('JavaScript') ? 'code_engineering' :
                      requirement.includes('内容') || requirement.includes('生成') || requirement.includes('短文') || requirement.includes('文章') ? 'content_generation' :
                      requirement.includes('行业') || requirement.includes('分析') || requirement.includes('报告') ? 'industry_application' :
                      'generic';
        
        const action = requirement.includes('部署') ? 'deploy' :
                      requirement.includes('修复') ? 'fix' :
                      requirement.includes('生成') || requirement.includes('写') ? 'generate' :
                      'execute';
        
        return { domain, action, details: requirement };
      }
      
      const response = await ApiClients.openai.call(requirement);
      return JSON.parse(response.choices[0].message.content);
    }
  },
  
  toolManager: {
    getTools(domain) {
      const toolMap = {
        device_control: ['iot-controller', 'device-monitor'],
        code_engineering: ['code-generator', 'debugger', 'deployer'],
        content_generation: ['text-generator', 'image-processor'],
        industry_application: ['data-analyzer', 'report-generator'],
        generic: ['generic-processor']
      };
      return toolMap[domain] || ['generic-processor'];
    },
    
    async execute(toolName, params, useApi) {
      useApi = useApi || false;
      switch (toolName) {
        case 'code-generator':
          if (useApi && process.env.OPENAI_API_KEY) {
            const language = params.language || 'JavaScript';
            const functionality = params.functionality || params.prompt || '示例功能';
            const response = await ApiClients.openai.call(
              '用' + language + '语言编写一个实现以下功能的代码：' + functionality
            );
            return { success: true, code: response.choices[0].message.content };
          }
          return { success: true, code: "// Generated code\nfunction hello() { return 'world'; }" };
          
        case 'deployer':
          if (params.repoUrl && process.env.GITHUB_TOKEN) {
            const repoInfo = await ApiClients.github.getRepoInfo(params.repoUrl);
            const platform = params.platform || 'Heroku';
            return { success: true, url: 'https://' + platform + '.com/' + repoInfo.name };
          }
          const repoUrl = params.repoUrl;
          const platform = params.platform || 'Heroku';
          const repoName = repoUrl ? repoUrl.split('/').pop() : 'app';
          return { success: true, url: 'https://' + platform + '.com/' + repoName };
          
        case 'text-generator':
          if (useApi && process.env.OPENAI_API_KEY) {
            const prompt = params.prompt || params.text || '示例主题';
            const length = params.length || 500;
            const response = await ApiClients.openai.call(
              '生成一篇关于"' + prompt + '"的文本，约' + length + '字。'
            );
            return { success: true, text: response.choices[0].message.content };
          }
          const textPrompt = params.prompt || params.text || 'your request';
          return { success: true, text: 'Generated content for: ' + textPrompt };
          
        case 'iot-controller':
          return { success: true, result: 'IoT设备控制成功: ' + JSON.stringify(params.device || params) };
          
        case 'device-monitor':
          return { success: true, status: 'online', metrics: { cpu: 25, memory: 45, network: 'connected' } };
          
        case 'data-analyzer':
          const task = params.task;
          return { success: true, analysis: '数据分析完成: ' + (task || '通用分析'), insights: ['趋势分析', '异常检测', '预测结果'] };
          
        case 'report-generator':
          const industry = params.industry;
          return { success: true, report: '报告已生成，涉及行业: ' + (industry || '通用'), sections: ['概述', '数据分析', '结论建议'] };
          
        case 'debugger':
          return { success: true, debugInfo: '调试完成，未发现错误', warnings: [] };
          
        case 'image-processor':
          const style = params.style || 'default';
          return { success: true, imageUrl: 'https://example.com/generated-image.png', style };
          
        case 'generic-processor':
        default:
          return { success: true, result: '工具' + toolName + '执行结果', processedAt: new Date().toISOString() };
      }
    }
  }
};

export async function cozeFullStackHandler(args) {
  const input = args.input;
  const logger = args.logger;
  
  try {
    const requirement = input && input.requirement !== undefined 
      ? input.requirement 
      : (typeof input === 'string' ? input : null);
    
    if (!requirement) {
      return {
        success: false,
        error: { 
          message: '缺少必要参数: requirement',
          code: 'MISSING_REQUIREMENT',
          timestamp: new Date().toISOString()
        }
      };
    }
    
    const useApi = input && input.useApi !== undefined ? input.useApi : false;
    
    if (logger && typeof logger.info === 'function') {
      logger.info('执行模式: ' + (useApi ? 'API模式' : '无API模式'));
      logger.info('处理需求: ' + requirement);
    }
    
    const parsedRequirement = await ToolModules.requirementParser.analyze(requirement, useApi);
    const tools = ToolModules.toolManager.getTools(parsedRequirement.domain);
    
    let finalResult;
    switch (parsedRequirement.domain) {
      case 'device_control':
        finalResult = await executeDeviceControl(tools, parsedRequirement);
        break;
      case 'code_engineering':
        finalResult = await executeCodeEngineering(tools, parsedRequirement, useApi);
        break;
      case 'content_generation':
        finalResult = await executeContentGeneration(tools, parsedRequirement, useApi);
        break;
      case 'industry_application':
        finalResult = await executeIndustryApplication(tools, parsedRequirement);
        break;
      default:
        finalResult = await executeGenericTask(tools, parsedRequirement);
    }
    
    return {
      success: true,
      result: finalResult,
      executionDetails: {
        requirement,
        parsedRequirement,
        toolsUsed: tools,
        mode: useApi ? 'API模式' : '无API模式',
        apiAvailable: !!(process.env.OPENAI_API_KEY || process.env.GITHUB_TOKEN),
        timestamp: new Date().toISOString(),
        executionId: 'exec_' + Date.now()
      },
      metadata: {
        pluginVersion: '1.0.0',
        pluginName: 'CozeFullStackAutomation',
        domain: parsedRequirement.domain,
        action: parsedRequirement.action
      }
    };
  } catch (error) {
    const err = error;
    const errorInfo = {
      message: err.message,
      stack: err.stack,
      code: 'EXECUTION_ERROR',
      timestamp: new Date().toISOString()
    };
    
    if (logger && typeof logger.error === 'function') {
      logger.error('执行错误: ' + err.message);
    }
    
    return {
      success: false,
      error: errorInfo
    };
  }
}

async function executeDeviceControl(tools, requirement) {
  const matchResult = requirement.details.match(/控制(.*)设备/);
  const deviceMatch = matchResult ? matchResult[1] : '智能设备';
  const deviceInfo = { 
    type: 'IoT设备', 
    model: 'SmartController', 
    status: 'online',
    name: deviceMatch,
    lastSeen: new Date().toISOString()
  };
  
  const actions = [];
  for (const tool of tools) {
    const toolResult = await ToolModules.toolManager.execute(tool, { device: deviceInfo });
    actions.push(Object.assign({ tool }, toolResult));
  }
  
  return {
    device: deviceInfo,
    actions,
    status: '设备操作完成',
    actionCount: actions.length
  };
}

async function executeCodeEngineering(tools, requirement, useApi) {
  const matchResult = requirement.details.match(/(.*)项目/);
  const projectMatch = matchResult ? matchResult[1] : '未知项目';
  const projectType = requirement.details.includes('前端') ? 'frontend' : 
                      requirement.details.includes('后端') ? 'backend' : 'fullstack';
  const targetPlatform = requirement.details.includes('AWS') ? 'AWS' : 
                        requirement.details.includes('Azure') ? 'Azure' : 'Heroku';
  
  const project = { 
    name: projectMatch, 
    type: projectType, 
    platform: targetPlatform,
    status: 'in_progress',
    createdAt: new Date().toISOString()
  };
  
  for (const tool of tools) {
    let params;
    if (tool === 'code-generator') {
      params = { 
        language: projectType === 'frontend' ? 'JavaScript/React' : 'Python/Node.js', 
        functionality: requirement.details 
      };
    } else if (tool === 'deployer') {
      params = { repoUrl: 'https://github.com/user/' + projectMatch, platform: targetPlatform };
    } else {
      params = { project };
    }
    
    const toolResult = await ToolModules.toolManager.execute(tool, params, useApi);
    if (toolResult.success === true) {
      if (tool === 'code-generator') {
        project.code = toolResult.code;
      } else if (tool === 'deployer') { 
        project.deploymentUrl = toolResult.url; 
        project.status = 'deployed';
        project.deployedAt = new Date().toISOString();
      }
    }
  }
  
  return {
    project,
    toolsUsed: tools,
    success: project.status === 'deployed',
    executionSteps: tools.length
  };
}

async function executeContentGeneration(tools, requirement, useApi) {
  const matchResult = requirement.details.match(/生成(.*)内容/);
  const contentMatch = matchResult ? matchResult[1] : '文本';
  const contentLength = requirement.details.includes('短') ? 300 : 
                       requirement.details.includes('长') ? 2000 : 800;
  const contentStyle = requirement.details.includes('现代') ? 'modern' : 
                       requirement.details.includes('传统') ? 'traditional' : 'default';
  
  const content = { 
    type: contentMatch, 
    status: 'generating',
    style: contentStyle,
    estimatedLength: contentLength,
    createdAt: new Date().toISOString()
  };
  
  for (const tool of tools) {
    let params;
    if (tool === 'text-generator') {
      const promptText = requirement.details.replace('生成', '').trim();
      params = { prompt: promptText, length: contentLength };
    } else if (tool === 'image-processor') {
      params = { text: content.text || '生成图片', style: contentStyle };
    } else {
      params = {};
    }
    
    const toolResult = await ToolModules.toolManager.execute(tool, params, useApi);
    if (toolResult.success === true) {
      if (tool === 'text-generator') {
        content.text = toolResult.text;
      } else if (tool === 'image-processor') {
        content.image = toolResult.imageUrl;
        content.imageStyle = toolResult.style;
      }
    }
  }
  
  content.status = 'completed';
  content.completedAt = new Date().toISOString();
  return content;
}

async function executeIndustryApplication(tools, requirement) {
  const matchResult = requirement.details.match(/(.*)行业/);
  const industry = matchResult ? matchResult[1] : '通用';
  const task = requirement.details.replace(industry + '行业', '').trim() || '数据分析';
  
  const data = { 
    industry, 
    task, 
    status: 'processing',
    industryType: industry,
    createdAt: new Date().toISOString()
  };
  
  for (const tool of tools) {
    const params = { industry, task, context: requirement };
    const toolResult = await ToolModules.toolManager.execute(tool, params);
    
    if (toolResult.success === true) {
      if (toolResult.result) {
        Object.assign(data, toolResult.result);
      }
      if (toolResult.analysis) {
        data.analysis = toolResult.analysis;
        data.insights = toolResult.insights || [];
      }
      if (toolResult.report) {
        data.report = toolResult.report;
        data.reportSections = toolResult.sections || [];
      }
    }
  }
  
  data.status = 'processed';
  data.completedAt = new Date().toISOString();
  return data;
}

async function executeGenericTask(tools, requirement) {
  const state = { 
    input: requirement.details, 
    status: 'in_progress',
    createdAt: new Date().toISOString()
  };
  
  for (const tool of tools) {
    const params = { input: state.input, context: requirement };
    const toolResult = await ToolModules.toolManager.execute(tool, params);
    Object.assign(state, toolResult);
  }
  
  state.status = 'processed';
  state.completedAt = new Date().toISOString();
  return state;
}

      metadataConfig,
      errorMsg: "无"
    };

  } catch (error) {
    if (logger && logger.error) {
      logger.error("处理失败：" + error);
    }
    return {
      generatedCode: "",
      metadataConfig: "",
      errorMsg: "系统错误：" + (error instanceof Error ? error.message : "未知错误")
    };
  }
}

// ============================================
// 代码问题诊断器 - CodeIssueDiagnoser
// ============================================
export async function codeIssueDiagnoserHandler({ input, logger }) {
  try {
    const content = input.inputContent ? input.inputContent.trim() : "";
    if (!content) {
      return {
        errorType: "",
        solution: "",
        recommendedCozePlugins: [],
        githubResources: [],
        diagnosisStatus: "失败（输入信息不完整）",
        summary: "未提供任何内容，请输入包含错误信息、项目场景和使用语言的文本"
      };
    }

    const langMatch = content.match(/(Python|Java|JavaScript|C\+\+|C#)/);
    const lang = langMatch ? langMatch[0] : "未知语言";
    const errorMatch = content.match(/(IndentationError|KeyError|NullPointerException|SyntaxError|IndexOutOfBoundsException)/);
    const errorName = errorMatch ? errorMatch[0] : "未识别错误";

    let errorType = "";
    if (lang === "Python" && errorName === "IndentationError") {
      errorType = "语法错误（Python缩进错误：代码块缩进不一致，可能混用空格和制表符）";
    } else if (lang === "Java" && errorName === "NullPointerException") {
      errorType = "运行时错误（Java空指针异常：调用了未初始化的对象方法）";
    } else if (errorName.includes("SyntaxError")) {
      errorType = "语法错误（" + lang + "语法不符合规范，如缺少分号、括号不匹配）";
    } else {
      errorType = "未明确分类错误（" + errorName + "，请提供更详细的错误信息）";
    }

    let solution = "";
    if (errorType.includes("缩进错误")) {
      solution = "1. 打开VS Code→选中报错代码→右键→格式化文档（或快捷键Shift+Alt+F）；\n2. 手动检查：确保循环/条件语句后的代码块缩进一致（推荐4个空格）；\n3. 修正示例：\nfor i in range(5):\n    print(i)";
    } else if (errorType.includes("空指针异常")) {
      solution = "1. 检查对象初始化：确保调用方法前对象已实例化；\n2. 添加非空判断：\nif (obj != null) {\n    obj.method();\n}";
    } else {
      solution = "建议：1. 复制完整错误堆栈到搜索引擎查询；2. 检查代码语法（如括号、分号）；3. 逐步调试定位问题位置。";
    }

    const recommendedPlugins = [];
    if (errorType.includes("语法错误")) {
      recommendedPlugins.push({
        name: "代码格式化助手",
        function: "自动修复缩进、分号缺失等语法问题，支持多语言"
      });
    }
    recommendedPlugins.push({
      name: "依赖管理工具",
      function: "检测并修复因依赖版本不兼容导致的运行时错误"
    });
    recommendedPlugins.push({
      name: "算法优化插件",
      function: "识别逻辑错误（如循环边界问题）并提供优化建议"
    });

    const githubResources = [];
    if (errorType.includes("缩进错误")) {
      githubResources.push({
        url: "https://github.com/python/cpython",
        coreFunction: "tokenize模块：解析Python代码并检测缩进问题"
      });
    } else if (errorType.includes("空指针异常")) {
      githubResources.push({
        url: "https://github.com/spring-projects/spring-framework",
        coreFunction: "NullSafeUtils：提供非空判断工具类，避免空指针异常"
      });
    }

    let status = "成功（已识别并提供解决方案）";
    if (githubResources.length === 0) {
      status = "部分完成（未匹配到GitHub项目，解决方案仍有效）";
    }
    if (errorType.includes("未明确分类")) {
      status = "部分完成（错误类型未明确，提供通用解决建议）";
    }

    const summary = "诊断结果：" + errorType + "\n核心解决步骤：" + solution.split('\n')[0] + "\n推荐工具：" + recommendedPlugins.map(function(p) { return p.name; }).join('、') + "\nGitHub资源：" + (githubResources.length > 0 ? githubResources[0].url : '无');

    return {
      errorType,
      solution,
      recommendedCozePlugins: recommendedPlugins,
      githubResources: githubResources,
      diagnosisStatus: status,
      summary: summary
    };

  } catch (error) {
    if (logger && logger.error) {
      logger.error("诊断处理失败：" + error);
    }
    return {
      errorType: "",
      solution: "",
      recommendedCozePlugins: [],
      githubResources: [],
      diagnosisStatus: "失败（系统错误）",
      summary: "处理过程出错：" + (error instanceof Error ? error.message : "未知错误")
    };
  }
}
