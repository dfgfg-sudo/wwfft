#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
OmniNeuro AI SuperFactory - 主程序
融合所有子系统，一键启动
"""

import os
import sys
import argparse
import logging
from pathlib import Path
from core.config_manager import ConfigManager
from core.data_processor import DataProcessor
from core.model_trainer import ModelTrainer
from core.inference_engine import InferenceEngine
from core.security_vault import SecurityVault
from core.quantum_memory import QuantumMemory
from pipelines.data_feeder import DataFeeder
from ui.web_interface import WebInterface
from ui.api_server import APIServer
from ui.cli_interface import CLIInterface

class OmniNeuroFactory:
    VERSION = "10.0.0"

    def __init__(self, config_path="config.yaml"):
        self.config = ConfigManager(config_path).config
        self._setup_logging()
        self._init_components()
        self._create_directories()
        logging.info(f"🚀 OmniNeuro AI SuperFactory v{self.VERSION} 初始化完成")

    def _setup_logging(self):
        log_dir = Path(self.config['paths']['log_dir'])
        log_dir.mkdir(parents=True, exist_ok=True)
        logging.basicConfig(
            level=getattr(logging, self.config['system']['log_level']),
            format='%(asctime)s [%(levelname)s] %(name)s: %(message)s',
            handlers=[
                logging.FileHandler(log_dir / 'system.log', encoding='utf-8'),
                logging.StreamHandler()
            ]
        )

    def _init_components(self):
        # 核心组件
        self.data_processor = DataProcessor(self.config)
        self.model_trainer = ModelTrainer(self.config)
        self.inference_engine = InferenceEngine(self.config)
        self.security_vault = SecurityVault(self.config)
        self.memory_system = QuantumMemory(self.config)

        # 数据投喂器（文件监控）
        self.data_feeder = DataFeeder(self.config, callback=self.on_new_data)

        # 界面
        self.web_ui = WebInterface(self.config)
        self.api_server = APIServer(self.config)
        self.cli = CLIInterface(self.config)

    def _create_directories(self):
        for key, path in self.config['paths'].items():
            if key.endswith('_dir'):
                Path(path).mkdir(parents=True, exist_ok=True)

    def on_new_data(self, filepath):
        """新数据到达时的回调"""
        logging.info(f"📥 新数据到达: {filepath}")
        # 自动触发增量训练（如果配置开启）
        if self.config['training'].get('auto_retrain', False):
            self.start_training()

    def start_training(self, data_path=None):
        """启动训练流水线"""
        logging.info("🧠 开始训练流水线...")
        dataset = self.data_processor.prepare_data(data_path)
        model_info = self.model_trainer.train(dataset)
        encrypted_path = self.security_vault.encrypt_model(
            model_info['model_path'], model_info['version']
        )
        self.inference_engine.load_model(encrypted_path)
        logging.info(f"✅ 训练完成，模型已加密存储于 {encrypted_path}")
        return model_info

    def run_inference(self, input_data, **kwargs):
        return self.inference_engine.infer(input_data, **kwargs)

    def start_web_ui(self, port=7860):
        self.web_ui.launch(port=port)

    def start_api(self, port=8000):
        self.api_server.start(port=port)

    def run_cli(self):
        self.cli.run()

    def start_all(self):
        """启动所有服务（GUI + API + 监控）"""
        self.data_feeder.start_monitoring()
        self.start_api()
        self.start_web_ui()

def main():
    parser = argparse.ArgumentParser(description="OmniNeuro AI SuperFactory")
    parser.add_argument("--mode", choices=["cli", "web", "api", "train", "all"],
                        default="all", help="运行模式")
    parser.add_argument("--config", default="config.yaml", help="配置文件路径")
    parser.add_argument("--port", type=int, default=7860, help="Web端口")
    parser.add_argument("--api-port", type=int, default=8000, help="API端口")
    args = parser.parse_args()

    factory = OmniNeuroFactory(args.config)

    if args.mode == "cli":
        factory.run_cli()
    elif args.mode == "web":
        factory.start_web_ui(args.port)
    elif args.mode == "api":
        factory.start_api(args.api_port)
    elif args.mode == "train":
        factory.start_training()
    else:  # all
        factory.start_all()

if __name__ == "__main__":
    main()

import pandas as pd
import json
from PIL import Image
import torchaudio
import fitz  # PyMuPDF
from pathlib import Path
from typing import Any, Dict, List

class DataProcessor:
    def __init__(self, config):
        self.config = config
        self.supported = {
            '.txt': self._load_text,
            '.json': self._load_json,
            '.csv': self._load_csv,
            '.xlsx': self._load_excel,
            '.jpg': self._load_image,
            '.png': self._load_image,
            '.wav': self._load_audio,
            '.mp3': self._load_audio,
            '.pdf': self._load_pdf,
        }

    def process(self, filepath: str) -> Any:
        ext = Path(filepath).suffix.lower()
        if ext in self.supported:
            return self.supported[ext](filepath)
        else:
            raise ValueError(f"Unsupported format: {ext}")

    def _load_text(self, path):
        with open(path, 'r', encoding='utf-8') as f:
            return f.read()

    def _load_json(self, path):
        with open(path, 'r', encoding='utf-8') as f:
            return json.load(f)

    def _load_csv(self, path):
        return pd.read_csv(path)

    def _load_excel(self, path):
        return pd.read_excel(path)

    def _load_image(self, path):
        return Image.open(path).convert('RGB')

    def _load_audio(self, path):
        waveform, sr = torchaudio.load(path)
        return {'waveform': waveform, 'sample_rate': sr}

    def _load_pdf(self, path):
        doc = fitz.open(path)
        text = ""
        for page in doc:
            text += page.get_text()
        return text

    def prepare_data(self, data_path=None):
        """从数据投喂目录加载所有数据，构建统一数据集"""
        inbox = Path(self.config['paths']['data_inbox'])
        all_data = []
        for file in inbox.rglob('*'):
            if file.is_file():
                try:
                    data = self.process(str(file))
                    all_data.append(data)
                except Exception as e:
                    logging.warning(f"跳过文件 {file}: {e}")
        # 这里可以进一步转化为训练数据集
        return all_data

from transformers import AutoModelForCausalLM, AutoTokenizer, TrainingArguments, Trainer
from peft import LoraConfig, get_peft_model
from bitsandbytes import BitsAndBytesConfig
import torch

class ModelTrainer:
    def __init__(self, config):
        self.config = config
        self.model = None
        self.tokenizer = None
        self._init_model()

    def _init_model(self):
        # 量化配置
        bnb_config = None
        if self.config['quantization']['enabled']:
            bnb_config = BitsAndBytesConfig(
                load_in_4bit=True,
                bnb_4bit_use_double_quant=self.config['quantization']['double_quant'],
                bnb_4bit_quant_type=self.config['quantization']['quant_type'],
                bnb_4bit_compute_dtype=getattr(torch, self.config['quantization']['compute_dtype'])
            )
        # 加载基础模型
        self.model = AutoModelForCausalLM.from_pretrained(
            self.config['training']['base_model'],
            quantization_config=bnb_config,
            device_map="auto"
        )
        self.tokenizer = AutoTokenizer.from_pretrained(
            self.config['training']['base_model']
        )
        # 注入LoRA
        if self.config['lora']['enabled']:
            lora_config = LoraConfig(
                r=self.config['lora']['r'],
                lora_alpha=self.config['lora']['alpha'],
                target_modules=self.config['lora']['target_modules'],
                lora_dropout=self.config['lora']['dropout'],
                bias="none",
                task_type="CAUSAL_LM"
            )
            self.model = get_peft_model(self.model, lora_config)

    def train(self, dataset):
        # 训练参数
        training_args = TrainingArguments(
            output_dir=self.config['paths']['model_dir'],
            per_device_train_batch_size=self.config['training']['batch_size'],
            num_train_epochs=self.config['training']['epochs'],
            learning_rate=self.config['training']['learning_rate'],
            warmup_steps=self.config['training']['warmup_steps'],
            gradient_accumulation_steps=self.config['training']['gradient_accumulation_steps'],
            logging_steps=self.config['training']['logging_steps'],
            save_steps=self.config['training']['save_steps'],
            save_total_limit=self.config['training']['save_total_limit'],
            fp16=(self.config['hardware']['precision'] == 'float16'),
            bf16=(self.config['hardware']['precision'] == 'bfloat16'),
            remove_unused_columns=False
        )
        trainer = Trainer(
            model=self.model,
            args=training_args,
            train_dataset=dataset,
            tokenizer=self.tokenizer,
            data_collator=self._get_data_collator()
        )
        trainer.train()
        model_path = f"{self.config['paths']['model_dir']}/final_model"
        self.model.save_pretrained(model_path)
        self.tokenizer.save_pretrained(model_path)
        return {'model_path': model_path, 'version': self.config['system']['version']}

    def _get_data_collator(self):
        # 自定义collator
        from transformers import DataCollatorForLanguageModeling
        return DataCollatorForLanguageModeling(tokenizer=self.tokenizer, mlm=False)

from cryptography.fernet import Fernet
import hashlib
import os

class SecurityVault:
    def __init__(self, config):
        self.config = config
        key = os.environ.get(config['security']['encryption_key_env'])
        if key:
            self.cipher = Fernet(key.encode())
        else:
            # 生成临时密钥（生产环境应妥善保管）
            self.cipher = Fernet.generate_key()

    def encrypt_file(self, filepath):
        with open(filepath, 'rb') as f:
            data = f.read()
        encrypted = self.cipher.encrypt(data)
        enc_path = filepath + '.enc'
        with open(enc_path, 'wb') as f:
            f.write(encrypted)
        return enc_path

    def decrypt_file(self, enc_path):
        with open(enc_path, 'rb') as f:
            encrypted = f.read()
        data = self.cipher.decrypt(encrypted)
        orig_path = enc_path.replace('.enc', '')
        with open(orig_path, 'wb') as f:
            f.write(data)
        return orig_path

    def verify_integrity(self, filepath, expected_hash):
        actual_hash = hashlib.sha256()
        with open(filepath, 'rb') as f:
            for chunk in iter(lambda: f.read(4096), b''):
                actual_hash.update(chunk)
        return actual_hash.hexdigest() == expected_hash
