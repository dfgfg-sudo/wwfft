/**
 * 编码检测：识别 UTF-8 / UTF-8-BOM / UTF-16LE / UTF-16BE
 * @param {Buffer} buffer 文件字节缓冲
 * @returns {{encoding: string, bomLength: number}}
 */
export function detectEncoding(buffer) {
  // 参数验证
  if (!buffer || !Buffer.isBuffer(buffer) || buffer.length === 0) {
    return { encoding: 'utf-8', bomLength: 0 };
  }
  // UTF-8 BOM: EF BB BF
  if (buffer.length >= 3 && buffer[0] === 0xef && buffer[1] === 0xbb && buffer[2] === 0xbf) {
    return { encoding: 'utf-8', bomLength: 3 };
  }
  // UTF-16 LE BOM: FF FE
  if (buffer.length >= 2 && buffer[0] === 0xff && buffer[1] === 0xfe) {
    return { encoding: 'utf-16le', bomLength: 2 };
  }
  // UTF-16 BE BOM: FE FF
  if (buffer.length >= 2 && buffer[0] === 0xfe && buffer[1] === 0xff) {
    return { encoding: 'utf-16be', bomLength: 2 };
  }
  // 启发式：判断是否包含大量 NULL 字节（UTF-16 无 BOM）
  let nullCount = 0;
  const sampleLen = Math.min(buffer.length, 1024);
  for (let i = 0; i < sampleLen; i++) {
    if (buffer[i] === 0x00) nullCount++;
  }
  if (nullCount > sampleLen / 4) {
    // 偶数位置大量 0x00 -> UTF-16LE
    if (buffer.length >= 2 && buffer[0] !== 0x00 && buffer[1] === 0x00) {
      return { encoding: 'utf-16le', bomLength: 0 };
    }
    return { encoding: 'utf-16be', bomLength: 0 };
  }
  return { encoding: 'utf-8', bomLength: 0 };
}

/**
 * 将 Buffer 按检测到的编码转为字符串
 * @param {Buffer} buffer
 * @returns {string}
 */
export function bufferToString(buffer) {
  if (!buffer || !Buffer.isBuffer(buffer) || buffer.length === 0) {
    return '';
  }
  const { encoding, bomLength } = detectEncoding(buffer);
  const body = bomLength > 0 ? buffer.slice(bomLength) : buffer;
  if (encoding === 'utf-16le') {
    return body.toString('utf16le');
  }
  if (encoding === 'utf-16be') {
    // 交换字节序后按 utf16le 解码
    const swapped = Buffer.alloc(body.length);
    for (let i = 0; i + 1 < body.length; i += 2) {
      swapped[i] = body[i + 1];
      swapped[i + 1] = body[i];
    }
    return swapped.toString('utf16le');
  }
  return body.toString('utf-8');
}

/**
 * 路径安全检查：防止路径遍历攻击（../、绝对路径、UNC 路径等）
 * @param {string} entryPath zip 内条目路径
 * @param {string} prefix 路径前缀
 * @returns {{safe: boolean, normalized: string, reason?: string}}
 */
export function safePath(entryPath, prefix) {
  // 参数验证
  if (typeof entryPath !== 'string' || entryPath.length === 0) {
    return { safe: false, normalized: '', reason: '路径为空' };
  }
  const safePrefix = typeof prefix === 'string' ? prefix : '';

  // 统一为正斜杠
  let normalized = entryPath.replace(/\\/g, '/');

  // 拒绝 Windows 绝对路径与 UNC 路径
  if (/^[a-zA-Z]:/.test(normalized) || normalized.startsWith('//')) {
    return { safe: false, normalized, reason: '检测到绝对路径' };
  }

  // 拼接前缀
  if (safePrefix) {
    normalized =
      safePrefix.replace(/\\/g, '/').replace(/\/+$/, '') +
      '/' +
      normalized.replace(/^\/+/, '');
  }

  // 规范化 .. 与 .
  const parts = [];
  for (const seg of normalized.split('/')) {
    if (seg === '' || seg === '.') continue;
    if (seg === '..') {
      if (parts.length === 0) {
        return { safe: false, normalized, reason: '路径遍历超出根目录' };
      }
      parts.pop();
      continue;
    }
    parts.push(seg);
  }
  normalized = parts.join('/');

  // 最终校验：不得包含 ..
  if (normalized.includes('..')) {
    return { safe: false, normalized, reason: '存在非法路径片段' };
  }
  return { safe: true, normalized };
}

/**
 * 格式化文件大小为人类可读字符串
 * @param {number} bytes 字节数
 * @returns {string} 格式化后的字符串（如 "1.5 MB"）
 */
export function formatSize(bytes) {
  if (typeof bytes !== 'number' || isNaN(bytes) || bytes === 0) return '0 B';
  if (bytes < 0) return '0 B';
  const units = ['B', 'KB', 'MB', 'GB'];
  const k = 1024;
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + units[i];
}

/**
 * 从文件名获取扩展名（含点号，小写）
 * @param {string} fileName 文件名
 * @returns {string} 扩展名（如 ".md"），无扩展名时返回空字符串
 */
export function getExtension(fileName) {
  if (typeof fileName !== 'string' || fileName.length === 0) return '';
  const dot = fileName.lastIndexOf('.');
  if (dot === -1 || dot === fileName.length - 1) return '';
  return fileName.substring(dot).toLowerCase();
}

/**
 * 从内容中提取标题
 * @param {string} content 文件文本内容
 * @param {string} fileName 文件名（作为回退标题）
 * @returns {string} 提取到的标题
 */
export function extractTitle(content, fileName) {
  const fallback = (typeof fileName === 'string' ? fileName : '未命名文档').replace(/\.[^.]+$/, '');
  if (!content || typeof content !== 'string' || content.trim().length === 0) return fallback;
  const lines = content.split('\n').map(l => l.trim()).filter(Boolean);
  if (lines.length === 0) return fallback;
  // 尝试 Markdown 标题
  const mdTitle = lines.find(l => /^#{1,6}\s+/.test(l));
  if (mdTitle) return mdTitle.replace(/^#{1,6}\s+/, '').trim();
  // 尝试 HTML 标题
  const htmlMatch = content.match(/<title[^>]*>(.*?)<\/title>/is);
  if (htmlMatch) return htmlMatch[1].trim();
  // 使用首行
  const first = lines[0];
  return first.length > 100 ? first.substring(0, 100) + '...' : first;
}

/**
 * 生成唯一文档 ID
 * @param {string} path 文件路径（用于哈希）
 * @returns {string} 唯一文档 ID
 */
export function genId(path) {
  if (typeof path !== 'string' || path.length === 0) path = 'unknown';
  const t = Date.now().toString(36);
  const r = Math.random().toString(36).substring(2, 8);
  let h = 0;
  for (let i = 0; i < path.length; i++) { h = ((h << 5) - h) + path.charCodeAt(i); h = h & h; }
  return 'doc_' + Math.abs(h).toString(36) + '_' + t + '_' + r;
}

/**
 * 解析 Markdown 内容，提取纯文本与标题
 * @param {string} content
 * @returns {{text: string, title: string}}
 */
export function parseMarkdown(content) {
  if (typeof content !== 'string' || content.length === 0) {
    return { text: '', title: '未命名文档', metadata: {} };
  }

  // ---- YAML frontmatter 解析 ----
  // 支持 `---\n...\n---\n` 格式的 YAML frontmatter
  const metadata = {};
  let bodyContent = content;
  const frontmatterMatch = content.match(/^---\r?\n([\s\S]*?)\r?\n---\r?\n?/);
  if (frontmatterMatch) {
    const yamlBlock = frontmatterMatch[1];
    bodyContent = content.substring(frontmatterMatch[0].length);
    // 逐行解析简单的 YAML key: value 格式
    for (const line of yamlBlock.split(/\r?\n/)) {
      const m = line.match(/^(\w[\w\-_]*)\s*:\s*(.*)$/);
      if (m) {
        let val = m[2].trim();
        // 去除引号
        if ((val.startsWith('"') && val.endsWith('"')) || (val.startsWith("'") && val.endsWith("'"))) {
          val = val.slice(1, -1);
        }
        metadata[m[1]] = val;
      }
    }
  }

  const lines = bodyContent.split(/\r?\n/);
  let title = '';
  const textParts = [];
  let inCodeBlock = false;
  for (const line of lines) {
    if (/^```/.test(line)) {
      inCodeBlock = !inCodeBlock;
      textParts.push(line);
      continue;
    }
    if (!inCodeBlock) {
      const m = line.match(/^#{1,2}\s+(.*)$/);
      if (m && !title) {
        title = m[1].trim();
      }
    }
    textParts.push(line);
  }
  // 去除 markdown 标记得到纯文本
  const text = textParts
    .join('\n')
    .replace(/!\[[^\]]*\]\([^)]*\)/g, '')   // 图片
    .replace(/\[([^\]]*)\]\([^)]*\)/g, '$1') // 链接保留文本
    .replace(/[*_`~>#-]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();

  // frontmatter 中的 title 优先于正文标题
  if (metadata.title) {
    title = metadata.title;
  }

  return { text, title: title || '未命名文档', metadata };
}

/**
 * 解析 HTML 内容，去除标签
 * @param {string} content
 * @returns {{text: string, title: string}}
 */
export function parseHtml(content) {
  if (typeof content !== 'string' || content.length === 0) {
    return { text: '', title: '未命名HTML文档' };
  }
  let title = '';
  const titleMatch = content.match(/<title[^>]*>([\s\S]*?)<\/title>/i);
  if (titleMatch) title = titleMatch[1].trim();
  // 去除 script/style/注释
  const stripped = content
    .replace(/<script[\s\S]*?<\/script>/gi, '')
    .replace(/<style[\s\S]*?<\/style>/gi, '')
    .replace(/<!--[\s\S]*?-->/g, '');
  const text = stripped
    .replace(/<[^>]+>/g, ' ')
    .replace(/&nbsp;/g, ' ')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&amp;/g, '&')
    .replace(/&quot;/g, '"')
    .replace(/&#39;/g, "'")
    .replace(/\s+/g, ' ')
    .trim();
  return { text, title: title || '未命名HTML文档' };
}

/**
 * 解析 JSON 内容
 * @param {string} content
 * @returns {{text: string, title: string}}
 */
export function parseJson(content) {
  if (typeof content !== 'string' || content.length === 0) {
    return { text: '', title: 'JSON文档', metadata: {} };
  }
  let text = content;
  let title = 'JSON文档';
  const metadata = {};
  try {
    const obj = JSON.parse(content);
    if (obj && typeof obj === 'object') {
      if (typeof obj.title === 'string') title = obj.title;
      else if (typeof obj.name === 'string') title = obj.name;
      // 提取所有 string 类型属性为 metadata
      for (const [key, value] of Object.entries(obj)) {
        if (typeof value === 'string' && key !== 'title' && key !== 'name') {
          metadata[key] = value;
        }
      }
    }
    text = JSON.stringify(obj, null, 2);
  } catch (e) {
    // 解析失败保留原文
  }
  return { text, title, metadata };
}

/**
 * 解析 XML 内容
 * @param {string} content
 * @returns {{text: string, title: string}}
 */
export function parseXml(content) {
  if (typeof content !== 'string' || content.length === 0) {
    return { text: '', title: 'XML文档' };
  }
  let title = 'XML文档';
  const titleMatch = content.match(
    /<(?:title|name|id)[^>]*>([\s\S]*?)<\/(?:title|name|id)>/i
  );
  if (titleMatch) title = titleMatch[1].trim();
  const text = content
    .replace(/<\?[\s\S]*?\?>/g, '')
    .replace(/<!--[\s\S]*?-->/g, '')
    .replace(/<[^>]+>/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
  return { text, title };
}

/**
 * 解析 CSV/TSV 内容
 * @param {string} content
 * @param {string} ext
 * @returns {{text: string, title: string}}
 */
export function parseCsv(content, ext) {
  if (typeof content !== 'string' || content.length === 0) {
    return { text: '', title: '表格文档', metadata: {} };
  }
  const delimiter = ext === 'tsv' ? '\t' : ',';
  const lines = content.split(/\r?\n/).filter((l) => l.length > 0);
  let title = '表格文档';
  const metadata = { columns: 0, rows: 0 };
  if (lines.length > 0) {
    const firstRow = lines[0].split(delimiter);
    metadata.columns = firstRow.length;
    metadata.rows = lines.length;
    const firstCell = firstRow[0].replace(/^"|"$/g, '');
    if (firstCell) title = firstCell + '（共' + lines.length + '行' + firstRow.length + '列）';
  }
  return { text: content, title, metadata };
}

/**
 * 解析代码文件
 * @param {string} content
 * @param {string} format
 * @returns {{text: string, title: string}}
 */
export function parseCode(content, format) {
  if (typeof content !== 'string' || content.length === 0) {
    return { text: '', title: (format || '代码') + '代码文件' };
  }
  const firstLine = content.split(/\r?\n/)[0] || '';
  let title = (format || '代码') + '代码文件';
  // 从注释提取标题
  const commentMatch = firstLine.match(/(?:\/\/|#|<!--|\/\*)\s*(.+)/);
  if (commentMatch) {
    title = commentMatch[1].trim().replace(/-->$|\*\/$/, '').trim();
  }
  return { text: content, title };
}

/**
 * 解析二进制文档（占位：Coze 运行时可能无法直接解析 docx/pdf）
 * @param {string} format
 * @param {string} fileName
 * @param {number} fileSize
 * @returns {{text: string, title: string}}
 */
export function parseBinaryDoc(format, fileName, fileSize, buffer) {
  const safeFormat = format || 'BINARY';
  const safeName = fileName || '未命名文档';
  const safeSize = typeof fileSize === 'number' ? fileSize : 0;

  // 尝试从二进制数据中提取可读文本
  let extractedText = '';
  if (buffer && Buffer.isBuffer(buffer) && buffer.length > 0) {
    // 移除控制字符，保留可打印 ASCII 和 UTF-8 中文字符
    const raw = buffer.toString('utf-8');
    const cleaned = raw.replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, '');
    // 统计可读字符比例
    const printableCount = (cleaned.match(/[\x20-\x7E\u4e00-\u9fa5]/g) || []).length;
    if (printableCount > 20) {
      extractedText = cleaned.substring(0, 2000); // 最多提取 2000 字符
    }
  }

  const text = extractedText
    ? '[二进制文档 ' + safeName + '] 文件大小: ' + safeSize + ' 字节\n\n提取的可读文本:\n' + extractedText
    : '[二进制文档 ' + safeName + '] 文件大小: ' + safeSize + ' 字节，需在运行时通过专门解析器提取';

  return {
    text,
    title: safeFormat.toUpperCase() + '文档',
    metadata: { format: safeFormat, file_name: safeName, file_size: safeSize, text_extracted: extractedText.length > 0 },
  };
}

/**
 * 根据格式解析文件内容
 * @param {Buffer} buffer
 * @param {string} format
 * @param {string} ext
 * @returns {{text: string, title: string}}
 */
export function parseContent(buffer, format, ext) {
  if (!buffer || !Buffer.isBuffer(buffer)) {
    return { text: '', title: '空文档', metadata: {} };
  }
  switch (format) {
    case 'markdown':
      return parseMarkdown(bufferToString(buffer));
    case 'html':
      return parseHtml(bufferToString(buffer));
    case 'json':
      return parseJson(bufferToString(buffer));
    case 'xml':
      return parseXml(bufferToString(buffer));
    case 'csv':
      return parseCsv(bufferToString(buffer), ext);
    case 'yaml':
    case 'config':
    case 'text':
      return { text: bufferToString(buffer), title: '文本文档', metadata: {} };
    case 'javascript':
    case 'typescript':
    case 'python':
    case 'java':
    case 'go':
    case 'rust':
    case 'c':
    case 'cpp':
    case 'csharp':
    case 'ruby':
    case 'php':
    case 'shell':
    case 'sql':
      return parseCode(bufferToString(buffer), format);
    case 'pdf':
    case 'doc':
    case 'docx':
    case 'xls':
    case 'xlsx':
    case 'ppt':
    case 'pptx':
    case 'rtf':
      return parseBinaryDoc(format, undefined, buffer.length, buffer);
    default:
      return { text: bufferToString(buffer), title: '未知格式文档', metadata: {} };
  }
}

/**
 * 统计字数（中文按字，英文按词）
 * @param {string} text
 * @returns {number}
 */
export function countWords(text) {
  if (!text || typeof text !== 'string') return 0;
  const cjk = (
    text.match(/[\u4e00-\u9fa5\u3040-\u30ff\uac00-\ud7af]/g) || []
  ).length;
  const words = (
    text
      .replace(/[\u4e00-\u9fa5\u3040-\u30ff\uac00-\ud7af]/g, ' ')
      .match(/[a-zA-Z0-9]+/g) || []
  ).length;
  return cjk + words;
}

/**
 * 生成目录树字符串
 * @param {string[]} paths 规范化后的文件路径列表
 * @returns {string}
 */
export function buildDirectoryTree(paths) {
  if (!Array.isArray(paths) || paths.length === 0) return '';
  const root = {};
  for (const p of paths) {
    if (typeof p !== 'string' || p.length === 0) continue;
    const segs = p.split('/');
    let node = root;
    segs.forEach((seg) => {
      if (!node[seg]) node[seg] = {};
      node = node[seg];
    });
  }
  const lines = [];
  function walk(node, prefix) {
    const keys = Object.keys(node).sort();
    keys.forEach((key, idx) => {
      const isLast = idx === keys.length - 1;
      lines.push(prefix + (isLast ? '└── ' : '├── ') + key);
      walk(node[key], prefix + (isLast ? '    ' : '│   '));
    });
  }
  walk(root, '');
  return lines.join('\n');
}

/**
 * knowledge_batch_upload 主处理函数
 * 原始入口函数名: handler
 *
 * @param {Object} params
 * @param {Object} params.input 输入参数
 * @param {string} params.input.zip_base64 ZIP文件的Base64编码
 * @param {string} [params.input.path_prefix] 路径前缀
 * @param {string[]} [params.input.allowed_extensions] 允许的扩展名白名单
 * @param {number} [params.input.max_file_size_mb] 单文件最大MB
 * @param {boolean} [params.input.skip_hidden] 跳过隐藏文件
 * @param {Object} params.logger 日志对象
 * @returns {Promise<Object>} 处理结果
 */
export async function batchUploadHandler({ input, logger }) {
  const startTime = Date.now();
  const logs = [];
  const log = (level, msg) => {
    const entry = { time: new Date().toISOString(), level, msg };
    logs.push(entry);
    if (logger && typeof logger[level] === 'function') {
      logger[level](msg);
    }
  };

  // ---------- 参数验证 ----------
  if (!input || typeof input !== 'object' || Array.isArray(input)) {
    log('error', 'input 参数缺失或类型错误');
    return {
      success: false,
      total_count: 0,
      success_count: 0,
      fail_count: 0,
      skipped_count: 0,
      processing_time_ms: Date.now() - startTime,
      directory_tree: '',
      documents: [],
      summary: '参数验证失败：input 缺失或类型错误',
      logs,
    };
  }
  if (typeof input.zip_base64 !== 'string' || input.zip_base64.length === 0) {
    log('error', 'zip_base64 为必填项且必须为非空字符串');
    return {
      success: false,
      total_count: 0,
      success_count: 0,
      fail_count: 0,
      skipped_count: 0,
      processing_time_ms: Date.now() - startTime,
      directory_tree: '',
      documents: [],
      summary: '参数验证失败：zip_base64 缺失或为空',
      logs,
    };
  }

  const pathPrefix = typeof input.path_prefix === 'string' ? input.path_prefix.trim().replace(/\/$/, '') : '';
  const allowedExtensions = Array.isArray(input.allowed_extensions)
    ? input.allowed_extensions
        .map((e) => (typeof e === 'string' ? e.toLowerCase().trim() : ''))
        .filter((e) => e.length > 0)
    : DEFAULT_EXTENSIONS;
  const maxFileSizeMb =
    typeof input.max_file_size_mb === 'number' && input.max_file_size_mb > 0
      ? input.max_file_size_mb
      : 20;
  const maxFileSize = maxFileSizeMb * 1024 * 1024;
  const skipHidden =
    typeof input.skip_hidden === 'boolean' ? input.skip_hidden : true;

  log('info', '=== Coze 批量知识库上传插件启动 ===');
  log(
    'info',
    '配置: 前缀=' +
      (pathPrefix || '(无)') +
      ', 白名单=' +
      (allowedExtensions ? allowedExtensions.join(',') : '(全部)') +
      ', 最大=' +
      formatSize(maxFileSize) +
      ', 跳过隐藏=' +
      skipHidden
  );

  // ---------- 解码 Base64 并打开 ZIP ----------
  let zip;
  let zipEntries;
  try {
    const zipBuffer = Buffer.from(input.zip_base64, 'base64');
    if (zipBuffer.length === 0) {
      throw new Error('Base64 解码后为空');
    }
    log('info', 'ZIP 解码成功，大小：' + formatSize(zipBuffer.length));
    zip = new AdmZip(zipBuffer);
    zipEntries = zip.getEntries();
    log('info', 'ZIP 解析完成，共 ' + zipEntries.length + ' 个条目');
  } catch (err) {
    log('error', 'ZIP 解压失败: ' + err.message);
    return {
      success: false,
      total_count: 0,
      success_count: 0,
      fail_count: 0,
      skipped_count: 0,
      processing_time_ms: Date.now() - startTime,
      directory_tree: '',
      documents: [],
      summary: 'ZIP 解压失败: ' + err.message,
      logs,
    };
  }

  // ---------- 遍历条目，收集有效文件 ----------
  const allEntries = [];
  let skippedCount = 0;

  for (const entry of zipEntries) {
    const rawName = entry.entryName;

    // 跳过目录
    if (entry.isDirectory) {
      continue;
    }

    // 跳过 macOS 系统文件
    if (rawName.startsWith('__MACOSX/') || rawName.includes('/._')) {
      continue;
    }

    const fileName = rawName.split('/').pop() || '';

    // 跳过系统文件
    if (fileName === '.DS_Store' || fileName === 'Thumbs.db') {
      continue;
    }

    // 路径安全检查
    const pathCheck = safePath(rawName, pathPrefix);
    if (!pathCheck.safe) {
      skippedCount++;
      log('warn', '路径安全检查未通过: ' + rawName + ' (' + pathCheck.reason + ')');
      continue;
    }
    const safeEntryPath = pathCheck.normalized;

    // 跳过隐藏文件
    if (skipHidden) {
      const baseName = rawName.split('/').pop() || '';
      const segs = rawName.split('/');
      if (baseName.startsWith('.') || segs.some((s) => s.startsWith('.'))) {
        skippedCount++;
        log('debug', '跳过隐藏文件: ' + rawName);
        continue;
      }
    }

    // 读取数据
    let data;
    try {
      data = entry.getData();
    } catch (err) {
      skippedCount++;
      log('warn', '读取条目数据失败: ' + rawName + ' - ' + err.message);
      continue;
    }

    // 空文件跳过
    if (data.length === 0) {
      skippedCount++;
      log('debug', '跳过空文件: ' + rawName);
      continue;
    }

    const ext = getExtension(fileName);

    // 扩展名白名单过滤
    if (allowedExtensions && !allowedExtensions.includes(ext)) {
      skippedCount++;
      log('debug', '跳过不在白名单的扩展名: ' + rawName + ' (' + ext + ')');
      continue;
    }

    // 文件大小过滤
    if (data.length > maxFileSize) {
      skippedCount++;
      log('warn', '文件超过大小限制: ' + rawName + ' (' + formatSize(data.length) + ' > ' + formatSize(maxFileSize) + ')');
      continue;
    }

    allEntries.push({
      entryPath: safeEntryPath,
      rawName: rawName,
      fileName: fileName,
      extension: ext,
      size: data.length,
      data: Buffer.from(data),
    });
  }

  log('info', '有效文件条目：' + allEntries.length + ' 个（跳过 ' + skippedCount + ' 个）');

  // ---------- 解析每个文件 ----------
  const documents = [];
  const treePaths = [];
  let successCount = 0;
  let failCount = 0;

  for (const entry of allEntries) {
    try {
      const format = getFormat(entry.extension);
      const { text, title, metadata } = parseContent(entry.data, format, entry.extension);
      const wordCount = countWords(text);
      const displayPath = pathPrefix ? pathPrefix + '/' + entry.entryPath : entry.entryPath;
      const docTitle = extractTitle(text, entry.fileName);

      const doc = {
        id: genId(entry.entryPath),
        title: docTitle || title,
        source_path: entry.entryPath,
        path: displayPath,
        content: text,
        format: format,
        file_size: entry.size,
        word_count: wordCount,
        metadata: metadata || {},
        success: true,
        processed_at: new Date().toISOString(),
      };

      documents.push(doc);
      treePaths.push(displayPath);
      successCount++;
      log('info', '  [OK] ' + displayPath + ' (' + formatSize(entry.size) + ', ' + wordCount + ' 字)');
    } catch (err) {
      failCount++;
      const displayPath = pathPrefix ? pathPrefix + '/' + entry.entryPath : entry.entryPath;
      log('error', '  [FAIL] ' + displayPath + ': ' + err.message);
      documents.push({
        id: genId(entry.entryPath),
        title: entry.fileName,
        source_path: entry.entryPath,
        path: displayPath,
        content: '',
        format: getFormat(entry.extension),
        file_size: entry.size,
        word_count: 0,
        success: false,
        error_message: String(err.message),
        processed_at: new Date().toISOString(),
      });
    }
  }

  // ---------- 生成目录树 ----------
  const directoryTree = buildDirectoryTree(treePaths);

  const processingTimeMs = Date.now() - startTime;
  const totalCount = allEntries.length + skippedCount;
  const summary = [
    '批量处理完成！',
    '成功：' + successCount + ' | 失败：' + failCount + ' | 跳过：' + skippedCount + ' | 总计：' + totalCount,
    '耗时：' + processingTimeMs + 'ms',
  ].join(' | ');

  log('info', summary);
  log('info', '=== 插件执行结束 ===');

  return {
    success: failCount === 0 && successCount > 0,
    total_count: totalCount,
    success_count: successCount,
    fail_count: failCount,
    skipped_count: skippedCount,
    processing_time_ms: processingTimeMs,
    directory_tree: directoryTree,
    documents: documents,
    summary: summary,
    logs: logs,
  };
}

/**
 * 输入消毒函数：对字符串进行 HTML 实体转义，防止注入攻击
 * @param {string|*} input 输入内容
 * @returns {string|*} 消毒后的内容
 */
export function sanitizeInput(input) {
  if (typeof input !== 'string') {
    return input;
  }
  const entities = { '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;', '\\': '&#92;' };
  return input.replace(/[<>"'\\]/g, (char) => entities[char] || char);
}

/**
 * 参数验证函数：验证输入参数是否符合 INPUT_SCHEMA 要求
 * @param {Object} params 输入参数对象
 * @returns {{valid: boolean, errors: Array}} 验证结果
 */
export function validateParameters(params) {
  const errors = [];
  if (!params || typeof params !== 'object') {
    errors.push({ field: 'params', message: '参数必须是对象' });
    return { valid: false, errors };
  }
  if (!params.user_input || typeof params.user_input !== 'string' || params.user_input.trim() === '') {
    errors.push({ field: 'user_input', message: 'user_input必须是非空字符串' });
  }
  if (params.action && typeof params.action !== 'string') {
    errors.push({ field: 'action', message: 'action必须是字符串' });
  }
  if (params.action && typeof params.action === 'string' && params.action !== 'universal' && params.action !== 'general' && !MODULES_DEFINITION[params.action]) {
    errors.push({ field: 'action', message: 'action必须是有效模块名: ' + Object.keys(MODULES_DEFINITION).join(', ') });
  }
  if (params.sub_action && typeof params.sub_action !== 'string') {
    errors.push({ field: 'sub_action', message: 'sub_action必须是字符串' });
  }
  if (params.options && typeof params.options !== 'object') {
    errors.push({ field: 'options', message: 'options必须是对象' });
  }
  return { valid: errors.length === 0, errors };
}

/**
 * 路由决策函数：根据用户输入和关键词匹配确定目标模块
 * @param {Object} params 包含 action 和 user_input 的参数对象
 * @returns {{module: string, sub_action: string, confidence: number}} 路由结果
 */
export function determineRoute(params) {
  if (!params || typeof params !== 'object') {
    return { module: 'universal', sub_action: 'auto_handle', confidence: 0.5 };
  }
  const { action, user_input } = params;
  // 若显式指定 action 且为有效模块，直接路由
  if (action && action !== 'universal' && action !== 'general' && MODULES_DEFINITION[action]) {
    return { module: action, sub_action: 'auto_handle', confidence: 1.0 };
  }
  // 关键词匹配
  const text = (user_input || '').toLowerCase();
  let maxScore = 0;
  let selectedModule = 'universal';
  for (const [module, keywords] of Object.entries(ROUTING_KEYWORDS)) {
    if (module === 'universal' || module === 'general') continue;
    let score = 0;
    for (const keyword of keywords) {
      if (text.includes(keyword.toLowerCase())) score += 1;
    }
    if (score > maxScore) { maxScore = score; selectedModule = module; }
  }
  const confidence = maxScore > 0 ? Math.min(0.5 + maxScore * 0.1, 1.0) : 0.5;
  return { module: selectedModule, sub_action: 'auto_handle', confidence };
}

/**
 * 意图检测函数：根据用户输入快速检测意图模块
 * @param {string} userInput 用户输入文本
 * @returns {string} 检测到的模块ID
 */
export function detectIntent(userInput) {
  if (typeof userInput !== 'string' || userInput.length === 0) {
    return 'universal';
  }
  const input = userInput.toLowerCase();
  for (const [module, keywords] of Object.entries(ROUTING_KEYWORDS)) {
    for (const keyword of keywords) {
      if (input.includes(keyword.toLowerCase())) return module;
    }
  }
  return 'universal';
}

/**
 * DeepSeekAIFactoryUltimate 主处理函数（入口函数）
 *
 * @param {Object|string} input 输入参数对象或 JSON 字符串
 * @param {string} input.user_input 用户输入内容
 * @param {string} [input.action] 指定模块
 * @param {string} [input.sub_action] 子操作
 * @param {Object} [input.options] 选项
 * @returns {Promise<Object>} 处理结果（符合 OUTPUT_SCHEMA）
 */
export async function handler(input) {
  try {
    const startTime = Date.now();
    const params = typeof input === 'string' ? JSON.parse(input) : input;
    const validation = validateParameters(params);

    if (!validation.valid) {
      return {
        success: false,
        status: 'failed',
        error: {
          code: 'INVALID_PARAMETERS',
          message: '参数验证失败',
          details: validation.errors
        },
        metadata: {
          timestamp: Date.now(),
          version: COZE_ULTIMATE_PLUGIN_CONFIG.version,
          request_id: 'req_' + Date.now()
        }
      };
    }

    const sanitizedInput = sanitizeInput(params.user_input);
    const route = determineRoute({ ...params, user_input: sanitizedInput });
    const moduleResult = await executeModule(route.module, route.sub_action || 'auto_handle', { ...params, user_input: sanitizedInput });

    const processingTime = Date.now() - startTime;

    return {
      success: true,
      status: 'success',
      module: route.module,
      module_name: MODULES_DEFINITION[route.module] ? MODULES_DEFINITION[route.module].name : route.module,
      detected_intent: route.module,
      action: route.sub_action,
      result: moduleResult,
      performance_metrics: {
        processing_time_ms: processingTime,
        confidence_score: route.confidence,
        modules_executed: [route.module]
      },
      next_actions: [],
      errors_fixed: [],
      metadata: {
        timestamp: Date.now(),
        version: COZE_ULTIMATE_PLUGIN_CONFIG.version,
        request_id: 'req_' + Date.now(),
        automation_enabled: true,
        total_modules: Object.keys(MODULES_DEFINITION).length,
        total_tools: COZE_ULTIMATE_PLUGIN_CONFIG.total_tools,
        routed_module: route.module,
        routing_confidence: route.confidence
      }
    };
  } catch (error) {
    return {
      success: false,
      status: 'failed',
      error: {
        code: 'INTERNAL_ERROR',
        message: error.message,
        stack: error.stack
      },
      metadata: {
        timestamp: Date.now(),
        version: COZE_ULTIMATE_PLUGIN_CONFIG.version,
        request_id: 'req_' + Date.now()
      }
    };
  }
}

export default handler;
