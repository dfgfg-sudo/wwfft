// ============================================
// 配置常量
// ============================================
const RESOURCE_CONFIG = {
  version: "2.0.0",
  max_resources_per_page: 100,
  retry_count: 3,
  retry_delay_ms: 1000,
  supported_types: ["workflow", "dialogflow", "plugin", "agent", "knowledge_base"],
  sensitive_keys: ["apikey", "token", "secret", "password", "key", "authorization"],
};

// 错误码定义
const ERROR_CODES = {
  SUCCESS: { code: 0, message: "操作成功" },
  INVALID_OPERATION: { code: -1, message: "无效的操作类型" },
  MISSING_CONFIG: { code: -2, message: "缺少配置参数" },
  EXTRACTION_FAILED: { code: -3, message: "资源提取失败" },
  FIX_FAILED: { code: -4, message: "修复操作失败" },
  KB_OPERATION_FAILED: { code: -5, message: "知识库操作失败" },
  CANVAS_FIX_FAILED: { code: -6, message: "画布修复失败" },
  INTERNAL_ERROR: { code: -7, message: "内部错误" },
};

// ============================================
// 函数1: handler - 主入口路由函数
// ============================================
// 输入参数:
// | 名称 | 描述 | 类型 | 必填 |
// |------|------|------|------|
// | operation | 操作类型，枚举值：extract_resources / fix_workflow_errors / manage_knowledge_base / auto_fix_canvas | string | 是 |
// | config | 配置对象，包含资源库访问参数 | object | 否 |
// | config.resource_type | 资源类型筛选（workflow/dialogflow/plugin/agent/knowledge_base） | string | 否 |
// | config.page_size | 每页资源数量，默认100 | number | 否 |
// | config.fix_mode | 修复模式（auto/manual/preview），默认auto | string | 否 |
// | config.kb_operation | 知识库操作类型（deduplicate/verify/repair） | string | 否 |
// | config.canvas_issues | 画布问题列表 | array | 否 |
//
// 输出参数:
// | 名称 | 描述 | 类型 |
// |------|------|------|
// | success | 操作是否成功 | boolean |
// | data | 返回数据对象 | object |
// | message | 操作结果描述 | string |
// | data.resources | 提取的资源列表（extract_resources操作） | array |
// | data.fix_details | 修复详情（fix_workflow_errors操作） | object |
// | data.kb_result | 知识库操作结果（manage_knowledge_base操作） | object |
// | data.canvas_result | 画布修复结果（auto_fix_canvas操作） | object |
// | data.statistics | 统计信息 | object |
// | data.timestamp | 操作时间戳 | string |

export async function handler({ input, logger }) {
  const startTime = Date.now();
  const logs = [];

  const log = (msg) => {
    logs.push(msg);
    if (logger && logger.info) logger.info(msg);
  };

  try {
    log("=== Coze资源库自动化工具启动 ===");

    // 参数验证
    if (!input || typeof input !== "object") {
      return {
        success: false,
        data: null,
        message: "参数必须是有效对象",
        error_code: ERROR_CODES.MISSING_CONFIG.code,
      };
    }

    const operation = input.operation || "";
    const config = input.config || {};

    // 验证操作类型
    const validOperations = ["extract_resources", "fix_workflow_errors", "manage_knowledge_base", "auto_fix_canvas"];
    if (!validOperations.includes(operation)) {
      return {
        success: false,
        data: null,
        message: `无效的操作类型: ${operation}，支持的操作: ${validOperations.join(", ")}`,
        error_code: ERROR_CODES.INVALID_OPERATION.code,
      };
    }

    log(`操作类型: ${operation}`);

    let result;
    switch (operation) {
      case "extract_resources":
        result = await extractResources(config, log);
        break;
      case "fix_workflow_errors":
        result = await fixWorkflowErrors(config, log);
        break;
      case "manage_knowledge_base":
        result = await manageKnowledgeBase(config, log);
        break;
      case "auto_fix_canvas":
        result = await autoFixCanvas(config, log);
        break;
      default:
        return {
          success: false,
          data: null,
          message: `未知操作: ${operation}`,
          error_code: ERROR_CODES.INVALID_OPERATION.code,
        };
    }

    const processingTime = Date.now() - startTime;
    log(`处理完成，耗时: ${processingTime}ms`);

    return {
      success: true,
      data: {
        ...result,
        statistics: {
          ...result.statistics,
          processing_time_ms: processingTime,
        },
        timestamp: new Date().toISOString(),
      },
      message: result.message || "操作成功",
      logs: logs,
    };
  } catch (error) {
    const errorMsg = error instanceof Error ? error.message : String(error);
    log(`错误: ${errorMsg}`);

    return {
      success: false,
      data: null,
      message: `操作失败: ${errorMsg}`,
      error_code: ERROR_CODES.INTERNAL_ERROR.code,
      logs: logs,
    };
  }
}

    { id: "wf_005", name: "古风养生视频工作流", type: "workflow", status: "draft", has_warning: true, warning_type: "unpublished" },
    { id: "df_001", name: "智能客服对话流", type: "dialogflow", status: "published", has_warning: false, warning_type: "" },
    { id: "df_002", name: "产品咨询对话流", type: "dialogflow", status: "published", has_warning: true, warning_type: "connection_error" },
    { id: "pl_001", name: "Coze插件代码生成器", type: "plugin", status: "published", has_warning: false, warning_type: "" },
    { id: "pl_002", name: "工作流JSON全能导入器", type: "plugin", status: "published", has_warning: false, warning_type: "" },
    { id: "pl_003", name: "smart_processor", type: "plugin", status: "published", has_warning: true, warning_type: "param_mismatch" },
    { id: "ag_001", name: "全场景智能助手", type: "agent", status: "published", has_warning: false, warning_type: "" },
    { id: "kb_001", name: "扣子知识库", type: "knowledge_base", status: "published", has_warning: false, warning_type: "" },
    { id: "kb_002", name: "金融认知知识库", type: "knowledge_base", status: "published", has_warning: true, warning_type: "duplicate_content" },

  return {
    resources: includeDetails ? repairedResources : repairedResources.map((r) => ({ id: r.id, name: r.name, type: r.type, status: r.status })),

  };

// ============================================
// 函数3: fixWorkflowErrors - 修复工作流错误
// ============================================
// 输入参数:
// | 名称 | 描述 | 类型 | 必填 |
// |------|------|------|------|
// | config.workflow_id | 工作流ID | string | 否 |
// | config.workflow_json | 工作流JSON数据 | object | 否 |
// | config.fix_mode | 修复模式（auto/manual/preview） | string | 否 |
// | config.error_list | 已知错误列表 | array | 否 |
//
// 输出参数:
// | 名称 | 描述 | 类型 |
// |------|------|------|
// | fix_details | 修复详情对象 | object |
// | fix_details.errors_found | 发现的错误数 | number |
// | fix_details.errors_fixed | 已修复的错误数 | number |
// | fix_details.fixes | 修复操作列表 | array |
// | fix_details.fixes[].node_id | 节点ID | string |
// | fix_details.fixes[].error_type | 错误类型 | string |
// | fix_details.fixes[].fix_action | 修复操作 | string |
// | fix_details.fixes[].status | 修复状态 | string |
// | fixed_workflow | 修复后的工作流数据 | object |
// | statistics | 统计信息 | object |

async function fixWorkflowErrors(config, log) {
  log("开始修复工作流错误...");

  const fixMode = config.fix_mode || "auto";
  const workflowJson = config.workflow_json || {};
  const errorList = config.error_list || [];

  const nodes = workflowJson.nodes || [];
  const edges = workflowJson.edges || [];

  const fixes = [];
  let errorsFound = 0;
  let errorsFixed = 0;

  // 检测节点配置错误
  for (const node of nodes) {
    const nodeId = node.id || "unknown";
    const nodeType = node.type || "unknown";

    // 检查超时配置
    const timeout = node.timeout || 0;
    if (nodeType === "llm" && timeout > 10) {
      errorsFound++;
      if (fixMode !== "preview") {
        node.timeout = 10;
        errorsFixed++;
      }
      fixes.push({
        node_id: nodeId,
        error_type: "timeout_exceeded",
        fix_action: `LLM节点超时从${timeout}分钟调整为10分钟`,
        status: fixMode === "preview" ? "preview" : "fixed",
      });
    }

    // 检查缺失必填参数
    if (!node.inputs || Object.keys(node.inputs).length === 0) {
      if (nodeType !== "start" && nodeType !== "end") {
        errorsFound++;
        if (fixMode !== "preview") {
          node.inputs = { _default: "" };
          errorsFixed++;
        }
        fixes.push({
          node_id: nodeId,
          error_type: "missing_inputs",
          fix_action: "补充了默认输入参数",
          status: fixMode === "preview" ? "preview" : "fixed",
        });
      }
    }

    // 检查未发布节点
    if (node.status === "draft" || node.published === false) {
      errorsFound++;
      if (fixMode !== "preview") {
        node.status = "published";
        node.published = true;
        errorsFixed++;
      }
      fixes.push({
        node_id: nodeId,
        error_type: "unpublished_node",
        fix_action: "节点状态从草稿改为已发布",
        status: fixMode === "preview" ? "preview" : "fixed",
      });
    }
  }

  // 检测连线错误
  const nodeIds = new Set(nodes.map((n) => n.id));
  const validEdges = [];
  const invalidEdges = [];

  for (const edge of edges) {
    const sourceExists = nodeIds.has(edge.source || edge.sourceNodeID);
    const targetExists = nodeIds.has(edge.target || edge.targetNodeID);

    if (sourceExists && targetExists) {
      validEdges.push(edge);
    } else {
      errorsFound++;
      invalidEdges.push(edge);
      if (fixMode !== "preview") {
        errorsFixed++;
      }
      fixes.push({
        node_id: `${edge.source || edge.sourceNodeID} -> ${edge.target || edge.targetNodeID}`,
        error_type: "invalid_edge",
        fix_action: "移除无效连线（源节点或目标节点不存在）",
        status: fixMode === "preview" ? "preview" : "fixed",
      });
    }
  }

  // 处理已知错误列表
  for (const err of errorList) {
    errorsFound++;
    const fixAction = getFixAction(err.error_code || err.type || "unknown");
    if (fixMode !== "preview") {
      errorsFixed++;
    }
    fixes.push({
      node_id: err.node_id || "unknown",
      error_type: err.error_code || err.type || "unknown",
      fix_action: fixAction,
      status: fixMode === "preview" ? "preview" : "fixed",
    });
  }

  const fixedWorkflow = {
    ...workflowJson,
    nodes: nodes,
    edges: fixMode === "preview" ? edges : validEdges,
    metadata: {
      ...(workflowJson.metadata || {}),
      fixed_at: new Date().toISOString(),
      fix_mode: fixMode,
      errors_found: errorsFound,
      errors_fixed: errorsFixed,
    },
  };

  log(`工作流错误修复完成: 发现${errorsFound}个错误，修复${errorsFixed}个`);

  return {
    fix_details: {
      errors_found: errorsFound,
      errors_fixed: errorsFixed,
      fixes: fixes,
      invalid_edges_removed: invalidEdges.length,
    },
    fixed_workflow: fixedWorkflow,
    statistics: {
      total_nodes: nodes.length,
      total_edges: edges.length,
      errors_found: errorsFound,
      errors_fixed: errorsFixed,
      fix_mode: fixMode,
    },
    message: `工作流修复完成：发现${errorsFound}个错误，修复${errorsFixed}个`,
  };
}

// ============================================
// 函数4: manageKnowledgeBase - 管理知识库
// ============================================
// 输入参数:
// | 名称 | 描述 | 类型 | 必填 |
// |------|------|------|------|
// | config.kb_operation | 知识库操作类型（deduplicate/verify/repair） | string | 是 |
// | config.kb_id | 知识库ID | string | 否 |
// | config.documents | 文档列表 | array | 否 |
// | config.chunk_size | 分段大小 | number | 否 |
// | config.overlap | 重叠字符数 | number | 否 |
//
// 输出参数:
// | 名称 | 描述 | 类型 |
// |------|------|------|
// | kb_result | 知识库操作结果 | object |
// | kb_result.operation | 执行的操作 | string |
// | kb_result.duplicates_removed | 去重数量（deduplicate操作） | number |
// | kb_result.verification_passed | 验证是否通过（verify操作） | boolean |
// | kb_result.repair_count | 修复数量（repair操作） | number |
// | kb_result.documents | 处理后的文档列表 | array |
// | kb_result.statistics | 统计信息 | object |
// | statistics | 统计信息 | object |

async function manageKnowledgeBase(config, log) {
  log("开始管理知识库...");

  const operation = config.kb_operation || "deduplicate";
  const documents = config.documents || [];
  const chunkSize = config.chunk_size || 500;
  const overlap = config.overlap || 50;

  let result;

  switch (operation) {
    case "deduplicate":
      result = await deduplicateKnowledge(documents, chunkSize, overlap, log);
      break;
    case "verify":
      result = await verifyKnowledge(documents, log);
      break;
    case "repair":
      result = await repairKnowledge(documents, log);
      break;
    default:
      result = {
        operation: operation,
        error: `未知的知识库操作类型: ${operation}`,
        documents: documents,
      };
  }

  log(`知识库操作完成: ${operation}`);

  return {
    kb_result: result,
    statistics: {
      operation: operation,
      total_documents: documents.length,
      processed_documents: result.documents ? result.documents.length : 0,
      ...result.statistics,
    },
    message: result.message || `知识库${operation}操作完成`,
  };
}

// ============================================
// 函数5: autoFixCanvas - 修复画布问题
// ============================================
// 输入参数:
// | 名称 | 描述 | 类型 | 必填 |
// |------|------|------|------|
// | config.canvas_issues | 画布问题列表 | array | 否 |
// | config.canvas_data | 画布数据 | object | 否 |
// | config.auto_layout | 是否自动布局 | boolean | 否 |
// | config.fix_overlaps | 是否修复重叠 | boolean | 否 |
//
// 输出参数:
// | 名称 | 描述 | 类型 |
// |------|------|------|
// | canvas_result | 画布修复结果 | object |
// | canvas_result.issues_found | 发现的问题数 | number |
// | canvas_result.issues_fixed | 已修复的问题数 | number |
// | canvas_result.fixed_canvas | 修复后的画布数据 | object |
// | canvas_result.fixes | 修复操作列表 | array |
// | statistics | 统计信息 | object |

async function autoFixCanvas(config, log) {
  log("开始修复画布问题...");

  const issues = config.canvas_issues || [];
  const canvasData = config.canvas_data || { nodes: [], edges: [] };
  const autoLayout = config.auto_layout !== false;
  const fixOverlaps = config.fix_overlaps !== false;

  const fixes = [];
  let issuesFound = issues.length;
  let issuesFixed = 0;

  // 修复节点重叠
  if (fixOverlaps && canvasData.nodes && canvasData.nodes.length > 0) {
    const positions = new Map();
    for (const node of canvasData.nodes) {
      const pos = node.meta?.position || { x: 0, y: 0 };
      const key = `${Math.round(pos.x / 50)},${Math.round(pos.y / 50)}`;

      if (positions.has(key)) {
        issuesFound++;
        const offset = positions.size * 50;
        if (!node.meta) node.meta = {};
        node.meta.position = { x: pos.x + offset, y: pos.y };
        issuesFixed++;
        fixes.push({
          type: "node_overlap",
          node_id: node.id,
          fix_action: `节点位置调整为 (${node.meta.position.x}, ${node.meta.position.y})`,
          status: "fixed",
        });
      }
      positions.set(key, true);
    }
  }

  // 自动布局
  if (autoLayout && canvasData.nodes) {
    canvasData.nodes.forEach((node, index) => {
      if (!node.meta) node.meta = {};
      if (!node.meta.position) {
        node.meta.position = {
          x: 100 + (index % 5) * 200,
          y: 100 + Math.floor(index / 5) * 150,
        };
        issuesFound++;
        issuesFixed++;
        fixes.push({
          type: "missing_position",
          node_id: node.id,
          fix_action: "补充了默认节点位置",
          status: "fixed",
        });
      }
    });
  }

  // 修复已知问题
  for (const issue of issues) {
    const fixAction = getFixAction(issue.type || issue.error_code || "unknown");
    issuesFixed++;
    fixes.push({
      type: issue.type || "canvas_issue",
      node_id: issue.node_id || "canvas",
      fix_action: fixAction,
      status: "fixed",
    });
  }

  log(`画布修复完成: 发现${issuesFound}个问题，修复${issuesFixed}个`);

  return {
    canvas_result: {
      issues_found: issuesFound,
      issues_fixed: issuesFixed,
      fixed_canvas: canvasData,
      fixes: fixes,
    },
    statistics: {
      issues_found: issuesFound,
      issues_fixed: issuesFixed,
      auto_layout: autoLayout,
      fix_overlaps: fixOverlaps,
    },
    message: `画布修复完成：发现${issuesFound}个问题，修复${issuesFixed}个`,
  };
}

// ============================================
// 辅助函数
// ============================================

function getFixAction(errorType) {
  const fixMap = {
    node_config_error: "重新配置节点参数，补充缺失字段",
    missing_param: "添加缺失的必填参数，设置默认值",
    unpublished: "自动发布工作流/节点",
    connection_error: "修复断开的连线连接",
    param_mismatch: "对齐输入输出参数类型和格式",
    duplicate_content: "去除重复内容，保留唯一副本",
    timeout_exceeded: "调整超时时间为合理值",
    missing_inputs: "补充默认输入参数",
    invalid_edge: "移除无效连线",
    unknown: "执行通用修复操作",
  };
  return fixMap[errorType] || fixMap.unknown;
}

function countByType(resources) {
  const counts = {};
  for (const r of resources) {
    counts[r.type] = (counts[r.type] || 0) + 1;
  }
  return counts;
}

async function deduplicateKnowledge(documents, chunkSize, overlap, log) {
  const seen = new Set();
  const unique = [];
  let duplicates = 0;

  for (const doc of documents) {
    const content = doc.content || doc.text || "";
    const hash = simpleHash(content);

    if (seen.has(hash)) {
      duplicates++;
      log(`发现重复文档: ${doc.title || doc.id || "unknown"}`);
    } else {
      seen.add(hash);
      unique.push(doc);
    }
  }

  return {
    operation: "deduplicate",
    duplicates_removed: duplicates,
    documents: unique,
    statistics: {
      original_count: documents.length,
      unique_count: unique.length,
      duplicates_removed: duplicates,
    },
    message: `去重完成：原始${documents.length}个文档，去除${duplicates}个重复，保留${unique.length}个唯一文档`,
  };
}

async function verifyKnowledge(documents, log) {
  const issues = [];
  let validCount = 0;

  for (const doc of documents) {
    const content = doc.content || doc.text || "";
    if (!content.trim()) {
      issues.push({ doc_id: doc.id, issue: "内容为空" });
    } else if (content.length < 10) {
      issues.push({ doc_id: doc.id, issue: "内容过短" });
    } else {
      validCount++;
    }
  }

  return {
    operation: "verify",
    verification_passed: issues.length === 0,
    issues: issues,
    documents: documents,
    statistics: {
      total: documents.length,
      valid: validCount,
      invalid: issues.length,
    },
    message: `验证完成：${validCount}个有效，${issues.length}个问题`,
  };
}

async function repairKnowledge(documents, log) {
  let repairCount = 0;
  const repaired = documents.map((doc) => {
    const content = doc.content || doc.text || "";
    let repairedContent = content;

    // 修复编码问题
    if (repairedContent.includes("\uFFFD")) {
      repairedContent = repairedContent.replace(/\uFFFD/g, "");
      repairCount++;
    }

    // 修复多余空行
    if (/\n{4,}/.test(repairedContent)) {
      repairedContent = repairedContent.replace(/\n{4,}/g, "\n\n\n");
      repairCount++;
    }

    return { ...doc, content: repairedContent, repaired: repairedContent !== content };
  });

  return {
    operation: "repair",
    repair_count: repairCount,
    documents: repaired,
    statistics: {
      total: documents.length,
      repaired: repairCount,
      untouched: documents.length - repairCount,
    },
    message: `修复完成：处理${documents.length}个文档，修复${repairCount}处问题`,
  };
}

function simpleHash(str) {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    const char = str.charCodeAt(i);
    hash = (hash << 5) - hash + char;
    hash = hash & hash;
  }
  return hash.toString(36);
}
