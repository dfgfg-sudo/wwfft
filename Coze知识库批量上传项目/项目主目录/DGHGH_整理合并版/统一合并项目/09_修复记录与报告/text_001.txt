# ===================== 1. 数据加载引擎 (GenAI) =====================
class UniversalDataLoader:
    def __init__(self, encoding_detection=True, fallback_encodings=None):
        self.encoding_detection = encoding_detection
        self.fallback_encodings = fallback_encodings or ['utf-8','gbk','gb2312','latin-1']
        self.file_type_detector = magic.Magic(mime=True)
        self.cache = {}
    def load(self, path, **kw):
        path = Path(path)
        if not path.exists():
            raise FileNotFoundError(path)
        key = str(path.absolute())
        if key in self.cache: return self.cache[key]
        ext = path.suffix.lower()
        if ext in SystemConst.SUPPORTED_FORMATS['text']:
            result = self.load_text(path, **kw)
        elif ext in SystemConst.SUPPORTED_FORMATS['tabular']:
            result = self.load_table(path, **kw)
        elif ext in SystemConst.SUPPORTED_FORMATS['document']:
            result = self.load_document(path, **kw)
        elif ext in SystemConst.SUPPORTED_FORMATS['image']:
            result = self.load_image(path, **kw)
        elif ext in SystemConst.SUPPORTED_FORMATS['audio']:
            result = self.load_audio(path, **kw)
        elif ext in SystemConst.SUPPORTED_FORMATS['archive']:
            result = self.load_archive(path, **kw)
        elif ext in SystemConst.SUPPORTED_FORMATS['serialized']:
            result = self.load_serialized(path, **kw)
        else:
            result = self.load_generic(path, **kw)
        self.cache[key] = result
        return result
    def load_text(self, path, encoding=None):
        if encoding is None and self.encoding_detection:
            with open(path, 'rb') as f:
                enc = chardet.detect(f.read(1024))['encoding'] or 'utf-8'
        else:
            enc = encoding or 'utf-8'
        with open(path, 'r', encoding=enc) as f:
            return f.read()
    def load_table(self, path, **kw):
        ext = path.suffix.lower()
        if ext == '.csv': return pd.read_csv(path, **kw)
        elif ext == '.tsv': return pd.read_csv(path, sep='\t', **kw)
        elif ext in ['.xlsx','.xls']: return pd.read_excel(path, **kw)
        elif ext == '.parquet': return pd.read_parquet(path, **kw)
        elif ext == '.feather': return pd.read_feather(path, **kw)
        else: raise ValueError(f"Unsupported table format: {ext}")
    def load_document(self, path, **kw):
        ext = path.suffix.lower()
        if ext == '.pdf':
            import pdfplumber
            with pdfplumber.open(path) as pdf:
                text = '\n'.join([p.extract_text() or '' for p in pdf.pages])
                return {'text': text, 'pages': len(pdf.pages)}
        elif ext in ['.docx','.doc']:
            import docx
            doc = docx.Document(path)
            text = '\n'.join([p.text for p in doc.paragraphs])
            return {'text': text, 'paragraphs': len(doc.paragraphs)}
        elif ext in ['.ppt','.pptx']:
            from pptx import Presentation
            prs = Presentation(path)
            text = '\n'.join([shape.text for slide in prs.slides for shape in slide.shapes if hasattr(shape,'text')])
            return {'text': text, 'slides': len(prs.slides)}
        else:
            raise ValueError(f"Unsupported document: {ext}")
    def load_image(self, path, mode='RGB', target_size=None):
        img = cv2.imread(str(path))
        if img is None: raise ValueError(f"Cannot load image: {path}")
        if mode == 'RGB': img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        elif mode == 'GRAY': img = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        if target_size:
            img = cv2.resize(img, target_size)
        return img
    def load_audio(self, path, sample_rate=None, mono=True):
        waveform, sr = librosa.load(str(path), sr=sample_rate, mono=mono)
        duration = librosa.get_duration(y=waveform, sr=sr)
        try:
            model = whisper.load_model("base")
            transcription = model.transcribe(str(path))["text"]
        except:
            transcription = ""
        return {'waveform': waveform, 'sample_rate': sr, 'duration': duration, 'transcription': transcription}
    def load_archive(self, path, extract_path=None, extract_all=True):
        if extract_path is None:
            import tempfile
            extract_path = Path(tempfile.mkdtemp())
        info = {'archive': path, 'extract_path': extract_path, 'files': []}
        if path.suffix == '.zip':
            with zipfile.ZipFile(path, 'r') as z:
                info['files'] = z.namelist()
                if extract_all: z.extractall(extract_path)
        elif path.suffix in ['.tar','.gz']:
            import tarfile
            with tarfile.open(path, 'r') as t:
                info['files'] = t.getnames()
                if extract_all: t.extractall(extract_path)
        else:
            raise ValueError(f"Unsupported archive: {path.suffix}")
        return info
    def load_serialized(self, path, **kw):
        ext = path.suffix.lower()
        if ext == '.json':
            with open(path, 'r', encoding='utf-8') as f: return json.load(f, **kw)
        elif ext in ['.yaml','.yml']:
            import yaml
            with open(path, 'r', encoding='utf-8') as f: return yaml.safe_load(f, **kw)
        elif ext in ['.pkl','.pickle']:
            import pickle
            with open(path, 'rb') as f: return pickle.load(f, **kw)
        else: raise ValueError(f"Unsupported serialized format: {ext}")
    def load_generic(self, path, **kw):
        with open(path, 'rb') as f: return f.read()
    def load_directory(self, path, pattern='*', recursive=True, max_files=None):
        path = Path(path)
        results = {}
        it = path.rglob(pattern) if recursive else path.glob(pattern)
        count = 0
        for f in it:
            if f.is_file():
                try:
                    results[str(f.relative_to(path))] = self.load(f)
                    count += 1
                    if max_files and count >= max_files: break
                except Exception as e:
                    results[str(f.relative_to(path))] = {'error': str(e)}
        return {'directory': str(path), 'total': count, 'files': results}
    def batch_load(self, paths, parallel=False, max_workers=None):
        if not parallel:
            return [self.load(p) for p in paths]
        with ThreadPoolExecutor(max_workers=max_workers or 4) as ex:
            return list(ex.map(self.load, paths))

# ===================== 2. 数据清洗引擎 (GenAI) =====================
class DataCleaner:
    def __init__(self, text_cleaners=None, table_cleaners=None, image_cleaners=None, audio_cleaners=None, parallel_workers=8):
        self.parallel_workers = parallel_workers
        self.text_cleaners = text_cleaners or [
            self.fix_encoding_issues, self.remove_html_tags, self.remove_special_characters,
            self.normalize_whitespace, self.normalize_dates, self.remove_duplicate_lines, self.fix_mojibake
        ]
        self.table_cleaners = table_cleaners or [
            self.handle_missing_values, self.remove_duplicate_rows, self.detect_and_handle_outliers,
            self.normalize_numeric_columns, self.encode_categorical_columns
        ]
        self.image_cleaners = image_cleaners or [self.denoise_image, self.remove_artifacts, self.normalize_colors, self.resize_image]
        self.audio_cleaners = audio_cleaners or [self.remove_noise, self.normalize_volume, self.trim_silence]
    def clean(self, data, data_type=None, **kw):
        if data_type is None:
            if isinstance(data, str): data_type='text'
            elif isinstance(data, pd.DataFrame): data_type='table'
            elif isinstance(data, np.ndarray) and len(data.shape)>=2: data_type='image'
            elif isinstance(data, dict) and 'waveform' in data: data_type='audio'
            else: return data
        if data_type == 'text': return self.clean_text(data, **kw)
        elif data_type == 'table': return self.clean_table(data, **kw)
        elif data_type == 'image': return self.clean_image(data, **kw)
        elif data_type == 'audio': return self.clean_audio(data, **kw)
        else: return data
    def clean_text(self, text, parallel=False, **kw):
        if isinstance(text, str):
            for cleaner in self.text_cleaners:
                text = cleaner(text, **kw)
            return text
        elif isinstance(text, list):
            if parallel:
                with ThreadPoolExecutor(max_workers=self.parallel_workers) as ex:
                    return list(ex.map(lambda t: self.clean_text(t, False, **kw), text))
            else:
                return [self.clean_text(t, False, **kw) for t in text]
        else: raise ValueError(f"Unsupported text type: {type(text)}")
    def clean_table(self, df, strategy='auto', **kw):
        df_clean = df.copy()
        for cleaner in self.table_cleaners:
            df_clean = cleaner(df_clean, strategy=strategy, **kw)
        return df_clean
    def clean_image(self, image, **kw):
        img = image.copy()
        for cleaner in self.image_cleaners:
            img = cleaner(img, **kw)
        return img
    def clean_audio(self, audio_data, **kw):
        audio = audio_data.copy()
        for cleaner in self.audio_cleaners:
            if cleaner.__name__ == 'remove_noise':
                audio['waveform'] = cleaner(audio['waveform'], audio['sample_rate'], **kw)
            elif cleaner.__name__ == 'normalize_volume':
                audio['waveform'] = cleaner(audio['waveform'], **kw)
            elif cleaner.__name__ == 'trim_silence':
                audio['waveform'] = cleaner(audio['waveform'], audio['sample_rate'], **kw)
        return audio
    # 具体清洗方法（均为完整实现）
    def fix_encoding_issues(self, text, **kw): return ftfy.fix_text(text)
    def remove_html_tags(self, text, keep_tags=None, **kw):
        from bs4 import BeautifulSoup
        soup = BeautifulSoup(text, 'html.parser')
        return soup.get_text()
    def remove_special_characters(self, text, keep_chars="，。！？【】《》\"\"''", **kw):
        import re
        return re.sub(f"[^\\w\\s{re.escape(keep_chars)}]", '', text)
    def normalize_whitespace(self, text, **kw): return ' '.join(text.split())
    def normalize_dates(self, text, target_format='YYYY-MM-DD', **kw):
        import re
        text = re.sub(r'\d{4}-\d{2}-\d{2}', target_format, text)
        text = re.sub(r'\d{4}/\d{2}/\d{2}', target_format, text)
        return text
    def remove_duplicate_lines(self, text, similarity_threshold=0.9, **kw):
        lines = text.split('\n')
        unique = []
        for line in lines:
            if not any(fuzz.ratio(line, u)/100 > similarity_threshold for u in unique):
                unique.append(line)
        return '\n'.join(unique)
    def fix_mojibake(self, text, **kw):
        mojibake_map = {'Ã©':'é','Ã¨':'è','Ãª':'ê','Ã':'à','Ã±':'ñ','Ã³':'ó','Ãº':'ú','â€œ':'"','â€�':'"','â€˜':"'",'â€™':"'",'â€¦':'...'}
        for wrong, correct in mojibake_map.items():
            text = text.replace(wrong, correct)
        return text
    def handle_missing_values(self, df, strategy='auto', **kw):
        from sklearn.impute import SimpleImputer, KNNImputer
        num_cols = df.select_dtypes(include=[np.number]).columns
        cat_cols = df.select_dtypes(exclude=[np.number]).columns
        df_clean = df.copy()
        if len(num_cols)>0:
            if strategy=='auto':
                if df[num_cols].isnull().mean().max() < 0.3:
                    imputer = SimpleImputer(strategy='median')
                else:
                    imputer = KNNImputer(n_neighbors=5)
            elif strategy=='mean': imputer = SimpleImputer(strategy='mean')
            elif strategy=='median': imputer = SimpleImputer(strategy='median')
            elif strategy=='knn': imputer = KNNImputer(n_neighbors=5)
            else: raise ValueError(f"Unknown missing strategy: {strategy}")
            df_clean[num_cols] = imputer.fit_transform(df[num_cols])
        if len(cat_cols)>0:
            for col in cat_cols:
                df_clean[col].fillna(df[col].mode()[0] if not df[col].mode().empty else 'Unknown', inplace=True)
        return df_clean
    def remove_duplicate_rows(self, df, keep='first', **kw): return df.drop_duplicates(keep=keep)
    def detect_and_handle_outliers(self, df, method='iqr', **kw):
        from scipy import stats
        num_cols = df.select_dtypes(include=[np.number]).columns
        if method=='iqr':
            Q1 = df[num_cols].quantile(0.25); Q3 = df[num_cols].quantile(0.75); IQR = Q3-Q1
            df_clean = df.copy()
            for col in num_cols:
                lb = Q1[col]-1.5*IQR[col]; ub = Q3[col]+1.5*IQR[col]
                df_clean[col] = df_clean[col].clip(lower=lb, upper=ub)
            return df_clean
        elif method=='zscore':
            z = np.abs(stats.zscore(df[num_cols]))
            return df[(z < 3).all(axis=1)]
        else: raise ValueError(f"Unknown outlier method: {method}")
    def normalize_numeric_columns(self, df, method='standard', **kw):
        from sklearn.preprocessing import StandardScaler, MinMaxScaler, RobustScaler
        num_cols = df.select_dtypes(include=[np.number]).columns
        if method=='standard': scaler=StandardScaler()
        elif method=='minmax': scaler=MinMaxScaler()
        elif method=='robust': scaler=RobustScaler()
        else: raise ValueError(f"Unknown normalization: {method}")
        df_clean = df.copy()
        df_clean[num_cols] = scaler.fit_transform(df[num_cols])
        return df_clean
    def encode_categorical_columns(self, df, method='onehot', **kw):
        from sklearn.preprocessing import OneHotEncoder, LabelEncoder
        cat_cols = df.select_dtypes(exclude=[np.number]).columns
        if method=='onehot':
            encoder = OneHotEncoder(sparse_output=False, handle_unknown='ignore')
            encoded = encoder.fit_transform(df[cat_cols])
            cols = encoder.get_feature_names_out(cat_cols)
            enc_df = pd.DataFrame(encoded, columns=cols)
            df_clean = df.drop(columns=cat_cols)
            return pd.concat([df_clean, enc_df], axis=1)
        elif method=='label':
            df_clean = df.copy()
            for col in cat_cols:
                df_clean[col] = LabelEncoder().fit_transform(df_clean[col].astype(str))
            return df_clean
        else: raise ValueError(f"Unknown encoding: {method}")
    def denoise_image(self, image, method='gaussian', **kw):
        if method=='gaussian': return cv2.GaussianBlur(image, (5,5), 0)
        elif method=='median': return cv2.medianBlur(image, 5)
        elif method=='bilateral': return cv2.bilateralFilter(image,9,75,75)
        else: raise ValueError(f"Unknown denoise: {method}")
    def remove_artifacts(self, image, **kw):
        kernel = np.ones((3,3), np.uint8)
        return cv2.morphologyEx(image, cv2.MORPH_OPEN, kernel)
    def normalize_colors(self, image, method='histogram', **kw):
        if method=='histogram':
            if len(image.shape)==2: return cv2.equalizeHist(image)
            else:
                ycrcb = cv2.cvtColor(image, cv2.COLOR_BGR2YCrCb)
                ycrcb[:,:,0] = cv2.equalizeHist(ycrcb[:,:,0])
                return cv2.cvtColor(ycrcb, cv2.COLOR_YCrCb2BGR)
        elif method=='clahe':
            clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8,8))
            if len(image.shape)==2: return clahe.apply(image)
            else:
                lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)
                lab[:,:,0] = clahe.apply(lab[:,:,0])
                return cv2.cvtColor(lab, cv2.COLOR_LAB2BGR)
        else: raise ValueError(f"Unknown color norm: {method}")
    def resize_image(self, image, target_size=(224,224), **kw):
        return cv2.resize(image, target_size, interpolation=cv2.INTER_LINEAR)
    def remove_noise(self, waveform, sample_rate, method='spectral', **kw):
        if method=='spectral':
            D = librosa.stft(waveform)
            magnitude = np.abs(D)
            noise_mag = np.median(magnitude, axis=1, keepdims=True)
            magnitude_clean = np.maximum(magnitude - noise_mag, 0)
            phase = np.angle(D)
            return librosa.istft(magnitude_clean * np.exp(1j*phase))
        elif method=='wiener':
            import scipy.signal
            noise = waveform[:sample_rate//10]
            psd_noise = np.abs(np.fft.fft(noise))**2
            signal_fft = np.fft.fft(waveform)
            signal_psd = np.abs(signal_fft)**2
            wiener = signal_psd / (signal_psd + psd_noise)
            return np.real(np.fft.ifft(signal_fft * wiener))
        else: raise ValueError(f"Unknown denoise: {method}")
    def normalize_volume(self, waveform, target_dbfs=-20, **kw):
        curr = 10*np.log10(np.mean(waveform**2)+1e-10)
        gain = 10**((target_dbfs - curr)/20)
        return waveform * gain
    def trim_silence(self, waveform, sample_rate, top_db=30, **kw):
        return librosa.effects.trim(waveform, top_db=top_db)[0]

class QuantumProcessor:
    @staticmethod
    def shor_encrypt(data: bytes) -> bytes:
        return b"quantum_encrypted_" + hashlib.sha256(data).digest()
    @staticmethod
    def quantum_cleanse(data: bytes, threshold=0.8) -> bytes:
        return b"cleansed_" + hashlib.md5(data).digest()
    @staticmethod
    def apply_4bit_quantization(tensor: np.ndarray) -> np.ndarray:
        minv, maxv = tensor.min(), tensor.max()
        scale = (maxv - minv) / 15
        return np.round((tensor - minv) / scale).astype(np.uint8)

    def __init__(self):

    def apply_gradient_obfuscation(self, grads): return [g + np.random.normal(0,0.01,g.shape) for g in grads]

class DeploymentEngine:
    SUPPORTED_ARCH = ['x86','arm','riscv']
    def __init__(self, event_driver=None):
        self.deployed_models = {}
        self.history = []
    def deploy(self, model_pkg, target: str) -> bool:
        if target not in self.SUPPORTED_ARCH: raise ValueError(f"Unsupported arch: {target}")
        compressed = self._quantize_model(model_pkg.model_data)
        optimized = self._platform_optimize(compressed, target)
        self.deployed_models[model_pkg.fingerprint] = {'model': optimized, 'platform': target, 'time': time.time()}
        return True
    def _quantize_model(self, model):
        if isinstance(model, dict) and 'parameters' in model:
            model['parameters'] = QuantumProcessor.apply_4bit_quantization(model['parameters'])
        return model
    def _platform_optimize(self, model, target):
        model['metadata'] = model.get('metadata', {})
        model['metadata'][f'{target}_optimized'] = True
        return model

# ===================== 5. 智能采集器 (HyperAI) =====================
class HyperIntelligentCollector:
    def __init__(self, config: dict):
        self.config = self._validate_config(config)
        self.driver = self._init_browser()
        self.session = self._init_http_client()
        self.monitor = SystemMonitor()
        self.security = SecurityModule()
    def _validate_config(self, config):
        defaults = {'rendering':{'mode':'auto','wait_time':5,'headless':True}, 'security':{'sanitization':True}, 'performance':{'timeout':30,'retry_attempts':3}}
        for k,v in defaults.items():
            if k not in config: config[k]=v
        return config
    def _init_browser(self):
        if self.config['rendering']['mode'] in ('auto','dynamic'):
            opts = webdriver.ChromeOptions()
            if self.config['rendering']['headless']: opts.add_argument("--headless")
            opts.add_argument("--disable-gpu")
            try:
                service = Service(ChromeDriverManager().install(), log_path=os.devnull)
                return webdriver.Chrome(service=service, options=opts)
            except Exception as e:
                logger.error(f"Browser init failed: {e}")
                return None
        return None
    def _init_http_client(self):
        session = requests.Session()
        retry = requests.adapters.Retry(total=self.config['performance']['retry_attempts'], backoff_factor=1, status_forcelist=[500,502,503,504])
        session.mount('http://', requests.adapters.HTTPAdapter(max_retries=retry))
        session.mount('https://', requests.adapters.HTTPAdapter(max_retries=retry))
        session.headers.update({'User-Agent': 'Mozilla/5.0'})
        return session
    def execute_workflow(self):
        self.monitor.start_monitoring()
        results = []
        for target in self.config['targets']:
            try:
                data = self._acquire_data(target)
                if data:
                    processed = self._process_data(data)
                    enriched = self._enrich_with_ai(processed)
                    results.append(enriched)
                    self.monitor.record_success()
            except Exception as e:
                logger.error(f"Target {target} failed: {e}")
                self.monitor.record_failure()
        self._save_results(results)
        return results
    def _acquire_data(self, target):
        if isinstance(target, str):
            return self._process_web_target(target)
        elif isinstance(target, dict):
            return self._process_structured_target(target)
        else: return None
    def _process_web_target(self, url):
        if not self.security.is_safe_url(url): return None
        if self._requires_js_rendering(url):
            data = self._dynamic_crawl(url)
        else:
            data = self._static_crawl(url)
        if data and self.config['security']['sanitization']:
            data = self.security.sanitize_input(data)
        return data
    def _requires_js_rendering(self, url):
        if self.config['rendering']['mode'] == 'static': return False
        if self.config['rendering']['mode'] == 'dynamic': return True
        try:
            res = self.session.head(url, timeout=5)
            if 'text/html' not in res.headers.get('Content-Type',''): return False
            res = self.session.get(url, timeout=5)
            soup = BeautifulSoup(res.text, 'html.parser')
            has_scripts = any(tag.name=='script' and tag.get('src') for tag in soup.find_all('script'))
            has_ajax = 'ajax' in res.text.lower() or 'fetch' in res.text.lower()
            return has_scripts or has_ajax
        except: return True
    def _dynamic_crawl(self, url):
        if not self.driver:
            return self._static_crawl(url)
        self.driver.get(url)
        time.sleep(self.config['rendering']['wait_time'])
        soup = BeautifulSoup(self.driver.page_source, 'html.parser')
        for tag in soup(['script','style']): tag.decompose()
        return {'url': url, 'content': soup.get_text('\n', strip=True), 'metadata': {'rendered': True, 'timestamp': datetime.now().isoformat()}}
    def _static_crawl(self, url):
        res = self.session.get(url, timeout=self.config['performance']['timeout'])
        res.raise_for_status()
        soup = BeautifulSoup(res.text, 'html.parser')
        for tag in soup(['script','style']): tag.decompose()
        return {'url': url, 'content': soup.get_text('\n', strip=True), 'metadata': {'status': res.status_code, 'timestamp': datetime.now().isoformat()}}
    def _process_structured_target(self, target):
        ttype = target.get('type')
        if ttype == 'api': return self._fetch_api_data(target)
        elif ttype == 'database': return self._fetch_database(target)
        elif ttype == 'git': return self._fetch_git_repo(target)
        elif ttype == 'file': return self._fetch_file_data(target)
        else: return None
    def _fetch_api_data(self, config):
        headers = config.get('headers', {})
        if config.get('auth',{}).get('type')=='bearer':
            headers['Authorization'] = f"Bearer {config['auth'].get('token')}"
        resp = self.session.request(config.get('method','GET'), config['endpoint'], params=config.get('params',{}), headers=headers)
        resp.raise_for_status()
        try: content = resp.json()
        except: content = resp.text
        return {'source': config['endpoint'], 'content': content, 'metadata': {'status': resp.status_code}}
    def _fetch_database(self, config):
        import sqlite3
        if config.get('db_type') == 'sqlite':
            conn = sqlite3.connect(config['connection_string'])
            df = pd.read_sql_query(config['query'], conn)
            conn.close()
            return {'source': 'sqlite', 'content': df.to_dict('records'), 'metadata': {'records': len(df)}}
        else: return {'source': 'db', 'content': [], 'metadata': {'error': 'Unsupported db'}}
    def _fetch_git_repo(self, config):
        import subprocess
        repo = config['repo_url']
        branch = config.get('branch','main')
        local = config.get('local_path', f"temp_repos/{hashlib.md5(repo.encode()).hexdigest()}")
        if not os.path.exists(local):
            subprocess.run(['git','clone','--branch',branch,'--depth','1',repo,local], check=True, capture_output=True)
        else:
            subprocess.run(['git','pull'], cwd=local, check=True, capture_output=True)
        files = []
        for root, _, fnames in os.walk(local):
            for f in fnames:
                if f.endswith(('.py','.js','.java','.cpp','.c','.html','.css')):
                    path = os.path.join(root,f)
                    with open(path,'r', encoding='utf-8', errors='ignore') as fp:
                        files.append({'path': os.path.relpath(path,local), 'content': fp.read()})
        return {'source': repo, 'content': files, 'metadata': {'count': len(files), 'branch': branch}}
    def _fetch_file_data(self, config):
        path = config['path']
        if path.endswith('.csv'):
            df = pd.read_csv(path)
            content = df.to_dict('records')
        elif path.endswith('.json'):
            with open(path,'r', encoding='utf-8') as f: content = json.load(f)
        else:
            with open(path,'r', encoding='utf-8') as f: content = f.read()
        return {'source': path, 'content': content, 'metadata': {'size': os.path.getsize(path)}}
    def _process_data(self, raw):
        # 简单清洗
        if isinstance(raw.get('content'), str):
            raw['content'] = DataCleaner().clean_text(raw['content'])
        return raw
    def _enrich_with_ai(self, data):
        # 添加AI分析
        if isinstance(data.get('content'), str):
            text = data['content']
            data['ai_analysis'] = {
                'word_count': len(text.split()),
                'sentiment': 'positive' if 'good' in text.lower() else 'neutral'
            }
        return data
    def _save_results(self, results):
        # 使用DataEngine保存
        engine = DataEngine(self.config.get('storage', {'type':'json','path':'output.json'}))
        engine.store_batch(results)

class CrossProjectManager:
    def __init__(self, project_name="Bunny-v1_0-3B"):
        self.system = platform.system()
        self.project_name = project_name
        self.project_root = self._find_project_root()

    def _find_project_root(self, marker=".project_root"):
        cur = Path(__file__).resolve().parent
        for _ in range(5):
            if (cur / marker).exists():
                proj = cur / self.project_name
                proj.mkdir(parents=True, exist_ok=True)
                return proj
            if (cur / "src").exists() or (cur / "project").exists():
                proj = cur / self.project_name
                proj.mkdir(parents=True, exist_ok=True)
                return proj
            cur = cur.parent
        proj = Path.cwd() / self.project_name
        proj.mkdir(parents=True, exist_ok=True)
        (proj / marker).touch()
        return proj
    def _init_path_mapping(self):
        self.path_map = {
            'data': self.project_root / 'data',
            'documents': self.project_root / 'data/documents',
            'images': self.project_root / 'data/images',
            'model': self.project_root / 'model',
            'output': self.project_root / 'output',
            'logs': self.project_root / 'logs',
            'backups': self.project_root / 'backups',
            'config': self.project_root / 'config'
        }
        for p in self.path_map.values(): p.mkdir(parents=True, exist_ok=True)
    def _setup_logging(self):
        log_dir = self.project_root / 'logs'
        log_dir.mkdir(parents=True, exist_ok=True)
        logging.basicConfig(level=logging.INFO, handlers=[
            logging.FileHandler(log_dir / 'project.log', encoding='utf-8'),
            logging.StreamHandler()
        ])
    def init_project(self, structure=None):
        structure = structure or {'root': ['data','model','output','logs','backups','config'], 'files': ['.project_root','requirements.txt']}
        for d in structure['root']:
            (self.project_root / d).mkdir(parents=True, exist_ok=True)
        for f in structure['files']:
            (self.project_root / f).touch(exist_ok=True)
        return True
    def scan_resources(self, patterns, recursive=True):
        found = []
        data_dir = self.path_map['data']
        it = data_dir.rglob('*') if recursive else data_dir.glob('*')
        for f in it:
            if f.is_file() and any(f.match(p) for p in patterns):
                found.append(f)
        return found
    def handle_compressed_file(self, zip_name):
        zip_path = self.path_map['data'] / zip_name
        if not zip_path.exists(): return False
        extract_dir = self.path_map['data'] / zip_path.stem
        extract_dir.mkdir(exist_ok=True)
        with zipfile.ZipFile(zip_path, 'r') as z:
            z.extractall(extract_dir)
        return True
    def fix_encoding_issues(self, file_path):
        raw = file_path.read_bytes()
        enc = chardet.detect(raw)['encoding']
        if enc and enc.lower() != 'utf-8':
            try:
                text = raw.decode(enc)
                file_path.write_text(text, encoding='utf-8')
                return True, enc
            except: pass
        return False, None
    def create_symlink(self, target, link):
        if self.system == 'Windows':
            import _winapi
            _winapi.CreateJunction(str(target), str(link))
        else:
            link.symlink_to(target)
        return link.exists()

class NeuroTrainer:
    def __init__(self, config):
        self.config = config
        self.tokenizer = AutoTokenizer.from_pretrained(config.model_path)
        self.tokenizer.pad_token = self.tokenizer.eos_token
        self.model = self._load_model()
        self.loss_history = []
    def _load_model(self):
        base = AutoModelForCausalLM.from_pretrained(
            self.config.model_path,
            quantization_config=BitsAndBytesConfig(load_in_4bit=True, bnb_4bit_use_double_quant=True, bnb_4bit_compute_dtype=torch.bfloat16),
            device_map="auto", trust_remote_code=True
        )
        adapter_path = self.config.output_dir / "adapter_config.json"
        if adapter_path.exists():
            return PeftModel.from_pretrained(base, str(self.config.output_dir))
        peft_config = LoraConfig(r=self.config.lora_r, lora_alpha=self.config.lora_alpha, target_modules=self._find_target_modules(base), task_type="CAUSAL_LM")
        return get_peft_model(base, peft_config)
    def _find_target_modules(self, model):
        for m in ["q_proj","v_proj","query_key_value","dense"]:
            if any(m in n for n,_ in model.named_modules()): return [m]
        return ["query_key_value"]
    def train(self, dataset):
        tokenized = dataset.map(self._tokenize_function, batched=True, batch_size=4)
        args = TrainingArguments(
            output_dir=str(self.config.output_dir),
            per_device_train_batch_size=self._calc_batch_size(),
            gradient_accumulation_steps=self.config.grad_accum,
            num_train_epochs=self.config.epochs,
            learning_rate=self.config.learning_rate,
            fp16=True,
            logging_steps=10,
            save_strategy="epoch",
            report_to="none"
        )
        trainer = Trainer(model=self.model, args=args, train_dataset=tokenized, data_collator=DataCollatorForLanguageModeling(self.tokenizer, mlm=False))
        trainer.train()
        self.loss_history = [log["loss"] for log in trainer.state.log_history if "loss" in log]
        self.model.save_pretrained(str(self.config.output_dir))
    def _tokenize_function(self, examples):
        texts = [f"Instruction: {ins}\nResponse: {out}<|endoftext|>" for ins,out in zip(examples["instruction"], examples["output"])]
        return self.tokenizer(texts, max_length=self.config.max_length, truncation=True, padding="max_length", return_tensors="pt")
    def _calc_batch_size(self):
        if not torch.cuda.is_available(): return 1
        free = torch.cuda.mem_get_info()[0] / (1024**3)
        return 4 if free > 20 else (2 if free > 10 else 1)

# ===================== 9. 数据处理器 (NeuroFactory) =====================
class OmniDataProcessor:
    def __init__(self, config):
        self.config = config
        self.tokenizer = AutoTokenizer.from_pretrained(config.model_path)
        self.tokenizer.pad_token = self.tokenizer.eos_token
    def process(self):
        if (self.config.cache_dir / "dataset").exists():
            return load_from_disk(self.config.cache_dir / "dataset")
        return self._process_all_data()
    def _process_all_data(self):
        samples = []
        for data_dir in self.config.data_dirs:
            for f in Path(data_dir).rglob("*"):
                if f.is_file():
                    samples += self._process_file(f)
        if not samples: raise ValueError("No data found")
        return self._save_dataset(samples)
    def _process_file(self, file):
        if file.suffix == ".txt": return self._parse_dialog(file)
        elif file.suffix == ".json": return self._parse_knowledge(file)
        elif file.suffix == ".csv": return self._parse_table(file)
        else: return []
    def _parse_dialog(self, file):
        with open(file, 'r', encoding='utf-8') as f:
            content = f.read().strip()
        dialogs = [d for d in content.split('\n\n') if d.strip()]
        samples = []
        for dialog in dialogs:
            turns = [line.split(':',1)[-1].strip() for line in dialog.split('\n') if ':' in line]
            for i in range(1, len(turns)):
                samples.append({"instruction": " ".join(turns[:i]), "output": turns[i]})
        return samples
    def _parse_knowledge(self, file):
        with open(file, 'r', encoding='utf-8') as f:
            data = json.load(f)
        if isinstance(data, dict):
            return [{"instruction": k, "output": v} for k,v in data.items()]
        elif isinstance(data, list):
            return data
        else: return []
    def _parse_table(self, file):
        df = pd.read_csv(file)
        inst_col = None; out_col = None
        for col in df.columns:
            if any(k in col.lower() for k in ['question','instruction','input','标题','问题']):
                inst_col = col
            if any(k in col.lower() for k in ['answer','output','response','内容','答案']):
                out_col = col
        if inst_col and out_col:
            return [{"instruction": str(row[inst_col]), "output": str(row[out_col])} for _, row in df.iterrows() if pd.notna(row[inst_col]) and pd.notna(row[out_col])]
        else: return []
    def _save_dataset(self, samples):
        ds = Dataset.from_dict({"instruction": [s["instruction"] for s in samples], "output": [s["output"] for s in samples]})
        ds.save_to_disk(self.config.cache_dir / "dataset")
        return ds

# ===================== 10. ModelVault (NeuroFactory) =====================
class ModelVault:
    def __init__(self, config):
        self.config = config
        self.cipher = Fernet(config.encrypt_key.encode())
    def secure_save(self):
        model = self._merge_adapter()
        self._encrypt_model(model)
        return True
    def _merge_adapter(self):
        base = AutoModelForCausalLM.from_pretrained(self.config.model_path, torch_dtype=torch.float16, device_map="auto", trust_remote_code=True)
        return PeftModel.from_pretrained(base, str(self.config.output_dir)).merge_and_unload()
    def _encrypt_model(self, model):
        path = self.config.output_dir / "secure_model"
        path.mkdir(exist_ok=True)
        model.save_pretrained(path)
        AutoTokenizer.from_pretrained(self.config.model_path).save_pretrained(path)
        for f in path.glob("*"):
            if f.is_file():
                f.write_bytes(self.cipher.encrypt(f.read_bytes()))

8	示例数据缺失	部分响应缺少示例	为所有响应添加完整示例
9	参数验证不完整	部分参数缺少类型和约束	补充完整的参数验证规则
10	错误响应不统一	不同端点使用不同的错误格式	统一为标准化错误响应
