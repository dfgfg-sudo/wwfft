#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
SuperAI Training Fusion System - 终极整合版
兼容Trae-AI-IDE / Trae-CN / 标准Python
"""

import os
import sys
import argparse
import logging
from pathlib import Path

# 添加src到路径
sys.path.insert(0, str(Path(__file__).parent))

from core.signal_handler import SystemSignalHandler
from core.data_handler import DataHandler
from core.model_core import ModelCore
from core.training_engine import TrainingEngine
from core.system_monitor import SystemMonitor
from ui.web_interface import WebInterface
from ui.cli_interface import CLIInterface
from ui.gui_interface import GUIInterface
from utils.logger import setup_logging, get_logger
from config.config_manager import ConfigManager

logger = get_logger(__name__)

def parse_args():
    parser = argparse.ArgumentParser(description="SuperAI Training Fusion System")
    parser.add_argument("--mode", choices=["web", "cli", "gui", "train", "infer", "deploy", "export"],
                        default="web", help="运行模式")
    parser.add_argument("--config", type=str, default="config.json", help="配置文件")
    parser.add_argument("--port", type=int, default=7860, help="Web端口")
    parser.add_argument("--debug", action="store_true", help="调试模式")
    parser.add_argument("--train_data", type=str, help="训练数据目录")
    parser.add_argument("--prompt", type=str, help="推理提示")
    parser.add_argument("--export_format", type=str, default="onnx", choices=["onnx", "torchscript", "tensorrt"])
    parser.add_argument("--model_path", type=str, help="模型路径")
    return parser.parse_args()

def main():
    args = parse_args()
    setup_logging(level=logging.DEBUG if args.debug else logging.INFO)

    # 加载配置
    config_manager = ConfigManager(args.config)
    config = config_manager.load_config()

    # 初始化核心组件
    signal_handler = SystemSignalHandler()
    data_handler = DataHandler(config)
    model_core = ModelCore(config)
    training_engine = TrainingEngine(config)
    monitor = SystemMonitor()

    # 信号回调注册
    signal_handler.register_callback(lambda: training_engine.stop_training())

    logger.info("🚀 SuperAI Training Fusion System 初始化完成")

    if args.mode == "web":
        ui = WebInterface(config)
        ui.initialize(model_core, training_engine, monitor, data_handler)
        ui.launch(server_port=args.port)
    elif args.mode == "cli":
        cli = CLIInterface(config)
        cli.initialize(model_core, training_engine, monitor, data_handler)
        cli.run()
    elif args.mode == "gui":
        gui = GUIInterface(config)
        gui.initialize(model_core, training_engine, monitor, data_handler)
        gui.run()
    elif args.mode == "train":
        if not args.train_data:
            logger.error("请指定--train_data")
            sys.exit(1)
        model_core.initialize()
        dataset = data_handler.process_directory(Path(args.train_data))
        training_engine.prepare_training(model_core, data_handler, dataset)
        training_engine.start_training()
    elif args.mode == "infer":
        if not args.prompt:
            logger.error("请指定--prompt")
            sys.exit(1)
        model_core.initialize()
        response = model_core.generate(args.prompt)
        print(f"🤖 响应: {response}")
    elif args.mode == "deploy":
        if args.model_path:
            model_core.initialize(args.model_path)
        else:
            model_core.initialize()
        # 启动部署服务
        from deployment.serving import ModelServer
        server = ModelServer(config)
        server.serve_model(model_core.model, model_core.tokenizer)
    elif args.mode == "export":
        model_core.initialize(args.model_path)
        from deployment.exporter import ModelExporter
        exporter = ModelExporter(config)
        exporter.export(model_core.model, args.export_format)

if __name__ == "__main__":
    main()

"""
智能数据处理器 - 支持TXT/JSON/CSV/PDF/DOCX/IMG/代码文件
合并融合版：所有文件格式后缀名相同的内容已合并
"""

import json
import csv
import logging
from pathlib import Path
from typing import Generator, Dict, Any, List, Optional, Union
import pandas as pd
from PIL import Image
from torchvision import transforms
import docx
import PyPDF2
import zipfile
import io

logger = logging.getLogger(__name__)

class DataHandler:
    """智能数据处理器 - 支持所有格式"""
    SUPPORTED_FORMATS = ('.txt', '.json', '.jsonl', '.csv', '.xlsx', '.xls',
                         '.pdf', '.docx', '.doc', '.png', '.jpg', '.jpeg',
                         '.md', '.py', '.java', '.js', '.zip')

    def __init__(self, config: dict):
        self.config = config
        self.knowledge_base = []
        self.file_counter = 0
        self.error_log = []
        self.image_transform = transforms.Compose([
            transforms.Resize((224, 224)),
            transforms.ToTensor(),
            transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
        ])
        self.supported_ext = set(self.SUPPORTED_FORMATS)
        self.task_markers = {
            '.pdf': '[PDF]', '.docx': '[DOC]', '.doc': '[DOC]',
            '.txt': '[TXT]', '.md': '[MD]', '.csv': '[CSV]',
            '.json': '[JSON]', '.py': '[PY]', '.java': '[JAVA]',
            '.js': '[JS]', '.png': '[IMG]', '.jpg': '[IMG]', '.jpeg': '[IMG]'
        }

    def process_directory(self, path: Union[str, Path]) -> List[Dict]:
        """递归扫描目录并处理支持的文件格式"""
        path_obj = Path(path)
        if not path_obj.exists():
            raise FileNotFoundError(f"目录不存在: {path}")

        self.knowledge_base.clear()
        self.file_counter = 0
        self.error_log.clear()

        for file_path in path_obj.rglob('*'):
            if file_path.is_file() and file_path.suffix.lower() in self.supported_ext:
                self._process_file(file_path)

        logger.info(f"处理完成，共 {self.file_counter} 个文件，{len(self.error_log)} 个错误")
        return self.knowledge_base

    def _process_file(self, file_path: Path):
        """处理单个文件"""
        try:
            suffix = file_path.suffix.lower()
            if suffix in ('.png', '.jpg', '.jpeg'):
                self._process_image(file_path)
            elif suffix == '.pdf':
                self._process_pdf(file_path)
            elif suffix in ('.docx', '.doc'):
                self._process_docx(file_path)
            elif suffix in ('.csv', '.xlsx', '.xls'):
                self._process_table(file_path)
            elif suffix in ('.txt', '.md'):
                self._process_text(file_path)
            elif suffix in ('.json', '.jsonl'):
                self._process_json(file_path)
            elif suffix in ('.py', '.java', '.js'):
                self._process_code(file_path)
            elif suffix == '.zip':
                self._process_zip(file_path)
            else:
                logger.warning(f"未处理格式: {suffix}")
                return

            self.file_counter += 1
            logger.debug(f"已处理: {file_path.name}")

        except Exception as e:
            error_msg = f"处理失败 {file_path}: {str(e)}"
            logger.error(error_msg)
            self.error_log.append(error_msg)

    def _process_image(self, path: Path):
        """处理图像文件"""
        img = Image.open(path).convert('RGB')
        tensor = self.image_transform(img)
        self.knowledge_base.append({
            "type": "image",
            "data": tensor.numpy(),
            "source": str(path),
            "format": "image"
        })

    def _process_pdf(self, path: Path):
        """处理PDF文件"""
        text = []
        with open(path, 'rb') as f:
            reader = PyPDF2.PdfReader(f)
            for page in reader.pages:
                page_text = page.extract_text() or ""
                if page_text.strip():
                    text.append(page_text)
        if text:
            self.knowledge_base.append({
                "type": "text",
                "data": f"[PDF] {''.join(text)}",
                "source": str(path),
                "format": "pdf"
            })

        doc = docx.Document(path)
        text = '\n'.join([p.text for p in doc.paragraphs if p.text.strip()])
        if text:
            self.knowledge_base.append({
                "type": "text",
                "data": f"[DOC] {text}",
                "source": str(path),
                "format": "docx"
            })

    def _process_table(self, path: Path):
        """处理表格文件 (CSV/XLSX/XLS)"""
        try:
            if path.suffix.lower() == '.csv':
                df = pd.read_csv(path)
            else:
                df = pd.read_excel(path)
            text = df.to_string(index=False)
            self.knowledge_base.append({
                "type": "table",
                "data": f"[TABLE] {text}",
                "source": str(path),
                "format": path.suffix[1:]

        except Exception as e:

            try:
                with open(path, 'r', encoding=enc) as f:
                    text = f.read()
                break
            except UnicodeDecodeError:
                continue
        else:
            raise UnicodeDecodeError(f"无法解码 {path}")
        self.knowledge_base.append({
            "type": "text",
            "data": text,
            "source": str(path),
            "format": path.suffix[1:]
        })

    def _process_json(self, path: Path):

        with open(path, 'r', encoding='utf-8') as f:
            data = json.load(f)
        if isinstance(data, list):
            for item in data:
                self.knowledge_base.append({
                    "type": "json",
                    "data": json.dumps(item, ensure_ascii=False),
                    "source": str(path),
                    "format": "json"
                })
        elif isinstance(data, dict):
            self.knowledge_base.append({
                "type": "json",
                "data": json.dumps(data, ensure_ascii=False),

                "format": "json"
            })

    def _process_code(self, path: Path):
        """处理代码文件"""
        with open(path, 'r', encoding='utf-8') as f:

        self.knowledge_base.append({
            "type": "code",
            "data": code,
            "source": str(path),
            "format": path.suffix[1:],

        })

    def _process_zip(self, path: Path):
        """处理ZIP压缩文件"""
        try:
            with zipfile.ZipFile(path, 'r') as zf:
                for file_info in zf.infolist():
                    if not file_info.filename.endswith('/'):
                        suffix = Path(file_info.filename).suffix.lower()
                        if suffix in self.supported_ext:
                            with zf.open(file_info) as f:
                                # 创建临时文件处理
                                import tempfile
                                with tempfile.NamedTemporaryFile(delete=False, suffix=suffix) as tmp:
                                    tmp.write(f.read())
                                    tmp.flush()
                                    self._process_file(Path(tmp.name))
                                os.unlink(tmp.name)
        except Exception as e:
            raise Exception(f"ZIP处理失败: {str(e)}")

    def get_processed_count(self) -> int:
        return self.file_counter

    def get_errors(self) -> List[str]:
        return self.error_log.copy()

    def export_data(self, output_path: str, format: str = "json"):

        elif format.lower() == "csv":
            df = pd.DataFrame(self.knowledge_base)
            df.to_csv(output_file, index=False, encoding='utf-8')
        logger.info(f"数据已导出到 {output_path}")

            if self.device.type != "cuda":
                self.model.to(self.device)

            # 配置LoRA

    def _setup_lora(self):

    @lru_cache(maxsize=128)
    def generate(self, prompt: str, max_length: int = 200, temperature: float = 0.7,
                 top_p: float = 0.9, top_k: int = 50, repetition_penalty: float = 1.1) -> str:
        """生成文本"""
        if not self.is_loaded:
            self.initialize()

        try:
            inputs = self.tokenizer(
                prompt,
                return_tensors="pt",
                truncation=True,
                max_length=512
            ).to(self.device)

            generate_kwargs = {
                "max_length": min(max_length + len(inputs.input_ids[0]), 2048),
                "temperature": temperature,
                "top_p": top_p,
                "top_k": top_k,
                "repetition_penalty": repetition_penalty,
                "do_sample": temperature > 0,
                "pad_token_id": self.tokenizer.pad_token_id,
                "eos_token_id": self.tokenizer.eos_token_id
            }

            with torch.no_grad():
                outputs = self.model.generate(**inputs, **generate_kwargs)

            response = self.tokenizer.decode(
                outputs[0][len(inputs.input_ids[0]):],
                skip_special_tokens=True
            )
            return response.strip()

        except Exception as e:
            logger.error(f"文本生成失败: {str(e)}")
            return f"生成错误: {str(e)}"

    # ===== 多模态融合 =====
    def multimodal_fusion(self, text_features: torch.Tensor, image_features: torch.Tensor,
                          audio_features: Optional[torch.Tensor] = None,
                          fusion_strategy: str = "adaptive_weighted") -> torch.Tensor:
        """多模态融合"""
        if fusion_strategy == "adaptive_weighted":
            text_norm = torch.norm(text_features, dim=1).mean()
            image_norm = torch.norm(image_features, dim=1).mean()
            if audio_features is not None:
                audio_norm = torch.norm(audio_features, dim=1).mean()
                total = text_norm + image_norm + audio_norm
                audio_weight = audio_norm / total
            else:
                total = text_norm + image_norm
                audio_weight = 0
            text_weight = text_norm / total
            image_weight = image_norm / total
            fused = text_weight * text_features + image_weight * image_features
            if audio_features is not None:
                fused += audio_weight * audio_features
            return fused
        elif fusion_strategy == "concat":
            if audio_features is not None:
                return torch.cat([text_features, image_features, audio_features], dim=-1)
            return torch.cat([text_features, image_features], dim=-1)
        else:
            return (text_features + image_features) / 2

    # ===== 知识蒸馏 =====
    def compute_distillation_loss(self, student_logits: torch.Tensor,
                                   teacher_logits: torch.Tensor,
                                   labels: torch.Tensor,
                                   temperature: float = 2.0,
                                   alpha: float = 0.7) -> torch.Tensor:
        """计算知识蒸馏损失"""
        kd_loss = nn.KLDivLoss(reduction='batchmean')(
            F.log_softmax(student_logits / temperature, dim=-1),
            F.softmax(teacher_logits / temperature, dim=-1)
        ) * (temperature ** 2)
        ce_loss = nn.CrossEntropyLoss()(student_logits, labels)
        return alpha * kd_loss + (1 - alpha) * ce_loss

    # ===== 持续学习 EWC =====
    def compute_ewc_penalty(self, lambda_ewc: float = 1000.0) -> torch.Tensor:
        """计算EWC惩罚"""
        if self.fisher is None or self.prev_params is None:
            return torch.tensor(0.0, device=self.device)

        loss = 0
        for n, p in self.model.named_parameters():
            if n in self.fisher and n in self.prev_params and p.requires_grad:
                loss += (self.fisher[n] * (p - self.prev_params[n]) ** 2).sum()
        return lambda_ewc * loss

    def update_fisher(self, dataset, batch_size: int = 8):
        """更新Fisher信息矩阵"""
        self.model.train()
        fisher = {}
        for n, p in self.model.named_parameters():
            if p.requires_grad:
                fisher[n] = torch.zeros_like(p)

        # 计算Fisher信息
        for batch in dataset:
            inputs = self._prepare_inputs(batch)
            if not inputs:
                continue
            outputs = self.model(**inputs, labels=inputs["input_ids"])
            loss = outputs.loss
            loss.backward()
            for n, p in self.model.named_parameters():
                if p.grad is not None and n in fisher:
                    fisher[n] += p.grad.data ** 2
            self.model.zero_grad()

        self.fisher = fisher
        self.prev_params = {n: p.clone().detach() for n, p in self.model.named_parameters() if p.requires_grad}
        logger.info("Fisher信息矩阵更新完成")

    # ===== 强化学习 PPO =====
    def ppo_step(self, old_logits: torch.Tensor, actions: torch.Tensor,
                 rewards: torch.Tensor, clip_epsilon: float = 0.2) -> torch.Tensor:
        """PPO单步更新"""
        # 获取新策略logits
        outputs = self.model(actions)  # 简化实现
        new_logits = outputs.logits if hasattr(outputs, 'logits') else outputs

        # 计算概率比率
        old_probs = F.softmax(old_logits, dim=-1)
        new_probs = F.softmax(new_logits, dim=-1)

        # 确保维度匹配
        if actions.dim() == 1:
            actions = actions.unsqueeze(1)
        ratio = new_probs.gather(1, actions) / (old_probs.gather(1, actions) + 1e-8)

        # PPO损失
        surr1 = ratio * rewards
        surr2 = torch.clamp(ratio, 1 - clip_epsilon, 1 + clip_epsilon) * rewards
        policy_loss = -torch.min(surr1, surr2).mean()

        return policy_loss

    def _prepare_inputs(self, batch):
        """准备模型输入"""
        if isinstance(batch, dict):
            return {k: v.to(self.device) for k, v in batch.items()}
        return batch.to(self.device)

"""
自动化训练引擎 - 合并融合版
"""

import os
import time
import json
import logging
from typing import Dict, Any, Optional, List, Callable
import pandas as pd
import torch
from torch.utils.data import Dataset
from transformers import (
    Trainer,
    TrainingArguments,
    DataCollatorForLanguageModeling,
    TrainerCallback
)
from datasets import Dataset as HFDataset

logger = logging.getLogger(__name__)

class StreamDataset(Dataset):
    """流式数据集"""
    def __init__(self, tokenizer, texts, max_length=512):
        self.tokenizer = tokenizer
        self.texts = texts
        self.max_length = max_length
        self.encodings = self._preprocess_texts()

    def _preprocess_texts(self):
        return self.tokenizer(
            self.texts,
            truncation=True,
            max_length=self.max_length,
            padding='max_length',
            return_tensors='pt'
        )

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        return {
            'input_ids': self.encodings['input_ids'][idx],
            'attention_mask': self.encodings['attention_mask'][idx],
            'labels': self.encodings['input_ids'][idx].clone()
        }

class TrainingEngine:
    """自动化训练引擎 - 完整功能版"""

    def __init__(self, config: dict):
        self.config = config
        self.training_config = config.get("training", {})
        self.model_core = None
        self.data_handler = None
        self.trainer = None
        self.dataset = None
        self.is_training = False
        self.should_stop = False
        self.current_epoch = 0
        self.completed_steps = 0
        self.total_steps = 0
        self.loss_history = []

    def prepare_training(self, model_core, data_handler, dataset=None):
        """准备训练"""
        self.model_core = model_core
        self.data_handler = data_handler

        if dataset is None and data_handler:
            # 从数据处理器获取数据
            raw_data = data_handler.knowledge_base
            texts = [item.get('data', '') for item in raw_data if item.get('type') == 'text']
            if not texts:
                raise ValueError("未找到有效的训练数据")

            # 创建数据集
            df = pd.DataFrame({"text": texts})
            hf_dataset = HFDataset.from_pandas(df)
            tokenized = hf_dataset.map(
                self._tokenize_function,
                batched=True,
                remove_columns=hf_dataset.column_names
            )
            self.dataset = tokenized
        else:
            self.dataset = dataset

        # 准备训练参数
        training_args = self._prepare_training_args()

        # 数据整理器
        data_collator = DataCollatorForLanguageModeling(
            tokenizer=self.model_core.tokenizer,
            mlm=False
        )

        # 创建Trainer
        self.trainer = Trainer(
            model=self.model_core.model,
            args=training_args,
            train_dataset=self.dataset,
            data_collator=data_collator,
            callbacks=[self._create_callback()]
        )

        logger.info("训练准备完成")

    def _tokenize_function(self, examples):
        return self.model_core.tokenizer(
            examples["text"],
            truncation=True,
            max_length=512,
            padding=False
        )

    def _prepare_training_args(self) -> TrainingArguments:
        """准备训练参数"""
        output_dir = self.training_config.get("output_dir", "./output")
        model_name = self.training_config.get("model_name", "finetuned_model")

        return TrainingArguments(
            output_dir=os.path.join(output_dir, model_name),
            num_train_epochs=self.training_config.get("num_train_epochs", 3),
            per_device_train_batch_size=self.training_config.get("batch_size", 2),
            gradient_accumulation_steps=self.training_config.get("gradient_accumulation_steps", 4),
            learning_rate=self.training_config.get("learning_rate", 2e-5),
            fp16=self.training_config.get("fp16", True) and torch.cuda.is_available(),
            logging_steps=self.training_config.get("logging_steps", 50),
            save_steps=self.training_config.get("save_steps", 1000),
            save_total_limit=self.training_config.get("save_total_limit", 3),
            evaluation_strategy="no",
            load_best_model_at_end=False,
            report_to="none",
            remove_unused_columns=False,
            warmup_steps=self.training_config.get("warmup_steps", 100),
            weight_decay=self.training_config.get("weight_decay", 0.01),
            lr_scheduler_type=self.training_config.get("lr_scheduler_type", "cosine"),
            dataloader_num_workers=self.training_config.get("num_workers", 0),
            max_grad_norm=self.training_config.get("max_grad_norm", 1.0)
        )

    def _create_callback(self):
        """创建训练回调"""
        class TrainingCallback(TrainerCallback):
            def __init__(self, engine):
                self.engine = engine

            def on_step_end(self, args, state, control, **kwargs):
                self.engine.completed_steps = state.global_step
                self.engine.current_epoch = state.epoch
                if self.engine.should_stop:
                    control.should_training_stop = True
                return control

            def on_log(self, args, state, control, logs=None, **kwargs):
                if logs:
                    self.engine.loss_history.append(logs.get('loss', 0))
                    logger.info(f"训练日志: {logs}")

        return TrainingCallback(self)

    def start_training(self) -> bool:
        """开始训练"""
        if not self.trainer:
            raise ValueError("训练器未初始化，请先调用prepare_training")

        self.is_training = True
        self.should_stop = False
        self.loss_history = []

        try:
            logger.info("开始模型训练...")
            result = self.trainer.train()
            self._save_model()

            logger.info(f"训练完成，最终损失: {result.training_loss}")
            self.is_training = False
            return True

        except Exception as e:
            logger.error(f"训练失败: {str(e)}")
            self.is_training = False
            return False

    def stop_training(self):
        """停止训练"""
        self.should_stop = True
        logger.info("正在停止训练...")

    def _save_model(self):
        """保存模型"""
        if self.model_core and self.trainer:
            save_path = os.path.join(
                self.training_config.get("output_dir", "./output"),
                f"checkpoint-{int(time.time())}"
            )
            self.model_core.save_model(save_path)

    def get_training_status(self) -> Dict[str, Any]:
        """获取训练状态"""
        return {
            "is_training": self.is_training,
            "current_epoch": self.current_epoch,
            "completed_steps": self.completed_steps,
            "total_steps": self.total_steps,
            "should_stop": self.should_stop,
            "loss_history": self.loss_history[-10:] if self.loss_history else []
        }

    def evaluate_model(self, eval_data: Optional[List[Dict]] = None) -> Dict[str, Any]:
        """评估模型"""
        if not self.trainer:
            return {"error": "训练器未初始化"}

        try:
            if eval_data:
                df = pd.DataFrame(eval_data)
                df["text"] = df.apply(lambda x: x.get('data', ''), axis=1)
                eval_dataset = HFDataset.from_pandas(df[["text"]])
                eval_dataset = eval_dataset.map(
                    self._tokenize_function,
                    batched=True,
                    remove_columns=eval_dataset.column_names
                )
            else:
                if self.dataset:
                    eval_dataset = self.dataset.shuffle().select(range(min(100, len(self.dataset))))
                else:
                    return {"error": "没有评估数据"}

            results = self.trainer.evaluate(eval_dataset)
            return results

        except Exception as e:
            logger.error(f"评估失败: {str(e)}")
            return {"error": str(e)}

import psutil
import GPUtil
import time
import logging
from typing import Dict, Any, List

logger = logging.getLogger(__name__)

class SystemMonitor:
    """系统资源监控器"""

    def __init__(self, update_interval: int = 5):
        self.update_interval = update_interval
        self.monitoring_data = []
        self.max_records = 1000

    def get_system_stats(self) -> Dict[str, Any]:
        """获取系统统计信息"""
        try:
            cpu_percent = psutil.cpu_percent(interval=0.1)
            memory = psutil.virtual_memory()
            disk = psutil.disk_usage('/')
            net_io = psutil.net_io_counters()

            stats = {
                "timestamp": time.time(),
                "cpu": {
                    "percent": cpu_percent,
                    "count": psutil.cpu_count(),
                    "frequency": psutil.cpu_freq().current if psutil.cpu_freq() else None
                },
                "memory": {
                    "percent": memory.percent,
                    "used_gb": round(memory.used / (1024**3), 2),
                    "available_gb": round(memory.available / (1024**3), 2),
                    "total_gb": round(memory.total / (1024**3), 2)
                },
                "disk": {
                    "percent": disk.percent,
                    "used_gb": round(disk.used / (1024**3), 2),
                    "free_gb": round(disk.free / (1024**3), 2),
                    "total_gb": round(disk.total / (1024**3), 2)
                },
                "network": {
                    "bytes_sent": net_io.bytes_sent,
                    "bytes_recv": net_io.bytes_recv
                },
                "gpu": self._get_gpu_info(),
                "process": self._get_process_info(),
                "load_average": self._get_load_average()
            }

            self._record_data(stats)
            return stats

        except Exception as e:
            logger.error(f"获取系统状态失败: {str(e)}")
            return {"error": str(e)}

    def _get_gpu_info(self) -> List[Dict[str, Any]]:
        """获取GPU信息"""
        gpu_info = []
        try:
            gpus = GPUtil.getGPUs()
            for gpu in gpus:
                gpu_info.append({
                    "id": gpu.id,
                    "name": gpu.name,
                    "load": round(gpu.load * 100, 2),
                    "memory_used": gpu.memoryUsed,
                    "memory_total": gpu.memoryTotal,
                    "memory_percent": round(gpu.memoryUsed / gpu.memoryTotal * 100, 2),
                    "temperature": gpu.temperature
                })
        except Exception as e:
            logger.warning(f"获取GPU信息失败: {str(e)}")
        return gpu_info

    def _get_process_info(self) -> Dict[str, Any]:
        """获取进程信息"""
        try:
            process = psutil.Process()
            children = process.children(recursive=True)
            return {
                "pid": process.pid,
                "name": process.name(),
                "cpu_percent": process.cpu_percent(),
                "memory_percent": process.memory_percent(),
                "memory_rss_mb": round(process.memory_info().rss / (1024**2), 2),
                "threads": process.num_threads(),
                "children_count": len(children)
            }
        except Exception as e:
            logger.warning(f"获取进程信息失败: {str(e)}")
            return {"error": str(e)}

    def _get_load_average(self) -> List[float]:
        try:
            return [round(x, 2) for x in psutil.getloadavg()]
        except:
            return [0, 0, 0]

    def _record_data(self, stats: Dict[str, Any]):
        self.monitoring_data.append(stats)
        if len(self.monitoring_data) > self.max_records:
            self.monitoring_data = self.monitoring_data[-self.max_records:]

    def get_monitoring_history(self, limit: int = 100) -> List[Dict[str, Any]]:
        return self.monitoring_data[-limit:]

    def check_resource_limits(self, cpu_threshold: float = 90.0,
                              memory_threshold: float = 90.0,
                              gpu_threshold: float = 90.0) -> Dict[str, bool]:
        """检查资源限制"""
        stats = self.get_system_stats()
        warnings = {
            "cpu_warning": stats.get("cpu", {}).get("percent", 0) > cpu_threshold,
            "memory_warning": stats.get("memory", {}).get("percent", 0) > memory_threshold,
            "disk_warning": stats.get("disk", {}).get("percent", 0) > 90.0,
            "gpu_warning": False
        }
        for gpu in stats.get("gpu", []):
            if gpu.get("memory_percent", 0) > gpu_threshold:
                warnings["gpu_warning"] = True
                break
        return warnings

    def get_resource_recommendations(self) -> List[str]:
        """获取资源优化建议"""
        stats = self.get_system_stats()
        recommendations = []

        if stats.get("cpu", {}).get("percent", 0) > 80:
            recommendations.append("CPU使用率过高，考虑减少并发任务")
        if stats.get("memory", {}).get("percent", 0) > 85:
            recommendations.append("内存使用率过高，建议增加内存或优化使用")
        if stats.get("disk", {}).get("percent", 0) > 90:
            recommendations.append("磁盘空间不足，建议清理文件")
        for gpu in stats.get("gpu", []):
            if gpu.get("memory_percent", 0) > 85:
                recommendations.append(f"GPU {gpu['name']} 显存使用率过高")

        return recommendations or ["系统资源使用正常"]

        """设置配置值"""
        keys = key_path.split('.')
        config = self.config

                config[key] = {}

        config[keys[-1]] = value

        """深度合并字典"""
        result = base.copy()
        for key, value in override.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = self._deep_merge(result[key], value)
            else:
                result[key] = value
        return result
