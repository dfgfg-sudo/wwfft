# ===================== 1. 数据采集与解析 =====================
class DataCollector:
    """多格式数据采集器（支持ZIP递归）"""
    
    SUPPORTED_EXTS = {'.txt', '.csv', '.xlsx', '.json', '.docx', '.pdf', '.jpg', '.jpeg', '.png', '.zip'}
    
    def __init__(self, root_dir: str, max_files: int = 10000, max_workers: int = 4):
        self.root_dir = Path(root_dir)
        self.max_files = max_files
        self.max_workers = max_workers
        self.parsed_data = []
        self.counter = 0
        
    def collect(self) -> List[Dict]:
        """主入口：递归扫描并解析所有支持的文件"""
        if not self.root_dir.exists():
            raise FileNotFoundError(f"目录不存在: {self.root_dir}")
        
        logger.info(f"开始采集数据: {self.root_dir}")
        file_paths = self._scan_files()
        logger.info(f"发现 {len(file_paths)} 个文件，开始解析...")
        
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            for result in executor.map(self._parse_file, file_paths):
                if result is not None:
                    self.parsed_data.append(result)
                    self.counter += 1
                    if self.counter >= self.max_files:
                        break
        
        logger.info(f"成功解析 {len(self.parsed_data)} 个有效数据文件")
        return self.parsed_data
    
    def _scan_files(self) -> List[Path]:
        """递归扫描所有支持扩展名的文件"""
        files = []
        for ext in self.SUPPORTED_EXTS:
            pattern = f"**/*{ext}"
            files.extend(self.root_dir.glob(pattern))
        # 去重并排序
        return sorted(set(files))
    
    def _parse_file(self, file_path: Path) -> Optional[Dict]:
        """根据扩展名调用相应的解析器"""
        ext = file_path.suffix.lower()
        try:
            if ext == '.zip':
                return self._parse_zip(file_path)
            elif ext in ('.txt', '.md'):
                return self._parse_text(file_path)
            elif ext == '.csv':
                return self._parse_csv(file_path)
            elif ext in ('.xlsx', '.xls'):
                return self._parse_excel(file_path)
            elif ext == '.json':
                return self._parse_json(file_path)
            elif ext == '.docx':
                return self._parse_docx(file_path)
            elif ext == '.pdf':
                return self._parse_pdf(file_path)
            elif ext in ('.jpg', '.jpeg', '.png'):
                return self._parse_image(file_path)
            else:
                return None
        except Exception as e:
            logger.warning(f"解析失败 {file_path}: {e}")
            return None
    
    def _parse_text(self, path: Path) -> Dict:
        with open(path, 'r', encoding='utf-8', errors='ignore') as f:
            content = f.read()
        return {'type': 'text', 'path': str(path), 'content': content}
    
    def _parse_csv(self, path: Path) -> Dict:
        df = pd.read_csv(path, encoding='utf-8', errors='ignore')
        return {'type': 'tabular', 'path': str(path), 'data': df.to_dict(orient='records')}
    
    def _parse_excel(self, path: Path) -> Dict:
        df = pd.read_excel(path)
        return {'type': 'tabular', 'path': str(path), 'data': df.to_dict(orient='records')}
    
    def _parse_json(self, path: Path) -> Dict:
        with open(path, 'r', encoding='utf-8') as f:
            data = json.load(f)
        return {'type': 'json', 'path': str(path), 'data': data}
    
    def _parse_docx(self, path: Path) -> Dict:
        try:
            from docx import Document
            doc = Document(path)
            text = '\n'.join(p.text for p in doc.paragraphs if p.text.strip())
            return {'type': 'text', 'path': str(path), 'content': text}
        except ImportError:
            logger.warning("python-docx未安装，无法解析DOCX")
            return None
    
    def _parse_pdf(self, path: Path) -> Dict:
        try:
            import PyPDF2
            text = ""
            with open(path, 'rb') as f:
                reader = PyPDF2.PdfReader(f)
                for page in reader.pages:
                    text += page.extract_text() or ""
            return {'type': 'text', 'path': str(path), 'content': text}
        except ImportError:
            logger.warning("PyPDF2未安装，无法解析PDF")
            return None
    
    def _parse_image(self, path: Path) -> Dict:
        img = Image.open(path).convert('RGB')
        return {'type': 'image', 'path': str(path), 'image': img}
    
    def _parse_zip(self, path: Path) -> Optional[Dict]:
        """解压ZIP并递归解析内部文件（仅支持一级）"""
        extracted = []
        try:
            with zipfile.ZipFile(path, 'r') as zf:
                for info in zf.infolist():
                    if info.is_dir():
                        continue
                    ext = Path(info.filename).suffix.lower()
                    if ext not in self.SUPPORTED_EXTS:
                        continue
                    data = zf.read(info.filename)
                    # 根据扩展名处理内存中的数据
                    if ext in ('.txt', '.json'):
                        content = data.decode('utf-8', errors='ignore')
                        extracted.append({'type': 'text', 'path': f"{path}/{info.filename}", 'content': content})
                    elif ext == '.csv':
                        df = pd.read_csv(pd.io.common.BytesIO(data))
                        extracted.append({'type': 'tabular', 'path': f"{path}/{info.filename}", 'data': df.to_dict(orient='records')})
                    elif ext in ('.jpg', '.jpeg', '.png'):
                        img = Image.open(pd.io.common.BytesIO(data)).convert('RGB')
                        extracted.append({'type': 'image', 'path': f"{path}/{info.filename}", 'image': img})
            if extracted:
                return {'type': 'zip', 'path': str(path), 'contents': extracted}
        except Exception as e:
            logger.warning(f"ZIP解析失败 {path}: {e}")
        return None


# ===================== 2. 数据清洗与标准化 =====================
class DataCleaner:
    """智能数据清洗器"""
    
    def __init__(self, min_text_len: int = 10, max_text_len: int = 10000):
        self.min_len = min_text_len
        self.max_len = max_text_len
    
    def clean(self, raw_data: List[Dict]) -> List[Dict]:
        """统一清洗入口"""
        cleaned = []
        for item in raw_data:
            item_type = item.get('type')
            if item_type == 'text':
                cleaned_item = self._clean_text(item)
            elif item_type == 'tabular':
                cleaned_item = self._clean_tabular(item)
            elif item_type == 'image':
                cleaned_item = self._clean_image(item)
            elif item_type == 'zip':
                # 递归清洗ZIP内容
                sub_cleaned = self.clean(item.get('contents', []))
                cleaned.extend(sub_cleaned)
                continue
            else:
                continue
            if cleaned_item:
                cleaned.append(cleaned_item)
        return cleaned
    
    def _clean_text(self, item: Dict) -> Optional[Dict]:
        text = item.get('content', '')
        if not isinstance(text, str):
            return None
        # 基本清洗
        text = text.lower()
        text = ' '.join(text.split())
        # 保留中英文、数字、基本标点
        import re
        text = re.sub(r'[^\u4e00-\u9fa5a-zA-Z0-9\.,!?;:()\-\'"\s]', '', text)
        if self.min_len <= len(text) <= self.max_len:
            return {'type': 'text', 'path': item.get('path', ''), 'content': text}
        return None
    
    def _clean_tabular(self, item: Dict) -> Optional[Dict]:
        data = item.get('data')
        if not data:
            return None
        # 简单处理：转为DataFrame并清洗
        df = pd.DataFrame(data)
        df = df.drop_duplicates()
        # 删除全空列
        df = df.dropna(axis=1, how='all')
        # 数值列填充中位数，类别列填充众数
        for col in df.columns:
            if pd.api.types.is_numeric_dtype(df[col]):
                df[col] = df[col].fillna(df[col].median())
            else:
                df[col] = df[col].fillna(df[col].mode()[0] if not df[col].mode().empty else '')
        # 转换为文本特征（简单合并）
        text_rep = df.astype(str).apply(lambda row: ' '.join(row), axis=1).tolist()
        # 将每条记录作为一个文本样本
        return {'type': 'text', 'path': item.get('path', ''), 'content': '\n'.join(text_rep)}
    
    def _clean_image(self, item: Dict) -> Optional[Dict]:
        img = item.get('image')
        if img is None:
            return None
        # 标准化尺寸和格式
        img = img.resize((224, 224))
        # 转换为numpy数组并归一化
        arr = np.array(img).astype(np.float32) / 255.0
        return {'type': 'image', 'path': item.get('path', ''), 'array': arr}


# ===================== 3. 训练数据集封装 =====================
class TextDataset(Dataset):
    """文本数据集（用于语言模型微调）"""
    def __init__(self, texts, tokenizer, max_length=512):
        self.texts = texts
        self.tokenizer = tokenizer
        self.max_length = max_length
    
    def __len__(self):
        return len(self.texts)
    
    def __getitem__(self, idx):
        text = self.texts[idx]
        enc = self.tokenizer(
            text,
            truncation=True,
            padding='max_length',
            max_length=self.max_length,
            return_tensors='pt'
        )
        input_ids = enc['input_ids'].squeeze(0)
        attention_mask = enc['attention_mask'].squeeze(0)
        labels = input_ids.clone()
        return {'input_ids': input_ids, 'attention_mask': attention_mask, 'labels': labels}


class ImageDataset(Dataset):
    """图像数据集（分类/回归）"""
    def __init__(self, arrays):
        self.arrays = arrays
    
    def __len__(self):
        return len(self.arrays)
    
    def __getitem__(self, idx):
        arr = self.arrays[idx]
        # 转换为tensor，并添加通道维度
        tensor = torch.tensor(arr, dtype=torch.float32).permute(2, 0, 1)
        return tensor


# ===================== 4. 模型训练引擎 =====================
class ModelTrainer:
    """支持LoRA微调的轻量训练引擎"""
    
    def __init__(self, model_path: str, device: str = "auto"):
        self.model_path = Path(model_path)
        self.device = self._get_device(device)
        self.tokenizer = None
        self.model = None
        self.lora_config = None
        self._load_model()
    
    def _get_device(self, device):
        if device == "auto":
            return torch.device("cuda" if torch.cuda.is_available() else "cpu")
        return torch.device(device)
    
    def _load_model(self):
        """加载预训练模型和分词器（兼容Bunny-v1_0-3B）"""
        if not self.model_path.exists():
            raise FileNotFoundError(f"模型目录不存在: {self.model_path}")
        logger.info(f"加载模型: {self.model_path}")
        self.tokenizer = AutoTokenizer.from_pretrained(str(self.model_path), trust_remote_code=True)
        if self.tokenizer.pad_token is None:
            self.tokenizer.pad_token = self.tokenizer.eos_token
        
        config = AutoConfig.from_pretrained(str(self.model_path), trust_remote_code=True)
        self.model = AutoModelForCausalLM.from_pretrained(
            str(self.model_path),
            config=config,
            torch_dtype=torch.float32,
            trust_remote_code=True,
            low_cpu_mem_usage=True
        ).to(self.device)
        
        # 配置LoRA
        self.lora_config = LoraConfig(
            r=8,
            lora_alpha=32,
            target_modules=["q_proj", "v_proj"],
            lora_dropout=0.05,
            bias="none",
            task_type="CAUSAL_LM"
        )
        self.model = get_peft_model(self.model, self.lora_config)
        self.model.print_trainable_parameters()
        logger.info("模型加载完成")
    
    def train(self, texts: List[str], epochs: int = 3, batch_size: int = 2, lr: float = 5e-5):
        """执行微调训练"""
        if not texts:
            raise ValueError("训练文本列表为空")
        
        logger.info(f"开始训练，样本数: {len(texts)}，轮次: {epochs}")
        dataset = TextDataset(texts, self.tokenizer, max_length=512)
        dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True)
        
        optimizer = torch.optim.AdamW(self.model.parameters(), lr=lr)
        self.model.train()
        
        for epoch in range(epochs):
            total_loss = 0
            for batch in dataloader:
                batch = {k: v.to(self.device) for k, v in batch.items()}
                outputs = self.model(**batch)
                loss = outputs.loss
                loss.backward()
                optimizer.step()
                optimizer.zero_grad()
                total_loss += loss.item()
            avg_loss = total_loss / len(dataloader)
            logger.info(f"Epoch {epoch+1}/{epochs} - Loss: {avg_loss:.4f}")
        
        logger.info("训练完成")
        return self.model
    
    def save(self, output_dir: str):
        """保存微调后的模型"""
        path = Path(output_dir)
        path.mkdir(parents=True, exist_ok=True)
        self.model.save_pretrained(path)
        self.tokenizer.save_pretrained(path)
        logger.info(f"模型已保存至: {path}")

class AutoMLFactory:
    """全自动管道控制器"""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.data_dir = Path(config.get('data_dir', './data'))
        self.model_path = config.get('model_path', '')
        self.output_dir = Path(config.get('output_dir', './output'))
        self.max_files = config.get('max_files', 10000)
        self.epochs = config.get('epochs', 3)
        self.batch_size = config.get('batch_size', 2)
        self.lr = config.get('learning_rate', 5e-5)
        
        # 创建输出目录
        self.output_dir.mkdir(parents=True, exist_ok=True)
        self.data_dir.mkdir(parents=True, exist_ok=True)
    
    def run(self):
        """执行完整流水线"""
        logger.info("=" * 50)
        logger.info("AutoML Factory 启动")
        logger.info(f"数据目录: {self.data_dir}")
        logger.info(f"模型路径: {self.model_path}")
        logger.info(f"输出目录: {self.output_dir}")
        logger.info("=" * 50)
        
        # 1. 采集
        collector = DataCollector(str(self.data_dir), max_files=self.max_files)
        raw = collector.collect()
        if not raw:
            logger.error("未采集到任何数据，退出")
            return
        
        # 2. 清洗
        cleaner = DataCleaner()
        cleaned = cleaner.clean(raw)
        if not cleaned:
            logger.error("清洗后无有效数据，退出")
            return
        
        # 3. 提取文本（将所有清洗后的文本合并）
        texts = []
        for item in cleaned:
            if item.get('type') == 'text':
                texts.append(item.get('content', ''))
            elif item.get('type') == 'image':
                # 图像数据可另行处理，此处忽略
                pass
        
        if len(texts) < 10:
            logger.warning("有效文本样本太少，可能影响训练效果")
        
        # 4. 训练
        if self.model_path:
            trainer = ModelTrainer(self.model_path)
            trainer.train(texts, epochs=self.epochs, batch_size=self.batch_size, lr=self.lr)
            trainer.save(str(self.output_dir / 'fine_tuned'))
        else:
            logger.warning("未指定模型路径，跳过训练")
        
        # 5. 保存清洗后的数据（可选）
        cleaned_path = self.output_dir / 'cleaned_data.json'
        with open(cleaned_path, 'w', encoding='utf-8') as f:
            json.dump(cleaned, f, indent=2, ensure_ascii=False)
        logger.info(f"清洗后数据已保存至: {cleaned_path}")
        
        logger.info("=" * 50)
        logger.info("AutoML Factory 执行完毕")
        logger.info("=" * 50)
