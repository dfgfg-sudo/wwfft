# ====================== 安全参数提取 ======================
def extract_params(args):
    """
    从 Coze 传入的 args 对象（可能是 Args、CustomNamespace、dict 等）中安全提取参数字典
    """
    params = {}
    try:
        # 1. 如果是字典
        if isinstance(args, dict):
            if "input" in args and isinstance(args["input"], dict):
                params = args["input"]
            else:
                params = args
        # 2. 如果有 input 属性
        elif hasattr(args, "input") and args.input is not None:
            if isinstance(args.input, dict):
                params = args.input
            elif isinstance(args.input, str):
                try:
                    params = json.loads(args.input)
                except:
                    pass
        # 3. 如果有 to_dict 方法
        elif hasattr(args, "to_dict"):
            try:
                params = args.to_dict()
            except:
                pass
        # 4. 如果有 __dict__ 属性
        elif hasattr(args, "__dict__"):
            # 只取基础类型字段，避免序列化问题
            for k, v in args.__dict__.items():
                if not k.startswith("_") and isinstance(v, (str, int, float, bool, dict, list, type(None))):
                    params[k] = v
        # 5. 尝试直接获取常用属性
        for attr in ["space_id", "spaceId", "auto_fix", "dry_run", "max_workflows", "coze_api_token"]:
            if hasattr(args, attr):
                val = getattr(args, attr)
                if val is not None and isinstance(val, (str, int, float, bool)):
                    params[attr] = val
    except Exception:
        pass
    return params

# ====================== 工具函数 ======================
def get_token(params: Dict) -> str:
    token = params.get("coze_api_token") or os.environ.get("COZE_API_TOKEN")
    if not token:
        raise ValueError("未找到 COZE_API_TOKEN，请设置环境变量或在输入参数中提供")
    return token

def extract_error_code(error_msg: str) -> str:
    match = re.search(r'\b(\d{9})\b', error_msg)
    if match:
        return match.group(1)
    match2 = re.search(r'code[:\s]*(\d+)', error_msg, re.IGNORECASE)
    if match2:
        return match2.group(1)
    return "UNKNOWN"

def api_request(method: str, url: str, token: str, json_data: Dict = None, params: Dict = None) -> Dict:
    headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}
    resp = requests.request(method, url, headers=headers, json=json_data, params=params, timeout=30)
    resp.raise_for_status()
    result = resp.json()
    if result.get("code") != 0:
        raise Exception(f"API错误: code={result.get('code')}, msg={result.get('msg')}")
    return result.get("data", {})

def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    workflows = []
    page = 1
    page_size = 50
    while len(workflows) < limit:
        params = {"space_id": space_id, "page_num": page, "page_size": page_size}
        data = api_request("GET", WORKFLOW_LIST_API, token, params=params)
        items = data.get("items", [])
        if not items:
            break
        workflows.extend(items)
        if len(workflows) >= limit or len(items) < page_size:
            break
        page += 1
    return workflows[:limit]

def diagnose_workflow(workflow_id: str, token: str) -> Dict:
    try:
        resp = requests.post(
            WORKFLOW_RUN_API,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            json={"workflow_id": workflow_id, "parameters": {"__diagnostic__": True}},
            timeout=30
        )
        if resp.status_code == 200:
            result = resp.json()
            if result.get("code") == 0:
                return {"status": "healthy", "error_code": None, "error_msg": None}
            else:
                error_msg = result.get("msg", "")
                return {"status": "error", "error_code": extract_error_code(error_msg), "error_msg": error_msg}
        else:
            return {"status": "error", "error_code": f"HTTP_{resp.status_code}", "error_msg": resp.text}
    except Exception as e:
        return {"status": "error", "error_code": "EXCEPTION", "error_msg": str(e)}

def publish_workflow(workflow_id: str, token: str) -> bool:
    url = WORKFLOW_PUBLISH_API.format(workflow_id=workflow_id)
    try:
        api_request("POST", url, token, json_data={})
        return True
    except Exception as e:
        print(f"[ERROR] 发布失败: {e}")
        return False

def adjust_timeout(workflow_id: str, token: str) -> bool:
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            node_type = node.get("type", "")
            if node_type in ("http", "code", "plugin"):
                if "timeout" in node:
                    old = node.get("timeout", 30)
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["timeout"] = new
                        modified = True
                elif "config" in node and "timeout" in node["config"]:
                    old = node["config"].get("timeout", 30)
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["config"]["timeout"] = new
                        modified = True
        if not modified:
            return True
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] 调整超时失败: {e}")
        return False

def fix_json_schema(workflow_id: str, token: str) -> bool:
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        if "nodes" in config:
            for node in config["nodes"]:
                if "output_schema" in node and isinstance(node["output_schema"], dict):
                    node["output_schema"] = json.loads(json.dumps(node["output_schema"]))
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] JSON修复失败: {e}")
        return False

def execute_fix(workflow_id: str, workflow_name: str, error_code: str, token: str) -> Dict:
    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy["action"]
    max_retries = strategy["retry"]
    if action in ("skip", "skip_manual"):
        return {"status": "skipped", "action": "none", "message": strategy["description"]}
    fix_map = {
        "publish": publish_workflow,
        "adjust_timeout": adjust_timeout,
        "fix_json_schema": fix_json_schema,
    }
    fix_func = fix_map.get(action)
    if fix_func is None:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}
    for attempt in range(max_retries + 1):
        try:
            success = fix_func(workflow_id, token)
            if success:
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功"}
                else:
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试 {max_retries} 次后失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": f"异常: {str(e)}"}
    return {"status": "failed", "action": action, "message": "未知错误"}

def handler(args):
    try:
        # 安全提取参数字典
        params = extract_params(args)
        
        # 获取 space_id（支持多种写法）
        space_id = params.get("space_id") or params.get("spaceId")
        if not space_id:
            # 尝试从原始 args 中直接获取（兜底）
            if hasattr(args, "space_id"):
                space_id = getattr(args, "space_id")
            elif hasattr(args, "spaceId"):
                space_id = getattr(args, "spaceId")
        
        if not space_id:
            return {
                "total": 0,
                "healthy": 0,
                "fixed": 0,
                "failed": 0,
                "report": json.dumps({"error": "space_id 不能为空，请在输入参数中提供 space_id"}, ensure_ascii=False)
            }
        
        space_id = str(space_id).strip()
        auto_fix = params.get("auto_fix", True)
        dry_run = params.get("dry_run", False)
        max_workflows = int(params.get("max_workflows", 100))
        
        # 获取 Token
        token = get_token(params)
        
        # 扫描工作流
        print(f"扫描空间 {space_id} 的工作流...")
        workflows = get_workflow_list(space_id, token, max_workflows)
        total = len(workflows)
        if total == 0:
            return {
                "total": 0,
                "healthy": 0,
                "fixed": 0,
                "failed": 0,
                "report": json.dumps({"message": "未找到任何工作流", "space_id": space_id}, ensure_ascii=False)
            }
        
        healthy_count = fixed_count = failed_count = skipped_count = 0
        report_details = []
        
        for idx, wf in enumerate(workflows, 1):
            wf_id = wf.get("id")
            wf_name = wf.get("name", f"workflow_{wf_id}")
            print(f"[{idx}/{total}] 诊断: {wf_name}")
            
            diagnosis = diagnose_workflow(wf_id, token)
            record = {
                "workflow_id": wf_id,
                "workflow_name": wf_name,
                "diagnosis": diagnosis
            }
            
            if diagnosis["status"] == "healthy":
                healthy_count += 1
                record["fix_result"] = {"status": "healthy"}
                print(f"  ✓ 健康")
            else:
                error_code = diagnosis.get("error_code", "UNKNOWN")
                error_msg = diagnosis.get("error_msg", "")
                print(f"  ✗ 错误码: {error_code} - {error_msg[:100]}")
                
                if dry_run or not auto_fix:
                    record["fix_result"] = {"status": "dry_run", "error_code": error_code}
                    skipped_count += 1
                    print(f"  ⏭️ 跳过（仅诊断模式）")
                else:
                    fix_result = execute_fix(wf_id, wf_name, error_code, token)
                    record["fix_result"] = fix_result
                    if fix_result["status"] == "fixed":
                        fixed_count += 1
                        print(f"  ✅ 修复成功")
                    elif fix_result["status"] in ("failed", "partial"):
                        failed_count += 1
                        print(f"  ❌ 修复失败: {fix_result.get('message')}")
                    else:
                        skipped_count += 1
                        print(f"  ⏭️ 跳过: {fix_result.get('message')}")
            
            report_details.append(record)
        
        summary = {
            "total": total,
            "healthy": healthy_count,
            "fixed": fixed_count,
            "failed": failed_count,
            "skipped": skipped_count,
            "dry_run": dry_run,
            "auto_fix": auto_fix,
            "space_id": space_id,
            "scan_time": time.strftime("%Y-%m-%d %H:%M:%S")
        }
        full_report = {"summary": summary, "details": report_details}
        
        return {
            "total": total,
            "healthy": healthy_count,
            "fixed": fixed_count,
            "failed": failed_count,
            "report": json.dumps(full_report, ensure_ascii=False, indent=2)
        }
        
    except Exception as e:
        return {
            "total": 0,
            "healthy": 0,
            "fixed": 0,
            "failed": 0,
            "report": json.dumps({"error": str(e), "type": type(e).__name__}, ensure_ascii=False)
        }

def extract_params(args):
    """
    从 Coze 传入的任何类型对象中安全提取参数字典
    支持：dict, SimpleNamespace, Args, 任意对象, 以及嵌套 input 的情况
    """

    # 调试输出（发布时可删除）
    print(f"[DEBUG] args type: {type(args)}")
    if hasattr(args, "__dict__"):
        print(f"[DEBUG] args.__dict__: {args.__dict__}")
    else:
        print(f"[DEBUG] args (as dict): {args}")

    # 方法1：如果 args 本身是字典
    if isinstance(args, dict):
        params.update(args)
        # 如果字典中有 input 键且是字典，则优先使用 input 内容
        if "input" in params and isinstance(params["input"], dict):
            params.update(params["input"])
            params.pop("input", None)
    else:
        # 方法2：如果 args 是对象，获取所有非私有属性
        if hasattr(args, "__dict__"):
            for k, v in args.__dict__.items():
                if not k.startswith("_"):
                    params[k] = v
        # 方法3：尝试直接 getattr 可能暴露的公共属性
        for attr in ["space_id", "spaceId", "auto_fix", "dry_run", "max_workflows", "coze_api_token"]:
            if hasattr(args, attr):
                val = getattr(args, attr)
                if val is not None:
                    params[attr] = val

    # 方法4：如果仍然为空，尝试从 args.input 中读取（Coze 有时会包装）
    if not params:
        if hasattr(args, "input"):
            inp = args.input
            if isinstance(inp, dict):
                params.update(inp)
            elif isinstance(inp, str):
                try:
                    params.update(json.loads(inp))
                except:
                    pass

    # 标准化参数名：将 spaceId 统一为 space_id
    if "spaceId" in params and "space_id" not in params:
        params["space_id"] = params.pop("spaceId")

    # 类型转换
    if "auto_fix" in params:
        if isinstance(params["auto_fix"], str):
            params["auto_fix"] = params["auto_fix"].lower() in ("true", "1", "yes")
    if "dry_run" in params:

def extract_error_code(error_msg: str) -> str:
    match = re.search(r'\b(\d{9})\b', error_msg)
    if match:
        return match.group(1)
    match2 = re.search(r'code[:\s]*(\d+)', error_msg, re.IGNORECASE)
    if match2:
        return match2.group(1)
    return "UNKNOWN"

def api_request(method: str, url: str, token: str, json_data: Dict = None, params: Dict = None) -> Dict:
    headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}
    resp = requests.request(method, url, headers=headers, json=json_data, params=params, timeout=30)
    resp.raise_for_status()
    result = resp.json()
    if result.get("code") != 0:
        raise Exception(f"API错误: code={result.get('code')}, msg={result.get('msg')}")
    return result.get("data", {})

def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    workflows = []
    page = 1
    page_size = 50
    while len(workflows) < limit:
        params = {"space_id": space_id, "page_num": page, "page_size": page_size}
        data = api_request("GET", WORKFLOW_LIST_API, token, params=params)
        items = data.get("items", [])
        if not items:
            break
        workflows.extend(items)
        if len(workflows) >= limit or len(items) < page_size:
            break
        page += 1
    return workflows[:limit]

def diagnose_workflow(workflow_id: str, token: str) -> Dict:
    try:
        resp = requests.post(
            WORKFLOW_RUN_API,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            json={"workflow_id": workflow_id, "parameters": {"__diagnostic__": True}},
            timeout=30
        )
        if resp.status_code == 200:
            result = resp.json()
            if result.get("code") == 0:
                return {"status": "healthy", "error_code": None, "error_msg": None}
            else:
                error_msg = result.get("msg", "")
                return {"status": "error", "error_code": extract_error_code(error_msg), "error_msg": error_msg}
        else:
            return {"status": "error", "error_code": f"HTTP_{resp.status_code}", "error_msg": resp.text}
    except Exception as e:
        return {"status": "error", "error_code": "EXCEPTION", "error_msg": str(e)}

def publish_workflow(workflow_id: str, token: str) -> bool:
    url = WORKFLOW_PUBLISH_API.format(workflow_id=workflow_id)
    try:
        api_request("POST", url, token, json_data={})
        return True
    except Exception as e:
        print(f"[ERROR] 发布失败: {e}")
        return False

def adjust_timeout(workflow_id: str, token: str) -> bool:
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            node_type = node.get("type", "")
            if node_type in ("http", "code", "plugin"):
                if "timeout" in node:
                    old = node.get("timeout", 30)
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["timeout"] = new
                        modified = True
                elif "config" in node and "timeout" in node["config"]:
                    old = node["config"].get("timeout", 30)
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["config"]["timeout"] = new
                        modified = True
        if not modified:
            return True
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] 调整超时失败: {e}")
        return False

def fix_json_schema(workflow_id: str, token: str) -> bool:
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        if "nodes" in config:
            for node in config["nodes"]:
                if "output_schema" in node and isinstance(node["output_schema"], dict):
                    node["output_schema"] = json.loads(json.dumps(node["output_schema"]))
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] JSON修复失败: {e}")
        return False

def execute_fix(workflow_id: str, workflow_name: str, error_code: str, token: str) -> Dict:
    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy["action"]
    max_retries = strategy["retry"]
    if action in ("skip", "skip_manual"):
        return {"status": "skipped", "action": "none", "message": strategy["description"]}
    fix_map = {
        "publish": publish_workflow,
        "adjust_timeout": adjust_timeout,
        "fix_json_schema": fix_json_schema,
    }
    fix_func = fix_map.get(action)
    if fix_func is None:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}
    for attempt in range(max_retries + 1):
        try:
            success = fix_func(workflow_id, token)
            if success:
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功"}
                else:
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试 {max_retries} 次后失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": f"异常: {str(e)}"}
    return {"status": "failed", "action": action, "message": "未知错误"}

def get_param(args, name: str, default=None):
    """
    从任何类型的 args 中安全获取参数值
    支持：对象属性、字典键、大小写变体
    """
    # 1. 如果 args 是字典
    if isinstance(args, dict):
        if name in args:
            return args[name]
        # 尝试大小写变体
        alt_name = "spaceId" if name == "space_id" else name.replace("_", "").lower()
        for k, v in args.items():
            if k.lower() == alt_name or k == name.replace("_", ""):
                return v
        return default
    
    # 2. 如果 args 是对象
    if hasattr(args, name):
        return getattr(args, name)
    # 尝试备用名称
    if name == "space_id" and hasattr(args, "spaceId"):
        return getattr(args, "spaceId")
    if name == "auto_fix" and hasattr(args, "autoFix"):
        return getattr(args, "autoFix")
    
    return default

def extract_params(args):
    """提取所需参数的字典"""
    params = {}
    
    # 定义需要提取的参数及其默认值
    param_names = ["space_id", "auto_fix", "dry_run", "max_workflows", "coze_api_token"]
    
    for name in param_names:
        value = get_param(args, name)
        if value is not None:
            params[name] = value
    
    # 类型转换
    if "auto_fix" in params:
        if isinstance(params["auto_fix"], str):
            params["auto_fix"] = params["auto_fix"].lower() in ("true", "1", "yes")
    if "dry_run" in params:
        if isinstance(params["dry_run"], str):
            params["dry_run"] = params["dry_run"].lower() in ("true", "1", "yes")
    if "max_workflows" in params:
        if isinstance(params["max_workflows"], str):
            params["max_workflows"] = int(params["max_workflows"])
        elif isinstance(params["max_workflows"], float):
            params["max_workflows"] = int(params["max_workflows"])
    
    # 设置默认值
    params.setdefault("auto_fix", True)
    params.setdefault("dry_run", False)
    params.setdefault("max_workflows", 100)
    
    # 调试打印（正式使用时可以删除或注释掉）
    print(f"[DEBUG] Extracted params: {params}")
    
    return params

def extract_error_code(error_msg: str) -> str:
    match = re.search(r'\b(\d{9})\b', error_msg)
    if match:
        return match.group(1)
    match2 = re.search(r'code[:\s]*(\d+)', error_msg, re.IGNORECASE)
    if match2:
        return match2.group(1)
    return "UNKNOWN"

def api_request(method: str, url: str, token: str, json_data: Dict = None, params: Dict = None) -> Dict:
    headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}
    resp = requests.request(method, url, headers=headers, json=json_data, params=params, timeout=30)
    resp.raise_for_status()
    result = resp.json()
    if result.get("code") != 0:
        raise Exception(f"API错误: code={result.get('code')}, msg={result.get('msg')}")
    return result.get("data", {})

def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    workflows = []
    page = 1
    page_size = 50
    while len(workflows) < limit:
        params = {"space_id": space_id, "page_num": page, "page_size": page_size}
        data = api_request("GET", WORKFLOW_LIST_API, token, params=params)
        items = data.get("items", [])
        if not items:
            break
        workflows.extend(items)
        if len(workflows) >= limit or len(items) < page_size:
            break
        page += 1
    return workflows[:limit]

def diagnose_workflow(workflow_id: str, token: str) -> Dict:
    try:
        resp = requests.post(
            WORKFLOW_RUN_API,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            json={"workflow_id": workflow_id, "parameters": {"__diagnostic__": True}},
            timeout=30
        )
        if resp.status_code == 200:
            result = resp.json()
            if result.get("code") == 0:
                return {"status": "healthy", "error_code": None, "error_msg": None}
            else:
                error_msg = result.get("msg", "")
                return {"status": "error", "error_code": extract_error_code(error_msg), "error_msg": error_msg}
        else:
            return {"status": "error", "error_code": f"HTTP_{resp.status_code}", "error_msg": resp.text}
    except Exception as e:
        return {"status": "error", "error_code": "EXCEPTION", "error_msg": str(e)}

def publish_workflow(workflow_id: str, token: str) -> bool:

    try:
        api_request("POST", url, token, json_data={})
        return True
    except Exception as e:

        return False

def adjust_timeout(workflow_id: str, token: str) -> bool:
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            node_type = node.get("type", "")
            if node_type in ("http", "code", "plugin"):
                if "timeout" in node:
                    old = node.get("timeout", 30)
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["timeout"] = new
                        modified = True
                elif "config" in node and "timeout" in node["config"]:
                    old = node["config"].get("timeout", 30)
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["config"]["timeout"] = new
                        modified = True
        if not modified:
            return True
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] 调整超时失败: {e}")
        return False

def fix_json_schema(workflow_id: str, token: str) -> bool:
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        if "nodes" in config:
            for node in config["nodes"]:
                if "output_schema" in node and isinstance(node["output_schema"], dict):
                    node["output_schema"] = json.loads(json.dumps(node["output_schema"]))
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] JSON修复失败: {e}")
        return False

def execute_fix(workflow_id: str, workflow_name: str, error_code: str, token: str) -> Dict:
    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy["action"]
    max_retries = strategy["retry"]
    if action in ("skip", "skip_manual"):
        return {"status": "skipped", "action": "none", "message": strategy["description"]}
    fix_map = {
        "publish": publish_workflow,
        "adjust_timeout": adjust_timeout,
        "fix_json_schema": fix_json_schema,
    }
    fix_func = fix_map.get(action)
    if fix_func is None:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}
    for attempt in range(max_retries + 1):
        try:
            success = fix_func(workflow_id, token)
            if success:
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功"}
                else:
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试 {max_retries} 次后失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": f"异常: {str(e)}"}
    return {"status": "failed", "action": action, "message": "未知错误"}

// 1. 安全清洗函数：对所有外部输入进行无害化处理
function sanitizeInput(input) {
    if (!input) return "";
    // 移除潜在的危险HTML标签和脚本，防止XSS
    input = input.replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '');
    input = input.replace(/<[^>]*>/g, '');
    // 转义特殊字符，防止注入
    input = input.replace(/&/g, '&amp;')
                 .replace(/</g, '&lt;')
                 .replace(/>/g, '&gt;')
                 .replace(/"/g, '&quot;')
                 .replace(/'/g, '&#039;');
    return input.trim();
}

def sanitize(text):
    """输入安全清洗：移除 HTML/脚本，转义部分特殊字符"""

    # 移除 <script> 标签及其内容

    text = re.sub(r'<[^>]*>', '', text)
    # 转义可能用于注入的特殊字符
    text = text.replace('&', '&amp;').replace('<', '&lt;').replace('>', '&gt;')
    text = text.replace('"', '&quot;').replace("'", '&#039;')
    return text.strip()

"""
Coze 工作流智能修复插件
功能：自动扫描空间内所有工作流 → 诊断运行错误 → 安全修复 → 输出审计报告
版本：1.0.0
兼容性：Coze IDE Python 3.8+
"""

import json
import time
import logging
from datetime import datetime
from typing import Dict, List, Optional, Any, Tuple

import requests

# ---------------------------- 日志配置 ----------------------------
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
)
logger = logging.getLogger("WorkflowAutoFix")

# ---------------------------- API 基础配置 ----------------------------
COZE_API_BASE = "https://api.coze.cn"  # 国内站使用 https://api.coze.cn；国际站使用 https://api.coze.com
DEFAULT_TIMEOUT = 30
REQUEST_RETRY = 3
RETRY_DELAY = 1


class CozeAPIClient:
    """Coze API 封装（带重试与错误处理）"""

    def __init__(self, api_token: str, base_url: str = COZE_API_BASE):
        self.api_token = api_token
        self.base_url = base_url.rstrip('/')
        self.headers = {
            "Authorization": f"Bearer {api_token}",
            "Content-Type": "application/json"
        }

    def _request(self, method: str, endpoint: str, **kwargs) -> Dict[str, Any]:
        url = f"{self.base_url}{endpoint}"
        retries = kwargs.pop('retries', REQUEST_RETRY)
        for attempt in range(1, retries + 1):
            try:
                resp = requests.request(
                    method, url,
                    headers=self.headers,
                    timeout=DEFAULT_TIMEOUT,
                    **kwargs
                )
                resp.raise_for_status()
                return resp.json()
            except requests.exceptions.RequestException as e:
                logger.warning(f"API请求失败 (尝试 {attempt}/{retries}): {e}")
                if attempt == retries:
                    raise RuntimeError(f"API请求最终失败: {e}")
                time.sleep(RETRY_DELAY)
        return {}

    def get(self, endpoint: str, params: Dict = None) -> Dict:
        return self._request("GET", endpoint, params=params)

    def patch(self, endpoint: str, data: Dict) -> Dict:
        return self._request("PATCH", endpoint, json=data)

    def get_workflows(self, space_id: str, page_size: int = 50) -> List[Dict]:
        """获取空间内所有工作流（自动分页）"""
        workflows = []
        page_token = None
        while True:
            params = {
                "space_id": space_id,
                "page_size": page_size
            }
            if page_token:
                params["page_token"] = page_token
            resp = self.get("/v1/workflows", params=params)
            items = resp.get("data", [])
            workflows.extend(items)
            page_token = resp.get("next_page_token")
            if not page_token:
                break
        return workflows

    def get_workflow_definition(self, workflow_id: str) -> Dict:
        """获取工作流完整定义"""
        return self.get(f"/v1/workflows/{workflow_id}")

    def get_workflow_runs(self, workflow_id: str, limit: int = 20) -> List[Dict]:
        """获取工作流最近运行记录"""
        resp = self.get(
            f"/v1/workflows/{workflow_id}/runs",
            params={"limit": limit, "order": "desc"}
        )
        return resp.get("data", [])

    def update_workflow(self, workflow_id: str, definition: Dict) -> Dict:
        """更新工作流定义"""
        payload = {"workflow_definition": definition}
        return self.patch(f"/v1/workflows/{workflow_id}", payload)


# ---------------------------- 错误诊断与修复策略 ----------------------------
class WorkflowDoctor:
    """工作流诊断器：分析运行错误，定位问题节点"""

    @staticmethod
    def diagnose(workflow_def: Dict, recent_runs: List[Dict]) -> List[Dict]:
        """
        诊断工作流，返回错误列表
        每项包含: {'type': str, 'node_id': str, 'details': str, 'fix_strategy': str}
        """
        errors = []
        nodes = workflow_def.get("nodes", [])
        edges = workflow_def.get("edges", [])

        # 1. 检查超时配置缺失
        for node in nodes:
            if node.get("type") in ("http", "llm", "code") and not node.get("timeout"):
                errors.append({
                    "type": "missing_timeout",
                    "node_id": node.get("id"),
                    "details": f"节点 [{node.get('name')}] 未设置超时时间",
                    "fix_strategy": "add_timeout",
                    "suggested_value": 30
                })

        # 2. 检查重试策略缺失 (仅对 HTTP/API 节点)
        for node in nodes:
            if node.get("type") == "http" and not node.get("retry"):
                errors.append({
                    "type": "missing_retry",
                    "node_id": node.get("id"),
                    "details": f"HTTP节点 [{node.get('name')}] 未配置重试策略",
                    "fix_strategy": "add_retry",
                    "suggested_value": {"max_attempts": 2, "delay": 1}
                })

        # 3. 检查条件分支是否缺少 else 分支
        for node in nodes:
            if node.get("type") == "condition":
                branches = node.get("branches", [])
                has_else = any(b.get("is_else", False) for b in branches)
                if not has_else:
                    errors.append({
                        "type": "missing_else_branch",
                        "node_id": node.get("id"),
                        "details": f"条件节点 [{node.get('name')}] 缺少 else 默认分支",
                        "fix_strategy": "add_else_branch",
                        "suggested_value": {"target": "default_next_node"}
                    })

        # 4. 检查变量引用是否存在未定义的变量 (通过运行日志具体错误)
        for run in recent_runs[:5]:
            if run.get("status") == "failed":
                err_msg = run.get("error_message", "").lower()
                if "variable" in err_msg and "not defined" in err_msg:
                    errors.append({
                        "type": "undefined_variable",
                        "node_id": run.get("failed_node_id", "unknown"),
                        "details": f"运行失败，变量未定义: {run.get('error_message')}",
                        "fix_strategy": "auto_fix_variable_reference"
                    })

        # 5. 检查节点配置缺失（如 HTTP 节点缺少 URL）
        for node in nodes:
            if node.get("type") == "http":
                if not node.get("url"):
                    errors.append({
                        "type": "missing_http_url",
                        "node_id": node.get("id"),
                        "details": f"HTTP节点 [{node.get('name')}] 缺失 url 配置",
                        "fix_strategy": "skip_or_prompt"
                    })

        # 6. 简单循环依赖检测 (直接检测 A->B 且 B->A)
        edge_map = {}
        for edge in edges:
            src = edge.get("source")
            tgt = edge.get("target")
            edge_map.setdefault(src, []).append(tgt)
        for edge in edges:
            src = edge.get("source")
            tgt = edge.get("target")
            if tgt in edge_map and src in edge_map.get(tgt, []):
                errors.append({
                    "type": "circular_dependency",
                    "node_id": f"{src}↔{tgt}",
                    "details": f"检测到循环依赖: {src} ↔ {tgt}",
                    "fix_strategy": "manual_review"
                })
        return errors

    @staticmethod
    def apply_fix(workflow_def: Dict, error: Dict) -> Tuple[Dict, str]:
        """根据错误类型修改工作流定义，返回(新定义, 修复说明)"""
        new_def = json.loads(json.dumps(workflow_def))  # 深拷贝
        fix_strategy = error.get("fix_strategy")
        node_id = error.get("node_id")
        if not node_id:
            return new_def, f"无法修复：缺少节点ID ({error})"

        # 定位节点
        node = None
        for n in new_def.get("nodes", []):
            if n.get("id") == node_id:
                node = n
                break

        if not node and fix_strategy not in ("circular_dependency",):
            return new_def, f"未找到节点 {node_id}，修复跳过"

        fix_msg = ""
        if fix_strategy == "add_timeout":
            node["timeout"] = error.get("suggested_value", 30)
            fix_msg = f"为节点 {node_id} 添加超时配置 {node['timeout']}s"
        elif fix_strategy == "add_retry":
            node["retry"] = error.get("suggested_value", {"max_attempts": 2, "delay": 1})
            fix_msg = f"为节点 {node_id} 添加重试策略: {node['retry']}"
        elif fix_strategy == "add_else_branch":
            if "branches" not in node:
                node["branches"] = []
            node["branches"].append({
                "is_else": True,
                "target": error.get("suggested_value", {}).get("target", "default_next")
            })
            fix_msg = f"为条件节点 {node_id} 添加 else 默认分支"
        elif fix_strategy == "auto_fix_variable_reference":
            # 简化处理：尝试将未定义变量替换为默认空字符串
            fix_msg = f"节点 {node_id} 存在未定义变量，建议人工审查并配置默认值。自动修复未实施"
        else:
            fix_msg = f"错误类型 '{error['type']}' 无自动修复策略，需人工介入"

        return new_def, fix_msg

class WorkflowAutoFixEngine:
    def __init__(self, api_client: CozeAPIClient, space_id: str, dry_run: bool = False):
        self.api = api_client
        self.space_id = space_id
        self.dry_run = dry_run
        self.report = {
            "start_time": datetime.utcnow().isoformat(),
            "space_id": space_id,
            "dry_run": dry_run,
            "workflows": [],
            "summary": {}
        }

    def run(self) -> str:
        """执行完整扫描修复流程，返回审计报告文本"""
        logger.info(f"开始扫描空间 {self.space_id}，Dry-run模式: {self.dry_run}")
        workflows = self.api.get_workflows(self.space_id)
        logger.info(f"找到 {len(workflows)} 个工作流")

        total_fixed = 0
        total_errors = 0

        for wf in workflows:
            wf_id = wf.get("workflow_id")
            wf_name = wf.get("name", wf_id)
            logger.info(f"处理工作流: {wf_name} ({wf_id})")

            # 获取完整定义
            try:
                full_def = self.api.get_workflow_definition(wf_id)
            except Exception as e:
                logger.error(f"获取工作流 {wf_id} 定义失败: {e}")
                self.report["workflows"].append({
                    "id": wf_id,
                    "name": wf_name,
                    "status": "failed",
                    "error": str(e)
                })
                continue

            # 获取最近运行记录
            runs = self.api.get_workflow_runs(wf_id, limit=10)

            # 诊断
            doctor = WorkflowDoctor()
            errors = doctor.diagnose(full_def, runs)
            if not errors:
                self.report["workflows"].append({
                    "id": wf_id,
                    "name": wf_name,
                    "status": "healthy",
                    "errors_count": 0
                })
                continue

            total_errors += len(errors)
            wf_report_item = {
                "id": wf_id,
                "name": wf_name,
                "errors": [],
                "fixes_applied": [],
                "status": "pending"
            }

            # 备份原始定义
            backup_key = f"{wf_id}_backup_{int(time.time())}"
            wf_report_item["backup_key"] = backup_key

            current_def = full_def
            fixes = []

            # 逐一应用修复
            for err in errors:
                new_def, fix_msg = doctor.apply_fix(current_def, err)
                if new_def != current_def:
                    fixes.append(fix_msg)
                    current_def = new_def

            if fixes and not self.dry_run:
                try:
                    self.api.update_workflow(wf_id, current_def)
                    wf_report_item["status"] = "fixed"
                    wf_report_item["fixes_applied"] = fixes
                    total_fixed += 1
                    logger.info(f"工作流 {wf_name} 修复成功，应用 {len(fixes)} 项修改")
                except Exception as e:
                    wf_report_item["status"] = "fix_failed"
                    wf_report_item["fix_error"] = str(e)
                    logger.error(f"工作流 {wf_name} 修复失败: {e}")
            elif fixes and self.dry_run:
                wf_report_item["status"] = "dry_run_fix_available"
                wf_report_item["fixes_applied"] = fixes
                logger.info(f"[Dry-run] 工作流 {wf_name} 需要修复: {fixes}")
            else:
                wf_report_item["status"] = "no_fix_possible"
                wf_report_item["unresolved_errors"] = [e["type"] for e in errors]

            wf_report_item["errors_count"] = len(errors)
            wf_report_item["errors_detected"] = errors
            self.report["workflows"].append(wf_report_item)

        # 汇总统计
        self.report["summary"] = {
            "total_workflows": len(workflows),
            "total_errors_detected": total_errors,
            "total_fixed": total_fixed if not self.dry_run else 0,
            "dry_run_suggested_fixes": sum(1 for w in self.report["workflows"] if w.get("status") == "dry_run_fix_available"),
            "failed_fixes": sum(1 for w in self.report["workflows"] if w.get("status") == "fix_failed"),
            "healthy_workflows": sum(1 for w in self.report["workflows"] if w.get("status") == "healthy")
        }
        self.report["end_time"] = datetime.utcnow().isoformat()
        return self._format_report()

    def _format_report(self) -> str:
        """生成可读的审计报告"""
        report_lines = []
        report_lines.append("=" * 80)
        report_lines.append("Coze 工作流智能修复审计报告")
        report_lines.append(f"空间ID: {self.report['space_id']}")
        report_lines.append(f"执行模式: {'Dry-Run (只读)' if self.report['dry_run'] else '实际修复'}")
        report_lines.append(f"开始时间: {self.report['start_time']}")
        report_lines.append(f"结束时间: {self.report['end_time']}")
        report_lines.append("-" * 80)

        summary = self.report["summary"]
        report_lines.append(f"📊 总计: {summary['total_workflows']} 个工作流")
        report_lines.append(f"🔍 检测到错误总数: {summary['total_errors_detected']}")
        if self.report['dry_run']:
            report_lines.append(f"💡 建议修复的工作流数: {summary['dry_run_suggested_fixes']}")
        else:
            report_lines.append(f"✅ 成功修复的工作流数: {summary['total_fixed']}")
            report_lines.append(f"⚠️ 修复失败的工作流数: {summary['failed_fixes']}")
        report_lines.append(f"❤️ 健康工作流数: {summary['healthy_workflows']}")
        report_lines.append("-" * 80)

        for wf in self.report["workflows"]:
            report_lines.append(f"\n📁 工作流: {wf['name']} ({wf['id']})")
            report_lines.append(f"状态: {wf['status']}")
            if wf.get("errors_count", 0) > 0:
                report_lines.append(f"错误数: {wf['errors_count']}")
                for err in wf.get("errors_detected", []):
                    report_lines.append(f"  - [{err['type']}] {err['details']}")
            if wf.get("fixes_applied"):
                report_lines.append("修复动作:")
                for fix in wf["fixes_applied"]:
                    report_lines.append(f"  ✓ {fix}")
            if wf.get("unresolved_errors"):
                report_lines.append(f"未解决错误类型: {', '.join(wf['unresolved_errors'])}")
        report_lines.append("\n" + "=" * 80)
        report_lines.append("报告生成完毕 | 安全自动化修复插件 v1.0")
        return "\n".join(report_lines)

def _try_parse_json(value):
    """如果是 JSON 字符串则解析，否则返回原值"""
    if isinstance(value, str):
        try:
            return json.loads(value)
        except (json.JSONDecodeError, TypeError):
            pass
    return value


def _search_key_in_obj(obj, key: str, default=None):
    """在 dict 或对象中搜索 key，支持驼峰/下划线/全小写/全大写"""
    if isinstance(obj, dict):
        if key in obj:
            return obj[key]
        parts = key.split("_")
        camel = parts[0] + "".join(p.capitalize() for p in parts[1:])
        if camel in obj:

def extract_error_code(msg: str) -> str:
    m = re.search(r'\b(\d{9})\b', msg)
    if m:
        return m.group(1)
    m = re.search(r'code[:\s]*(\d+)', msg, re.IGNORECASE)
    return m.group(1) if m else "UNKNOWN"

def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    workflows = []

    while len(workflows) < limit:

            "space_id": space_id, "page_num": page, "page_size": page_size

        items = data.get("items", [])
        if not items:
            break
        workflows.extend(items)
        page += 1
        if len(items) < page_size:

def diagnose_workflow(workflow_id: str, token: str) -> Dict:

        if resp.status_code == 200:

def adjust_timeout(workflow_id: str, token: str) -> bool:
    """翻倍超时时间，上限300秒"""
    try:
        url = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url, token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            if node.get("type") in ("http", "code", "plugin"):
                for path in [("timeout",), ("config", "timeout")]:
                    cur = node
                    for k in path[:-1]:
                        cur = cur.get(k, {})
                    if isinstance(cur, dict) and path[-1] in cur:
                        old = cur[path[-1]]
                        new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                        if new != old:
                            cur[path[-1]] = new
                            modified = True
        if not modified:
            return True
        url_upd = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_upd, token, json_data=config)
        return True
    except Exception as e:
        print(f"[超时调整失败] {workflow_id}: {e}")
        return False

def execute_fix(workflow_id: str, wf_name: str, error_code: str, token: str) -> Dict:
    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy["action"]
    max_retries = strategy["retry"]
    desc = strategy["description"]

    if action in ("skip", "skip_manual"):
        return {"status": "skipped", "action": "none", "message": f"跳过: {desc}"}

    fix_map = {"publish": publish_workflow, "adjust_timeout": adjust_timeout, "fix_json_schema": fix_json_schema}
    fix_func = fix_map.get(action)
    if not fix_func:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}

    for attempt in range(max_retries + 1):
        try:
            if fix_func(workflow_id, token):
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功 ({desc})"}
                else:
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试{max_retries}次后失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": f"异常: {str(e)}"}
    return {"status": "failed", "action": action, "message": "未知错误"}

                    record["fix_result"] = {"status": "dry_run", "error_code": err_code}
                    skipped_count += 1
                    print("  ⏭️ 跳过修复")
                else:
                    fix_result = execute_fix(wf_id, wf_name, err_code, token)
                    record["fix_result"] = fix_result
                    if fix_result["status"] == "fixed":
                        fixed_count += 1
                        print(f"  ✅ {fix_result['message']}")
                    else:
                        failed_count += 1
                        print(f"  ❌ {fix_result['message']}")
            details.append(record)

        summary = {
            "total": total,
            "healthy": healthy_count,
            "fixed": fixed_count,
            "failed": failed_count,
            "skipped": skipped_count,
            "dry_run": dry_run,
            "auto_fix": auto_fix,
            "space_id": space_id,
            "scan_time": time.strftime("%Y-%m-%d %H:%M:%S")
        }
        full_report = {"summary": summary, "details": details}

        print(f"\n📊 扫描完成：总计 {total}，健康 {healthy_count}，修复 {fixed_count}，失败 {failed_count}，跳过 {skipped_count}")

        return {
            "total": total,
            "healthy": healthy_count,
            "fixed": fixed_count,
            "failed": failed_count,
            "report": json.dumps(full_report, ensure_ascii=False, indent=2)
        }

    except Exception as e:
        import traceback
        traceback.print_exc()
        return {
            "total": 0, "healthy": 0, "fixed": 0, "failed": 0,
            "report": json.dumps({"error": str(e), "type": type(e).__name__}, ensure_ascii=False)
        }

def _search(obj, key, default=None):
    if isinstance(obj, dict):
        if key in obj: return obj[key]
        camel = key[0].lower() + key.title().replace("_","")[1:] if "_" in key else None
        if camel and camel in obj: return obj[camel]
        if key.lower() in obj: return obj[key.lower()]
        if key.upper() in obj: return obj[key.upper()]
        return default
    for attr in (key, key.replace("_",""), key.lower(), key.upper()):
        if hasattr(obj, attr):
            v = getattr(obj, attr)
            if v is not None: return v
    return default

def api_call(method, url, token, json_data=None, params=None, retry=2):
    h = {"Authorization":f"Bearer {token}","Content-Type":"application/json"}
    for i in range(retry+1):
        try:
            r = requests.request(method, url, headers=h, json=json_data, params=params, timeout=30)
            r.raise_for_status()
            res = r.json()
            if res.get("code")!=0: raise Exception(f"API错误 {res.get('code')}: {res.get('msg')}")
            return res.get("data",{})
        except Exception as e:
            if i==retry: raise
            time.sleep(2)

def extract_code(msg): 
    m = re.search(r'\b(\d{9})\b', msg)
    if m: return m.group(1)
    m = re.search(r'code[:\s]*(\d+)', msg, re.I)
    return m.group(1) if m else "UNKNOWN"

# ==================== 终极参数提取（解决 space_id 为空的核心） ====================
def _try_parse_json(val):
    """安全解析 JSON 字符串，失败返回原值"""
    if isinstance(val, str):
        try:
            return json.loads(val)
        except (json.JSONDecodeError, TypeError):
            pass
    return val


def _search_key(obj, key: str, default=None):
    """
    在 dict 或任意对象中递归搜索 key，兼容：
    - 原始 key（space_id）
    - 小驼峰（spaceId）
    - 全小写（space_id）
    - 全大写（SPACE_ID）
    """
    if isinstance(obj, dict):
        # 直接匹配
        if key in obj:
            return obj[key]
        # 小驼峰
        parts = key.split('_')
        camel = parts[0] + ''.join(p.capitalize() for p in parts[1:])
        if camel in obj:
            return obj[camel]
        # 全小写 / 全大写
        if key.lower() in obj:
            return obj[key.lower()]
        if key.upper() in obj:
            return obj[key.upper()]
        return default

    # 对象属性
    for attr in (key, key.replace('_', ''), key.lower(), key.upper()):
        if hasattr(obj, attr):
            val = getattr(obj, attr)
            if val is not None:
                return val
    return default


def extract_params(args) -> Dict[str, Any]:
    """
    终极参数提取，彻底解决参数丢失问题。能处理：
    - args 为 dict，参数可能在顶层或嵌套在 'input' 键内（值为 dict 或 JSON 字符串）
    - args 为对象，参数可能直接在属性或 'input' 属性内（值为 dict 或 JSON 字符串）
    - args 本身是 JSON 字符串
    - 驼峰/下划线/全大写/全小写等各种命名风格
    - 保证提取后统一命名为下划线风格（如 space_id）
    """
    # 1. 如果是字符串，尝试解析为 JSON
    if isinstance(args, str):
        args = _try_parse_json(args)

    # 2. 尝试从 'input' 容器中提取（优先级更高，因为 Coze 经常把插件输入放在这里）
    input_container = None
    if isinstance(args, dict) and "input" in args:
        raw = args["input"]
        input_container = raw if isinstance(raw, dict) else _try_parse_json(raw)
    elif hasattr(args, "input"):
        raw = getattr(args, "input", None)
        if raw is not None:
            input_container = raw if isinstance(raw, dict) else _try_parse_json(raw)

    # 3. 从 args 本身提取基础值
    base = {}
    for std_key, alt_keys in [
        ("space_id", ["space_id", "spaceId"]),
        ("coze_api_token", ["coze_api_token", "cozeApiToken"]),
        ("auto_fix", ["auto_fix", "autoFix"]),
        ("dry_run", ["dry_run", "dryRun"]),
        ("max_workflows", ["max_workflows", "maxWorkflows"]),
        ("requirement", ["requirement", "req"]),
        ("action", ["action"]),
    ]:
        val = None
        for k in alt_keys:
            val = _search_key(args, k)
            if val is not None:
                break
        if val is not None:
            base[std_key] = val

    # 4. 用 'input' 容器中的值覆盖（优先级更高）
    if input_container and isinstance(input_container, dict):
        for std_key, alt_keys in [
            ("space_id", ["space_id", "spaceId"]),
            ("coze_api_token", ["coze_api_token", "cozeApiToken"]),
            ("auto_fix", ["auto_fix", "autoFix"]),
            ("dry_run", ["dry_run", "dryRun"]),
            ("max_workflows", ["max_workflows", "maxWorkflows"]),
            ("requirement", ["requirement", "req"]),
            ("action", ["action"]),
        ]:
            val = None
            for k in alt_keys:
                val = _search_key(input_container, k)
                if val is not None:
                    break
            if val is not None:
                base[std_key] = val

    # 5. 默认值
    base.setdefault("action", "fix")
    base.setdefault("auto_fix", True)
    base.setdefault("dry_run", False)
    base.setdefault("max_workflows", 100)

    # 6. 类型转换
    if isinstance(base.get("auto_fix"), str):
        base["auto_fix"] = base["auto_fix"].lower() in ("true", "1", "yes")
    if isinstance(base.get("dry_run"), str):
        base["dry_run"] = base["dry_run"].lower() in ("true", "1", "yes")
    try:
        base["max_workflows"] = int(base["max_workflows"])
    except (ValueError, TypeError):
        base["max_workflows"] = 100

    return base


# ==================== API 工具函数（带重试） ====================
def get_token(params: Dict) -> str:
    token = params.get("coze_api_token") or os.environ.get("COZE_API_TOKEN")
    if not token:
        raise ValueError("未找到 coze_api_token，请提供令牌或设置环境变量 COZE_API_TOKEN")
    return token


def api_request(method, url, token, json_data=None, query_params=None, retries=2):
    """带重试的 API 请求，自动处理鉴权和错误码"""
    headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}
    for attempt in range(retries + 1):
        try:
            resp = requests.request(method, url, headers=headers,
                                    json=json_data, params=query_params, timeout=30)
            resp.raise_for_status()
            result = resp.json()
            if result.get("code") != 0:
                raise Exception(f"API 错误 code={result.get('code')} msg={result.get('msg')}")
            return result.get("data", {})
        except Exception as e:
            if attempt == retries:
                raise
            time.sleep(2)


def extract_error_code(msg: str) -> str:
    """从错误消息中提取 9 位数字错误码"""
    m = re.search(r'\b(\d{9})\b', msg)
    if m:
        return m.group(1)
    m = re.search(r'code[:\s]*(\d+)', msg, re.IGNORECASE)
    return m.group(1) if m else "UNKNOWN"


# ==================== 工作流扫描与诊断 ====================
def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    """分页获取空间下所有工作流"""
    workflows, page, page_size = [], 1, 50
    while len(workflows) < limit:
        data = api_request("GET", LIST_API, token, query_params={
            "space_id": space_id, "page_num": page, "page_size": page_size
        })
        items = data.get("items", [])
        if not items:
            break
        workflows.extend(items)
        if len(items) < page_size:
            break
        page += 1
    return workflows[:limit]


def diagnose_workflow(workflow_id: str, token: str) -> Dict:
    """运行一次空跑诊断，返回健康状态或错误码"""
    try:
        resp = requests.post(RUN_API,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            json={"workflow_id": workflow_id, "parameters": {"__diagnostic__": True}},
            timeout=30)
        if resp.status_code == 200:
            result = resp.json()
            if result.get("code") == 0:
                return {"status": "healthy", "error_code": None, "error_msg": None}
            else:
                error_msg = result.get("msg", "")
                return {"status": "error", "error_code": extract_error_code(error_msg), "error_msg": error_msg}
        else:
            return {"status": "error", "error_code": f"HTTP_{resp.status_code}", "error_msg": resp.text}
    except Exception as e:
        return {"status": "error", "error_code": "NETWORK_EXCEPTION", "error_msg": str(e)}


# ==================== 修复操作 ====================
def publish_workflow(workflow_id: str, token: str) -> bool:
    """发布工作流"""
    try:
        api_request("POST", PUBLISH_API.format(workflow_id=workflow_id), token, json_data={})
        return True
    except Exception as e:
        print(f"[发布失败] {workflow_id}: {e}")
        return False


def adjust_timeout(workflow_id: str, token: str) -> bool:
    """将超时时间翻倍，上限 300 秒"""
    try:
        config = api_request("GET", GET_API.format(workflow_id=workflow_id), token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            if node.get("type") in ("http", "code", "plugin"):
                for path in [("timeout",), ("config", "timeout")]:
                    cur = node
                    for k in path[:-1]:
                        cur = cur.get(k, {})
                    if isinstance(cur, dict) and path[-1] in cur:
                        old = cur[path[-1]]
                        new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                        if new != old:
                            cur[path[-1]] = new
                            modified = True
        if modified:
            api_request("PUT", UPDATE_API.format(workflow_id=workflow_id), token, json_data=config)
        return True
    except Exception as e:
        print(f"[超时调整失败] {workflow_id}: {e}")
        return False


def fix_json_schema(workflow_id: str, token: str) -> bool:
    """重新序列化 JSON Schema，修复格式错误"""
    try:
        config = api_request("GET", GET_API.format(workflow_id=workflow_id), token)
        for node in config.get("nodes", []):
            if "output_schema" in node and isinstance(node["output_schema"], dict):
                node["output_schema"] = json.loads(json.dumps(node["output_schema"]))
        api_request("PUT", UPDATE_API.format(workflow_id=workflow_id), token, json_data=config)
        return True
    except Exception as e:
        print(f"[JSON修复失败] {workflow_id}: {e}")
        return False


# ==================== 修复执行与验证 ====================
def execute_fix(workflow_id: str, wf_name: str, error_code: str, token: str) -> Dict:
    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy["action"]
    max_retries = strategy["retry"]
    desc = strategy["desc"]

    if action in ("skip", "skip_manual"):
        return {"status": "skipped", "action": "none", "message": f"跳过: {desc}"}

    fix_map = {
        "publish": publish_workflow,
        "adjust_timeout": adjust_timeout,
        "fix_json_schema": fix_json_schema,
    }
    fix_func = fix_map.get(action)
    if not fix_func:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}

    for attempt in range(max_retries + 1):
        try:
            if fix_func(workflow_id, token):
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功 ({desc})"}
                else:
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试 {max_retries} 次后失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": str(e)}
    return {"status": "failed", "action": action, "message": "未知错误"}


# ==================== 工作流智能生成 ====================
def generate_workflow(requirement: str) -> Dict:
    """
    根据自然语言需求自动生成安全可用的工作流蓝图。
    自动添加输入清洗、敏感信息脱敏、异常处理分支、高风险操作审批、审计日志等安全节点。
    """
    lower = requirement.lower()
    has_http = any(w in lower for w in ["http", "api", "请求", "获取数据", "网页"])
    has_llm = any(w in lower for w in ["llm", "大模型", "回答", "生成", "聊天", "文本", "写作"])
    has_code = any(w in lower for w in ["代码", "计算", "处理", "转换", "逻辑"])
    has_db = any(w in lower for w in ["数据库", "存储", "写入", "查询", "记录", "保存"])
    has_approve = any(w in lower for w in ["审批", "确认", "高风险", "重要", "关键"])

    nodes = [{"id": "start", "type": "start", "config": {}}]
    edges = []

    # 输入清洗节点（强制执行）
    nodes.append({"id": "clean_input", "type": "code",
                  "config": {"code": "def clean(data): return data.strip()[:1000]"}})
    edges.append({"source": "start", "target": "clean_input"})

    prev = "clean_input"
    nid = 1

    for need, ntype, cfg in [
        (has_llm, "llm", {"prompt": "请根据输入内容进行智能处理"}),
        (has_http, "http", {"url": "https://api.example.com", "method": "GET"}),
        (has_code, "code", {"code": "def transform(data): return data"}),
        (has_db, "database", {"action": "insert"}),
        (has_approve, "approval", {"level": "high"}),
    ]:
        if need:
            nid += 1
            node = {"id": f"{ntype}_{nid}", "type": ntype, "config": cfg}
            # 每个节点都配置异常处理分支（错误隔离）
            node["on_error"] = {"target": "error_handler", "output": "error"}
            nodes.append(node)
            edges.append({"source": prev, "target": f"{ntype}_{nid}"})
            prev = f"{ntype}_{nid}"

    # 输出脱敏节点
    nodes.append({"id": "sanitize_out", "type": "code",
                  "config": {"code": "import re\ndef sanitize(x):\n    x = str(x)\n    x = re.sub(r'\\d{15,19}', '****', x)\n    x = re.sub(r'1[3-9]\\d{9}', '****', x)\n    return x"}})
    edges.append({"source": prev, "target": "sanitize_out"})

    # 结束节点
    nodes.append({"id": "end", "type": "end", "config": {}})
    edges.append({"source": "sanitize_out", "target": "end"})

    # 全局错误处理器
    nodes.append({"id": "error_handler", "type": "code",
                  "config": {"code": "def handle_error(e): return {'error': str(e)[:200], 'fallback': '请稍后重试'}"}})

    # 审计日志节点
    nodes.append({"id": "audit_log", "type": "code",
                  "config": {"code": "import time\ndef log(event):\n    print(f'AUDIT [{time.strftime(\"%Y-%m-%d %H:%M:%S\")}]: {event}')\n    return event"}})

    return {
        "name": f"自动生成工作流_{int(time.time())}",
        "description": f"根据需求自动生成: {requirement}",
        "nodes": nodes,
        "edges": edges,
        "security": {
            "input_cleaning": True,
            "output_sanitization": True,
            "error_isolation": True,
            "data_masking": True,
            "approval_on_high_risk": has_approve,
            "audit_logging": True
        }
    }

def handler(args):
    """
    Coze 插件入口函数。
    输入：args（由 Coze 平台传入，可能是 dict、对象或 JSON 字符串）
    输出：一个包含 total、healthy、fixed、failed 和 report 的字典。
    """
    try:
        # 1. 极致参数提取
        params = extract_params(args)
        action = params.get("action", "fix")

        # 2. 分流：生成 vs 修复
        if action == "generate":
            requirement = params.get("requirement", "创建一个安全可用的工作流")
            generated = generate_workflow(requirement)
            return {
                "total": 0, "healthy": 0, "fixed": 0, "failed": 0,
                "report": json.dumps({"action": "generate", "generated_workflow": generated},
                                     ensure_ascii=False, indent=2)
            }

        # 3. 修复模式
        space_id = params.get("space_id")
        if not space_id:
            return {
                "total": 0, "healthy": 0, "fixed": 0, "failed": 0,
                "report": json.dumps({"error": "space_id 不能为空，请在插件输入参数中提供 space_id 或 spaceId"},
                                     ensure_ascii=False)
            }

        token = get_token(params)
        auto_fix = params["auto_fix"]
        dry_run = params["dry_run"]
        max_workflows = params["max_workflows"]

        print(f"🚀 开始扫描空间 {space_id} (最多 {max_workflows} 个) | 自动修复: {auto_fix} | 仅诊断: {dry_run}")

        workflows = get_workflow_list(space_id, token, max_workflows)
        total = len(workflows)
        if total == 0:
            return {
                "total": 0, "healthy": 0, "fixed": 0, "failed": 0,
                "report": json.dumps({"message": "未找到任何工作流", "space_id": space_id}, ensure_ascii=False)
            }

        healthy_count = fixed_count = failed_count = skipped_count = 0
        details = []

        for idx, wf in enumerate(workflows, 1):
            wf_id = wf.get("workflow_id") or wf.get("id")
            wf_name = wf.get("name", f"workflow_{wf_id}")
            print(f"\n📋 [{idx}/{total}] {wf_name} ({wf_id})")

            diagnosis = diagnose_workflow(wf_id, token)
            record = {"workflow_id": wf_id, "workflow_name": wf_name, "diagnosis": diagnosis}

            if diagnosis["status"] == "healthy":
                healthy_count += 1
                record["fix_result"] = {"status": "healthy"}
                print("   ✅ 健康")
            else:
                err_code = diagnosis.get("error_code", "UNKNOWN")
                print(f"   ❌ 错误码: {err_code}")
                if dry_run or not auto_fix:
                    record["fix_result"] = {"status": "dry_run", "error_code": err_code}
                    skipped_count += 1
                    print("   ⏭️ 跳过修复")
                else:
                    fix_result = execute_fix(wf_id, wf_name, err_code, token)
                    record["fix_result"] = fix_result
                    if fix_result["status"] == "fixed":
                        fixed_count += 1
                        print(f"   ✅ {fix_result['message']}")
                    else:
                        failed_count += 1
                        print(f"   ❌ {fix_result['message']}")
            details.append(record)

        summary = {
            "total": total, "healthy": healthy_count, "fixed": fixed_count,
            "failed": failed_count, "skipped": skipped_count,
            "dry_run": dry_run, "auto_fix": auto_fix,
            "space_id": space_id, "scan_time": time.strftime("%Y-%m-%d %H:%M:%S")
        }
        full_report = {"summary": summary, "details": details}

        print(f"\n📊 扫描完成：总计 {total}，健康 {healthy_count}，已修复 {fixed_count}，失败 {failed_count}")

# -*- coding: utf-8 -*-
"""
Coze 插件：工作流自动修复器
功能：扫描空间内所有工作流 -> 诊断错误 -> 自动修复 -> 输出审计报告
安全特性：Token 从环境变量读取，支持 Dry-run 模式，所有写操作均有重试和验证
"""

import os
import json
import time
import re
from typing import Dict, List, Any, Optional
import requests

# ====================== 常量与配置 ======================
COZE_API_BASE = "https://api.coze.cn/v1"
WORKFLOW_LIST_API = f"{COZE_API_BASE}/workflows"
WORKFLOW_RUN_API = f"{COZE_API_BASE}/workflow/run"
WORKFLOW_PUBLISH_API = f"{COZE_API_BASE}/workflows/{{workflow_id}}/publish"
WORKFLOW_GET_API = f"{COZE_API_BASE}/workflows/{{workflow_id}}"
WORKFLOW_UPDATE_API = f"{COZE_API_BASE}/workflows/{{workflow_id}}"

# 错误码 -> 修复策略映射
ERROR_STRATEGY = {
    "720702011": {  # 工作流未发布
        "action": "publish",
        "retry": 1,
        "description": "工作流已保存但未发布"
    },
    "777777776": {  # 节点超时
        "action": "adjust_timeout",
        "retry": 2,
        "description": "节点执行超时，需要调大超时阈值"
    },
    "720712023": {  # 节点输出解析失败（JSON格式错误）
        "action": "fix_json_schema",
        "retry": 2,
        "description": "节点输出格式与预期不匹配"
    },
    "720702004": {  # 工作流不存在
        "action": "skip",
        "retry": 0,
        "description": "工作流可能已被删除，需人工检查"
    },
    "720702002": {  # 缺少必填参数
        "action": "skip_manual",
        "retry": 0,
        "description": "缺少API调用参数，需人工补充"
    },
    "720700801": {  # 数据库错误
        "action": "skip_manual",
        "retry": 0,
        "description": "数据库异常，需人工排查"
    },
    "UNKNOWN": {
        "action": "skip",
        "retry": 0,
        "description": "未知错误，建议人工介入"
    }

# ====================== 工具函数 ======================
def get_token(input_args: Dict) -> str:
    """安全获取 API Token（优先从输入参数，其次从环境变量）"""
    token = input_args.get("coze_api_token") or os.environ.get("COZE_API_TOKEN")
    if not token:
        raise ValueError("未找到 COZE_API_TOKEN，请在环境变量或输入参数中设置")
    return token


def extract_error_code(error_msg: str) -> str:
    """从错误消息中提取 Coze 错误码（9位数字）"""
    match = re.search(r'\b(\d{9})\b', error_msg)
    if match:
        return match.group(1)
    # 也支持常见的 HTTP 风格错误码
    match2 = re.search(r'code[:\s]*(\d+)', error_msg, re.IGNORECASE)
    if match2:
        return match2.group(1)
    return "UNKNOWN"


def api_request(method: str, url: str, token: str, json_data: Dict = None, params: Dict = None) -> Dict:
    """统一的 API 请求封装，自动处理 Coze 标准响应"""
    headers = {
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json"
    }
    resp = requests.request(method, url, headers=headers, json=json_data, params=params)
    resp.raise_for_status()
    result = resp.json()
    # Coze API 通常返回 code:0 表示成功
    if result.get("code") != 0:
        raise Exception(f"API 错误: code={result.get('code')}, msg={result.get('msg')}")
    return result.get("data", {})


def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    """分页获取空间下所有工作流"""
    workflows = []
    page = 1
    page_size = 50
    while len(workflows) < limit:
        params = {
            "space_id": space_id,
            "page_num": page,
            "page_size": page_size
        }
        data = api_request("GET", WORKFLOW_LIST_API, token, params=params)
        items = data.get("items", [])
        if not items:
            break
        workflows.extend(items)
        if len(workflows) >= limit or len(items) < page_size:
            break
        page += 1
    return workflows[:limit]


def diagnose_workflow(workflow_id: str, token: str) -> Dict:
    """
    诊断工作流：使用空参数执行一次，捕获错误码
    返回: {"status": "healthy"|"error", "error_code": str, "error_msg": str}
    """
    try:
        resp = requests.post(
            WORKFLOW_RUN_API,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            json={"workflow_id": workflow_id, "parameters": {"__diagnostic__": True}}
        )
        if resp.status_code == 200:
            result = resp.json()
            if result.get("code") == 0:
                return {"status": "healthy", "error_code": None, "error_msg": None}
            else:
                error_msg = result.get("msg", "")
                error_code = extract_error_code(error_msg)
                return {"status": "error", "error_code": error_code, "error_msg": error_msg}
        else:
            return {"status": "error", "error_code": f"HTTP_{resp.status_code}", "error_msg": resp.text}
    except Exception as e:
        return {"status": "error", "error_code": "EXCEPTION", "error_msg": str(e)}


def publish_workflow(workflow_id: str, token: str) -> bool:
    """发布工作流"""
    url = WORKFLOW_PUBLISH_API.format(workflow_id=workflow_id)
    try:
        api_request("POST", url, token, json_data={})
        return True
    except Exception as e:
        print(f"[ERROR] 发布失败: {e}")
        return False


def adjust_timeout(workflow_id: str, token: str) -> bool:
    """
    调整工作流中所有 HTTP/代码节点的超时时间
    将超时值翻倍，最大不超过 300 秒
    """
    try:
        # 获取当前工作流配置
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            node_type = node.get("type", "")
            if node_type in ("http", "code", "plugin"):
                # 查找 timeout 字段
                if "timeout" in node:
                    old = node["timeout"]
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["timeout"] = new
                        modified = True
                        print(f"[INFO] 节点 {node.get('name')} 超时 {old}s -> {new}s")
                elif "config" in node and "timeout" in node["config"]:
                    old = node["config"]["timeout"]
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["config"]["timeout"] = new
                        modified = True
                        print(f"[INFO] 节点 {node.get('name')} 超时 {old}s -> {new}s")
        if not modified:
            print("[INFO] 未发现需要调整超时的节点")
            return True
        # 保存更新后的配置
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] 调整超时失败: {e}")
        return False


def fix_json_schema(workflow_id: str, token: str) -> bool:
    """
    修复节点输出 JSON 格式问题：
    通过重新保存工作流配置，触发 Coze 后端的格式校验和自动修正。
    """
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        # 简单触发更新：对每个节点的输出 schema 做一次重新序列化
        # （Coze 后端会自动修正部分不规范的 schema）
        if "nodes" in config:
            for node in config["nodes"]:
                if "output_schema" in node and isinstance(node["output_schema"], dict):
                    # 重新序列化以触发格式校验
                    node["output_schema"] = json.loads(json.dumps(node["output_schema"]))
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] JSON 格式修复失败: {e}")
        return False


def execute_fix(workflow_id: str, workflow_name: str, error_code: str, token: str) -> Dict:
    """根据错误码执行对应的修复操作，支持重试"""
    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy["action"]
    max_retries = strategy["retry"]
    
    if action == "skip":
        return {"status": "skipped", "action": "none", "message": strategy["description"]}
    if action == "skip_manual":
        return {"status": "manual_required", "action": "none", "message": strategy["description"]}
    
    # 需要自动修复的动作
    fix_func = None
    if action == "publish":
        fix_func = publish_workflow
    elif action == "adjust_timeout":
        fix_func = adjust_timeout
    elif action == "fix_json_schema":
        fix_func = fix_json_schema
    else:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}
    
    # 重试逻辑
    for attempt in range(max_retries + 1):
        try:
            success = fix_func(workflow_id, token)
            if success:
                # 修复后等待生效再验证
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功 (尝试 {attempt+1})"}
                else:
                    # 修复后仍报错，可能是其他问题
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试 {max_retries} 次后仍失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": f"异常: {str(e)}"}
    return {"status": "failed", "action": action, "message": "未知错误"}


# ====================== 插件主入口 ======================
def handler(args):
    """
    Coze IDE 插件入口
    """
    # 1. 解析输入
    space_id = args.input.get("space_id", "").strip()
    if not space_id:
        return {
            "total": 0,
            "healthy": 0,
            "fixed": 0,
            "failed": 0,
            "report": json.dumps({"error": "space_id 不能为空"}, ensure_ascii=False)
        }
    
    auto_fix = args.input.get("auto_fix", True)
    dry_run = args.input.get("dry_run", False)
    max_workflows = args.input.get("max_workflows", 100)
    
    try:
        token = get_token(args.input)
    except ValueError as e:
        return {
            "total": 0,
            "healthy": 0,
            "fixed": 0,
            "failed": 0,
            "report": json.dumps({"error": str(e)}, ensure_ascii=False)
        }
    
    # 2. 获取工作流列表
    print(f"[Phase 1] 扫描空间 {space_id} 的工作流...")
    workflows = get_workflow_list(space_id, token, max_workflows)
    total = len(workflows)
    if total == 0:
        return {
            "total": 0,
            "healthy": 0,
            "fixed": 0,
            "failed": 0,
            "report": json.dumps({"message": "未找到任何工作流"}, ensure_ascii=False)
        }
    
    # 3. 诊断与修复
    healthy_count = 0
    fixed_count = 0
    failed_count = 0
    skipped_count = 0
    report_details = []
    
    for idx, wf in enumerate(workflows, 1):
        wf_id = wf.get("id")
        wf_name = wf.get("name", f"workflow_{wf_id}")
        print(f"[Phase 2] [{idx}/{total}] 诊断: {wf_name}")
        
        diagnosis = diagnose_workflow(wf_id, token)
        record = {
            "workflow_id": wf_id,
            "workflow_name": wf_name,
            "diagnosis": diagnosis
        }
        
        if diagnosis["status"] == "healthy":
            healthy_count += 1
            record["fix_result"] = {"status": "healthy", "action": "none"}
            print(f"  ✓ 健康")
        else:
            error_code = diagnosis.get("error_code", "UNKNOWN")
            error_msg = diagnosis.get("error_msg", "")
            print(f"  ✗ 错误码: {error_code} - {error_msg[:100]}")
            
            if dry_run or not auto_fix:
                record["fix_result"] = {"status": "dry_run", "action": "none", "error_code": error_code}
                skipped_count += 1
            else:
                fix_result = execute_fix(wf_id, wf_name, error_code, token)
                record["fix_result"] = fix_result
                if fix_result["status"] == "fixed":
                    fixed_count += 1
                    print(f"  ✅ 修复成功")
                elif fix_result["status"] in ("failed", "partial"):
                    failed_count += 1
                    print(f"  ❌ 修复失败: {fix_result.get('message')}")
                else:
                    skipped_count += 1
                    print(f"  ⏭️ 跳过: {fix_result.get('message')}")
        
        report_details.append(record)
    
    # 4. 生成审计报告
    summary = {
        "total": total,
        "healthy": healthy_count,
        "fixed": fixed_count,
        "failed": failed_count,
        "skipped": skipped_count,
        "dry_run": dry_run,
        "auto_fix": auto_fix,
        "space_id": space_id,
        "scan_time": time.strftime("%Y-%m-%d %H:%M:%S")
    }
    
    full_report = {
        "summary": summary,
        "details": report_details
    }
    
    print("\n[Phase 5] 审计报告摘要:")
    print(json.dumps(summary, indent=2, ensure_ascii=False))
    
    return {
        "total": total,
        "healthy": healthy_count,
        "fixed": fixed_count,
        "failed": failed_count,
        "report": json.dumps(full_report, ensure_ascii=False, indent=2)
    }

def get_token(input_args: Dict) -> str:
    """安全获取 API Token（优先从输入参数，其次从环境变量）"""
    token = input_args.get("coze_api_token") or os.environ.get("COZE_API_TOKEN")
    if not token:
        raise ValueError("未找到 COZE_API_TOKEN，请在环境变量或输入参数中设置")
    return token


def is_workflow_id(target: str) -> bool:
    """判断 target 是 workflow_id 还是 space_id"""
    return len(target) > 0 and target.isdigit() and len(target) > 10


def extract_error_code(error_msg: str) -> str:
    """从错误消息中提取 Coze 错误码（9位数字）"""
    match = re.search(r'\b(\d{9})\b', error_msg)
    if match:
        return match.group(1)
    match2 = re.search(r'code[:\s]*(\d+)', error_msg, re.IGNORECASE)
    if match2:
        return match2.group(1)
    return "UNKNOWN"


def api_request(method: str, url: str, token: str, json_data: Dict = None, params: Dict = None) -> Dict:
    """统一的 API 请求封装，自动处理 Coze 标准响应"""
    headers = {
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json"
    }
    resp = requests.request(method, url, headers=headers, json=json_data, params=params)
    resp.raise_for_status()
    result = resp.json()
    if "code" in result and result["code"] != 0:
        code = result.get("code")
        msg = result.get("msg", "")
        raise Exception(f"API Error: Code {code}, Msg: {msg}")
    if "data" in result:
        return result["data"]
    return result


def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    """分页获取空间下所有工作流"""
    workflows = []
    page = 1
    page_size = 50
    while True:
        params = {
            "space_id": space_id,
            "page_num": page,
            "page_size": page_size
        }
        try:
            data = api_request("GET", WORKFLOW_LIST_API, token, params=params)
            items = data.get("items", [])
            if not items:
                break
            workflows.extend(items)
            if len(workflows) >= limit or len(items) < page_size:
                break
            page += 1
        except Exception as e:
            print(f"[ERROR] 获取工作流列表失败: {e}")
            break
    return workflows[:limit]


def diagnose_workflow(workflow_id: str, token: str) -> Dict:
    """
    诊断工作流：使用空参数执行一次，捕获错误码
    返回: {"status": "healthy"|"error", "error_code": str, "error_msg": str, "code": int}
    """
    try:
        resp = requests.post(
            WORKFLOW_RUN_API,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            json={"workflow_id": workflow_id, "parameters": {"__diagnostic__": True}}
        )
        if resp.status_code == 200:
            result = resp.json()
            code = result.get("code", 0)
            if code == 0:
                return {"status": "healthy", "error_code": None, "error_msg": None, "code": 0}
            else:
                error_msg = result.get("msg", "")
                error_code = extract_error_code(error_msg)
                return {"status": "error", "error_code": error_code, "error_msg": error_msg, "code": code}
        else:
            return {"status": "error", "error_code": f"HTTP_{resp.status_code}", "error_msg": resp.text, "code": resp.status_code}
    except Exception as e:
        return {"status": "error", "error_code": "EXCEPTION", "error_msg": str(e), "code": -1}


def publish_workflow(workflow_id: str, token: str) -> bool:
    """发布工作流"""
    url = WORKFLOW_PUBLISH_API.format(workflow_id=workflow_id)
    try:
        api_request("POST", url, token, json_data={})
        return True
    except Exception as e:
        print(f"[ERROR] 发布失败: {e}")
        return False


def adjust_timeout(workflow_id: str, token: str) -> bool:
    """
    调整工作流中所有 HTTP/代码节点的超时时间
    将超时值翻倍，最大不超过 300 秒
    """
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            node_type = node.get("type", "")
            if node_type in ["http", "code", "plugin"]:
                if "timeout" in node:
                    old = node["timeout"]
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["timeout"] = new
                        modified = True
                        print(f"[INFO] 节点 {node.get('name')} 超时 {old}s -> {new}s")
                elif "config" in node and "timeout" in node["config"]:
                    old = node["config"]["timeout"]
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["config"]["timeout"] = new
                        modified = True
                        print(f"[INFO] 节点 {node.get('name')} 超时 {old}s -> {new}s")
        if not modified:
            print("[INFO] 未发现需要调整超时的节点")
            return True
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] 调整超时失败: {e}")
        return False


def fix_json_schema(workflow_id: str, token: str) -> bool:
    """
    修复节点输出 JSON 格式问题：
    通过重新保存工作流配置，触发 Coze 后端的格式校验和自动修正。
    """
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        if "nodes" in config:
            for node in config["nodes"]:
                if "output_schema" in node and isinstance(node["output_schema"], dict):
                    # 重新序列化以触发格式校验
                    node["output_schema"] = json.loads(json.dumps(node["output_schema"]))
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] JSON 格式修复失败: {e}")
        return False


def execute_fix(workflow_id: str, workflow_name: str, error_code: str, token: str, dry_run: bool = False) -> Dict:
    """根据错误码执行对应的修复操作，支持重试"""
    if dry_run:
        return {"status": "dry_run", "action": "none", "message": "预演模式，未执行修复"}

    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy.get("action", "skip")
    max_retries = strategy.get("retry", 0)
    
    if action == "skip":
        return {"status": "skipped", "action": "none", "message": strategy["description"]}
    if action == "skip_manual":
        return {"status": "manual_required", "action": "none", "message": strategy["description"]}
    
    fix_func = None
    if action == "publish":
        fix_func = publish_workflow
    elif action == "adjust_timeout":
        fix_func = adjust_timeout
    elif action == "fix_json_schema":
        fix_func = fix_json_schema
    else:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}
    
    for attempt in range(max_retries + 1):
        try:
            success = fix_func(workflow_id, token)
            if success:
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功 (尝试 {attempt+1})"}
                else:
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试 {max_retries} 次后仍失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": f"异常: {str(e)}"}
    return {"status": "failed", "action": action, "message": "未知错误"}


def process_single_workflow(workflow_id: str, token: str, auto_fix: bool, dry_run: bool) -> Dict:
    """处理单个工作流"""
    print(f"[Processing] 工作流 ID: {workflow_id}")
    diagnosis = diagnose_workflow(workflow_id, token)
    record = {
        "workflow_id": workflow_id,
        "diagnosis": diagnosis
    }
    if diagnosis["status"] == "healthy":
        record["fix_result"] = {"status": "healthy", "action": "none"}
        print(f"  ✓ 健康")
        return record
    error_code = diagnosis.get("error_code", "UNKNOWN")
    error_msg = diagnosis.get("error_msg", "")
    print(f"  ✗ 错误码: {error_code} - {error_msg[:100]}")
    if dry_run or not auto_fix:
        record["fix_result"] = {"status": "dry_run", "action": "none", "error_code": error_code}
    else:
        fix_result = execute_fix(workflow_id, "", error_code, token, dry_run)
        record["fix_result"] = fix_result
        if fix_result["status"] == "fixed":
            print(f"  ✅ 修复成功")
        elif fix_result["status"] in ("failed", "partial"):
            print(f"  ❌ 修复失败: {fix_result.get('message')}")
        else:
            print(f"  ⏭️ 跳过: {fix_result.get('message')}")
    return record


def process_space(space_id: str, token: str, auto_fix: bool, dry_run: bool, max_workflows: int) -> Dict:

def repair_workflows(**kwargs) -> Dict:
    """
    修复工作流的入口函数。安全、自动化地修复工作流错误。
    
    参数:
        target: str - 可以是工作流 ID 或空间 ID
        auto_fix: bool - 是否自动修复，默认 True
        dry_run: bool - 是否仅诊断，默认 False
        max_workflows: int - 最大处理数量，仅对空间有效，默认 100
    
    返回:
        Dict - 包含 total, fixed, report 的字典

def extract_id_from_url(url_or_id: str) -> str:
    """
    从 Coze 工作流 URL 中提取 workflow_id 或 space_id。
    支持格式：
      - 纯数字 ID
      - https://www.coze.cn/work_flow?workflow_id=7386766004514209844&space_id=...
      - https://www.coze.cn/space/7382283479335403547/workflow/...
    若提取不到，返回原字符串。
    """
    # 如果已经是纯数字且长度大于10，直接返回
    if url_or_id.isdigit() and len(url_or_id) >= 10:
        return url_or_id
    
    # 尝试匹配 workflow_id 参数
    match = re.search(r'workflow_id[=: ](\d+)', url_or_id)
    if match:
        return match.group(1)
    
    # 尝试匹配 space_id 参数
    match = re.search(r'space_id[=: ](\d+)', url_or_id)
    if match:
        return match.group(1)
    
    # 尝试匹配路径中的数字段（如 /space/123456/...）
    match = re.search(r'/space/(\d+)', url_or_id)
    if match:
        return match.group(1)
    
    # 最后返回清理后的原始字符串（去掉首尾空格）
    return url_or_id.strip()

def get_token_from_env_or_args(args_obj: Any) -> str:

    token = os.environ.get("COZE_API_TOKEN")

    # 如果 args 是字典

def api_request(method: str, url: str, token: str, json_data: Dict = None, params: Dict = None) -> Dict:
    """统一的 API 请求封装，自动处理 Coze 标准响应"""
    headers = {
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json"
    }
    resp = requests.request(method, url, headers=headers, json=json_data, params=params)
    resp.raise_for_status()
    result = resp.json()
    if "code" in result and result["code"] != 0:
        code = result.get("code")
        msg = result.get("msg", "")
        raise Exception(f"API Error: Code {code}, Msg: {msg}")
    if "data" in result:
        return result["data"]
    return result


def get_workflow_list(space_id: str, token: str, limit: int = 100) -> List[Dict]:
    """分页获取空间下所有工作流"""
    workflows = []
    page = 1
    page_size = 50
    while True:
        params = {
            "space_id": space_id,
            "page_num": page,
            "page_size": page_size
        }
        try:
            data = api_request("GET", WORKFLOW_LIST_API, token, params=params)
            items = data.get("items", [])
            if not items:
                break
            workflows.extend(items)
            if len(workflows) >= limit or len(items) < page_size:
                break
            page += 1
        except Exception as e:
            print(f"[ERROR] 获取工作流列表失败: {e}")
            break
    return workflows[:limit]


def diagnose_workflow(workflow_id: str, token: str) -> Dict:
    """诊断工作流：使用空参数执行一次，捕获错误码"""
    try:
        resp = requests.post(
            WORKFLOW_RUN_API,
            headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            json={"workflow_id": workflow_id, "parameters": {"__diagnostic__": True}}
        )
        if resp.status_code == 200:
            result = resp.json()
            code = result.get("code", 0)
            if code == 0:
                return {"status": "healthy", "error_code": None, "error_msg": None, "code": 0}
            else:
                error_msg = result.get("msg", "")
                error_code = re.search(r'\b(\d{9})\b', error_msg)
                error_code = error_code.group(1) if error_code else "UNKNOWN"
                return {"status": "error", "error_code": error_code, "error_msg": error_msg, "code": code}
        else:
            return {"status": "error", "error_code": f"HTTP_{resp.status_code}", "error_msg": resp.text, "code": resp.status_code}
    except Exception as e:
        return {"status": "error", "error_code": "EXCEPTION", "error_msg": str(e), "code": -1}

def adjust_timeout(workflow_id: str, token: str) -> bool:
    """调整工作流中所有 HTTP/代码节点的超时时间，翻倍且不超过300秒"""
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        nodes = config.get("nodes", [])
        modified = False
        for node in nodes:
            node_type = node.get("type", "")
            if node_type in ["http", "code", "plugin"]:
                if "timeout" in node:
                    old = node["timeout"]
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["timeout"] = new
                        modified = True
                        print(f"[INFO] 节点 {node.get('name')} 超时 {old}s -> {new}s")
                elif "config" in node and "timeout" in node["config"]:
                    old = node["config"]["timeout"]
                    new = min(old * 2 if isinstance(old, (int, float)) else 60, 300)
                    if new != old:
                        node["config"]["timeout"] = new
                        modified = True
                        print(f"[INFO] 节点 {node.get('name')} 超时 {old}s -> {new}s")
        if not modified:
            print("[INFO] 未发现需要调整超时的节点")
            return True
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] 调整超时失败: {e}")
        return False


def fix_json_schema(workflow_id: str, token: str) -> bool:
    """修复节点输出 JSON 格式问题：重新保存以触发后端校验"""
    try:
        url_get = WORKFLOW_GET_API.format(workflow_id=workflow_id)
        config = api_request("GET", url_get, token)
        if "nodes" in config:
            for node in config["nodes"]:
                if "output_schema" in node and isinstance(node["output_schema"], dict):
                    node["output_schema"] = json.loads(json.dumps(node["output_schema"]))
        url_update = WORKFLOW_UPDATE_API.format(workflow_id=workflow_id)
        api_request("PUT", url_update, token, json_data=config)
        return True
    except Exception as e:
        print(f"[ERROR] JSON 格式修复失败: {e}")
        return False


def execute_fix(workflow_id: str, workflow_name: str, error_code: str, token: str, dry_run: bool = False) -> Dict:
    """根据错误码执行修复，支持重试"""
    if dry_run:
        return {"status": "dry_run", "action": "none", "message": "预演模式，未执行修复"}

    strategy = ERROR_STRATEGY.get(error_code, ERROR_STRATEGY["UNKNOWN"])
    action = strategy.get("action", "skip")
    max_retries = strategy.get("retry", 0)
    
    if action == "skip":
        return {"status": "skipped", "action": "none", "message": strategy["description"]}
    if action == "skip_manual":
        return {"status": "manual_required", "action": "none", "message": strategy["description"]}
    
    fix_func = None
    if action == "publish":
        fix_func = publish_workflow
    elif action == "adjust_timeout":
        fix_func = adjust_timeout
    elif action == "fix_json_schema":
        fix_func = fix_json_schema
    else:
        return {"status": "failed", "action": action, "message": f"未知修复动作: {action}"}
    
    for attempt in range(max_retries + 1):
        try:
            success = fix_func(workflow_id, token)
            if success:
                time.sleep(2)
                verify = diagnose_workflow(workflow_id, token)
                if verify["status"] == "healthy":
                    return {"status": "fixed", "action": action, "message": f"修复成功 (尝试 {attempt+1})"}
                else:
                    return {"status": "partial", "action": action, "message": f"修复后仍有错误: {verify.get('error_code')}"}
            else:
                if attempt < max_retries:
                    time.sleep(3)
                    continue
                return {"status": "failed", "action": action, "message": f"重试 {max_retries} 次后仍失败"}
        except Exception as e:
            if attempt < max_retries:
                time.sleep(3)
            else:
                return {"status": "failed", "action": action, "message": f"异常: {str(e)}"}
    return {"status": "failed", "action": action, "message": "未知错误"}


def process_single_workflow(workflow_id: str, token: str, auto_fix: bool, dry_run: bool) -> Dict:
    """处理单个工作流"""
    print(f"[Processing] 工作流 ID: {workflow_id}")

        "workflow_id": workflow_id,

    }
    if diagnosis["status"] == "healthy":
        record["fix_result"] = {"status": "healthy", "action": "none"}
        print(f"  ✓ 健康")

    error_code = diagnosis.get("error_code", "UNKNOWN")
    error_msg = diagnosis.get("error_msg", "")
    print(f"  ✗ 错误码: {error_code} - {error_msg[:100]}")
    if dry_run or not auto_fix:
        record["fix_result"] = {"status": "dry_run", "action": "none", "error_code": error_code}
    else:
        fix_result = execute_fix(workflow_id, "", error_code, token, dry_run)
        record["fix_result"] = fix_result
        if fix_result["status"] == "fixed":
            print(f"  ✅ 修复成功")
        elif fix_result["status"] in ("failed", "partial"):
            print(f"  ❌ 修复失败: {fix_result.get('message')}")
        else:
            print(f"  ⏭️ 跳过: {fix_result.get('message')}")
    return record


def process_space(space_id: str, token: str, auto_fix: bool, dry_run: bool, max_workflows: int) -> Dict:
    """处理整个空间的工作流"""
    print(f"[Phase 1] 扫描空间 {space_id} 的工作流...")
    workflows = get_workflow_list(space_id, token, max_workflows)
    total = len(workflows)
    if total == 0:
        return {"total": 0, "healthy": 0, "fixed": 0, "failed": 0, "skipped": 0, "report_details": []}
    fixed_count = 0
    failed_count = 0
    skipped_count = 0
    healthy_count = 0
    report_details = []
    for idx, wf in enumerate(workflows, 1):
        wf_id = wf.get("id")
        wf_name = wf.get("name", f"workflow_{wf_id}")
        print(f"[Phase 2] [{idx}/{total}] 诊断: {wf_name}")
        diagnosis = diagnose_workflow(wf_id, token)
        record = {
            "workflow_id": wf_id,
            "workflow_name": wf_name,
            "diagnosis": diagnosis
        }
        if diagnosis["status"] == "healthy":
            healthy_count += 1
            record["fix_result"] = {"status": "healthy", "action": "none"}
            print(f"  ✓ 健康")
        else:
            error_code = diagnosis.get("error_code", "UNKNOWN")
            error_msg = diagnosis.get("error_msg", "")
            print(f"  ✗ 错误码: {error_code} - {error_msg[:100]}")
            if dry_run or not auto_fix:
                record["fix_result"] = {"status": "dry_run", "action": "none", "error_code": error_code}
                skipped_count += 1
            else:
                fix_result = execute_fix(wf_id, wf_name, error_code, token, dry_run)
                record["fix_result"] = fix_result
                if fix_result["status"] == "fixed":
                    fixed_count += 1
                    print(f"  ✅ 修复成功")
                elif fix_result["status"] in ("failed", "partial"):
                    failed_count += 1
                    print(f"  ❌ 修复失败: {fix_result.get('message')}")
                else:
                    skipped_count += 1
                    print(f"  ⏭️ 跳过: {fix_result.get('message')}")
        report_details.append(record)
    return {
        "total": total,
        "healthy": healthy_count,
        "fixed": fixed_count,
        "failed": failed_count,
        "skipped": skipped_count,
        "report_details": report_details
    }


# ====================== 核心修复函数 ======================
def repair_workflows(target: str, auto_fix: bool = True, dry_run: bool = False, max_workflows: int = 100, coze_api_token: str = None) -> Dict:
    """
    修复工作流的入口函数。安全、自动化地修复工作流错误。
    参数:
        target: str - 可以是工作流 ID、空间 ID 或 Coze URL
        auto_fix: bool - 是否自动修复，默认 True
        dry_run: bool - 是否仅诊断，默认 False
        max_workflows: int - 最大处理数量，仅对空间有效，默认 100
        coze_api_token: str - API Token（可选，优先使用环境变量）
    返回:
        Dict - 包含 total, fixed, report 的字典
    """
    # 1. 从输入中提取真实的 target（支持URL）
    raw_target = target.strip()
    real_target = extract_id_from_url(raw_target)
    print(f"[Target] 输入: {raw_target} -> 解析为: {real_target}")
    
    # 2. 获取 Token
    try:
        token = coze_api_token or os.environ.get("COZE_API_TOKEN")
        if not token:
            raise ValueError("未找到 COZE_API_TOKEN，请在环境变量或输入参数中设置")
    except ValueError as e:
        return {
            "total": 0,
            "fixed": 0,
            "report": json.dumps({"error": str(e)}, ensure_ascii=False)
        }
    
    # 3. 判断是单个工作流还是空间
    if is_workflow_id(real_target):
        print(f"[Mode] 单工作流模式，ID: {real_target}")
        record = process_single_workflow(real_target, token, auto_fix, dry_run)
        fix_status = record.get("fix_result", {}).get("status", "unknown")
        total = 1
        fixed = 1 if fix_status == "fixed" else 0
        summary = {
            "total": total,
            "fixed": fixed,
            "dry_run": dry_run,
            "auto_fix": auto_fix,
            "scan_time": time.strftime("%Y-%m-%d %H:%M:%S")
        }
        full_report = {
            "summary": summary,
            "details": [record]
        }
        return {
            "total": total,
            "fixed": fixed,
            "report": json.dumps(full_report, ensure_ascii=False, indent=2)
        }
    else:
        print(f"[Mode] 空间模式，ID: {real_target}")
        result = process_space(real_target, token, auto_fix, dry_run, max_workflows)
        summary = {
            "total": result["total"],
            "healthy": result["healthy"],
            "fixed": result["fixed"],
            "failed": result["failed"],
            "skipped": result["skipped"],
            "dry_run": dry_run,
            "auto_fix": auto_fix,
            "space_id": real_target,
            "scan_time": time.strftime("%Y-%m-%d %H:%M:%S")
        }
        full_report = {
            "summary": summary,
            "details": result["report_details"]
        }
        return {
            "total": result["total"],
            "fixed": result["fixed"],
            "report": json.dumps(full_report, ensure_ascii=False, indent=2)
        }


# ====================== Coze 标准入口（兼容对象和字典）======================
def handler(args):
    """
    Coze IDE 标准插件入口函数。
    兼容两种调用方式：
      - args 是字典（包含 input 键）
      - args 是对象（具有 input 属性）
    """
    try:
        # 兼容处理：获取输入参数字典
        if isinstance(args, dict):
            input_args = args.get("input", {})
        elif hasattr(args, "input"):
            # 对象情况，input 可能是字典或对象
            if isinstance(args.input, dict):
                input_args = args.input
            elif hasattr(args.input, "__dict__"):
                input_args = vars(args.input)
            else:
                input_args = {}
        else:
            # 如果 args 本身就是参数对象（没有 input 包装）
            if hasattr(args, "__dict__"):
                input_args = vars(args)
            else:
                input_args = {}
        
        # 调用核心修复函数
        result = repair_workflows(**input_args)
        return result
    except Exception as e:
        # 捕获所有异常，返回友好错误信息
        error_report = {
            "total": 0,
            "fixed": 0,
            "report": json.dumps({"error": str(e), "trace": "请检查输入参数和网络环境"}, ensure_ascii=False)
        }
        return error_report
